    34→function getNodeType(obj: THREE.Object3D): HierarchyNode['type'] {
    35→  if (obj.userData.isLightHelper || obj.userData.isProxy || obj.userData.isGrid) return 'helper';
    36→  if (obj instanceof THREE.Light) return 'light';
    37→  if (obj instanceof THREE.Camera) return 'camera';
    38→  if (obj instanceof THREE.Mesh) return 'mesh';
    39→  if (obj instanceof THREE.Group && obj.children.length > 0) return 'group';
    40→  return 'other';
    41→}
    42→
    43→function isCollection(obj: THREE.Object3D): boolean {
    44→  return obj instanceof THREE.Group &&
    45→    !obj.userData.isGrid &&
    46→    !obj.userData.isProxy &&
    47→    obj.userData._isCollection === true;
    48→}
    49→
    50→function buildTree(scene: THREE.Scene): HierarchyNode[] {
    51→  const nodes: HierarchyNode[] = [];
    52→  for (const child of scene.children) {
    53→    if (child.userData.isGrid || child.userData.isProxy) continue;
    54→    nodes.push(buildNode(child));
    55→  }
    56→  return nodes;
    57→}
    58→
    59→function buildNode(obj: THREE.Object3D): HierarchyNode {
    60→  const children: HierarchyNode[] = [];
    61→  for (const child of obj.children) {
    62→    if (child.userData.isGrid || child.userData.isProxy) continue;
    63→    children.push(buildNode(child));
    64→  }
    65→  return {
    66→    name: obj.name || `${obj.type}_${obj.id}`,
    67→    type: getNodeType(obj),
    68→    object: obj,
    69→    children,
    70→    isCollection: isCollection(obj),
    71→  };
    72→}
    73→
    74→function hasDescendantMatch(node: HierarchyNode, query: string): boolean {
    75→  const q = query.toLowerCase();
    76→  if (node.name.toLowerCase().includes(q)) return true;
    77→  return node.children.some((c) => hasDescendantMatch(c, q));
    78→}

    80→function findObjectByUuid(scene: THREE.Scene, uuid: string): THREE.Object3D | null {
    81→  let result: THREE.Object3D | null = null;
    82→  scene.traverse((child) => {
    83→    if (child.uuid === uuid) result = child;
    84→  });
    85→  return result;
    86→}
    87→
    88→function getAllDescendantIds(obj: THREE.Object3D): string[] {
    89→  const ids: string[] = [obj.uuid];
    90→  obj.traverse((child) => {
    91→    if (child.uuid !== obj.uuid) ids.push(child.uuid);
    92→  });
    93→  return ids;
    94→}
    95→
    96→function getAllCollections(scene: THREE.Scene): THREE.Group[] {
    97→  const groups: THREE.Group[] = [];
    98→  scene.traverse((child) => {
    99→    if (isCollection(child)) groups.push(child as THREE.Group);
   100→  });
   101→  return groups;
   102→}
   103→
   104→function countCollections(scene: THREE.Scene): number {
   105→  let count = 0;
   106→  for (const child of scene.children) {
   107→    if (isCollection(child)) count++;
   108→  }
   109→  scene.traverse((child) => {
   110→    if (child !== scene && isCollection(child)) count++;
   111→  });
   112→  return count;
   113→}
