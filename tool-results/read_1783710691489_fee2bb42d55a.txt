    33→function getNodeType(obj: THREE.Object3D): HierarchyNode['type'] {
    34→  if (obj.userData.isLightHelper || obj.userData.isProxy || obj.userData.isGrid) return 'helper';
    35→  if (obj instanceof THREE.Light) return 'light';
    36→  if (obj instanceof THREE.Camera) return 'camera';
    37→  if (obj instanceof THREE.Mesh) return 'mesh';
    38→  if (obj instanceof THREE.Group && obj.children.length > 0) return 'group';
    39→  return 'other';
    40→}
    41→
    42→function buildTree(scene: THREE.Scene): HierarchyNode[] {
    43→  const nodes: HierarchyNode[] = [];
    44→  for (const child of scene.children) {
    45→    if (child.userData.isGrid || child.userData.isProxy) continue;
    46→    nodes.push(buildNode(child));
    47→  }
    48→  return nodes;
    49→}
    50→
    51→function buildNode(obj: THREE.Object3D): HierarchyNode {
    52→  const children: HierarchyNode[] = [];
    53→  for (const child of obj.children) {
    54→    if (child.userData.isGrid || child.userData.isProxy) continue;
    55→    children.push(buildNode(child));
    56→  }
    57→  return {
    58→    name: obj.name || `${obj.type}_${obj.id}`,
    59→    type: getNodeType(obj),
    60→    object: obj,
    61→    children,
    62→  };
    63→}
    64→
    65→function hasDescendantMatch(node: HierarchyNode, query: string): boolean {
    66→  const q = query.toLowerCase();
    67→  if (node.name.toLowerCase().includes(q)) return true;
    68→  return node.children.some((c) => hasDescendantMatch(c, q));
    69→}
    70→
    71→function findObjectByUuid(scene: THREE.Scene, uuid: string): THREE.Object3D | null {
    72→  let result: THREE.Object3D | null = null;
    73→  scene.traverse((child) => {
    74→    if (child.uuid === uuid) result = child;
    75→  });
    76→  return result;
    77→}
    78→
    79→function getAllDescendantIds(obj: THREE.Object3D): string[] {
    80→  const ids: string[] = [obj.uuid];
    81→  obj.traverse((child) => {
    82→    if (child.uuid !== obj.uuid) ids.push(child.uuid);
    83→  });
    84→  return ids;
    85→}
    86→
    87→function getAllGroupsInScene(scene: THREE.Scene): THREE.Group[] {
    88→  const groups: THREE.Group[] = [];
    89→  scene.traverse((child) => {
    90→    if (child instanceof THREE.Group && child.children.length > 0 && !child.userData.isGrid && !child.userData.isProxy) {
    91→      groups.push(child);
    92→    }
    93→  });
    94→  return groups;
    95→}
    96→
    97→function countCollections(scene: THREE.Scene): number {
    98→  let count = 0;
    99→  for (const child of scene.children) {
   100→    if (child instanceof THREE.Group && !child.userData.isGrid && !child.userData.isProxy) {
   101→      count++;
   102→    }
   103→  }
   104→  return count;
   105→}
