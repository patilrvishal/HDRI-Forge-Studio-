   133→  /** Remove existing ground objects from the scene and dispose their resources. */
   134→  private _disposeGround(): void {
   135→    if (this.groundOverlay) {
   136→      this.scene.remove(this.groundOverlay);
   137→      this.groundOverlay.geometry.dispose();
   138→      (this.groundOverlay.material as THREE.Material).dispose();
   139→      this.groundOverlay = null;
   140→    }
   141→    if (this.ground) {
   142→      this.scene.remove(this.ground);
   143→      this.ground.geometry.dispose();
   144→      if (Array.isArray(this.ground.material)) {
   145→        this.ground.material.forEach((m) => m.dispose());
   146→      } else {
   147→        this.ground.material.dispose();
   148→      }
   149→      this.ground = null;
   150→    }
   151→    if (this._floorCubeCamera) {
   152→      this._floorCubeCamera.dispose();
   153→      this._floorCubeCamera = null;
   154→    }
   155→    if (this._floorCubeRT) {
   156→      this._floorCubeRT.dispose();
   157→      this._floorCubeRT = null;
   158→    }
   159→    this._floorMaterial = null;
   160→  }
   161→
   162→  /**
   163→   * Rebuild the ground plane according to the given settings.
   164→   * Always uses MeshStandardMaterial (PBR) so roughness/metalness work correctly.
   165→   * - reflections=true  → CubeCamera real-time reflections via envMap
   166→   * - reflections=false → Scene environment only (or none)
   167→   */
   168→  updateGround(settings?: GroundSettings | null): void {
   169→    if (!settings) return;
   170→    // Merge with defaults so old scene files missing new fields don't crash
   171→    const merged: GroundSettings = {
   172→      visible: true,
   173→      reflections: true,
   174→      reflectionSharpness: 0.85,
   175→      color: '#111118',
   176→      roughness: 0.15,
   177→      metalness: 0.95,
   178→      fadeRadius: 8.0,
   179→      ...settings,
   180→    };
   181→    this._groundSettings = merged;
   182→    this._disposeGround();
   183→
   184→    if (!merged.visible) return;
   185→
   186→    const GROUND_SIZE = 40;
   187→    const groundGeo = new THREE.PlaneGeometry(GROUND_SIZE, GROUND_SIZE);
   188→    const color = new THREE.Color(merged.color);
   189→
   190→    // ── PBR ground material (always MeshStandardMaterial) ──
   191→    const groundMat = new THREE.MeshStandardMaterial({
   192→      color: color.getHex(),
   193→      metalness: merged.metalness,
   194→      roughness: merged.roughness,
   195→      envMapIntensity: 1.0,
   196→      side: THREE.FrontSide,
   197→    });
   198→    this._floorMaterial = groundMat;
   199→
   200→    if (merged.reflections) {
   201→      // ── CubeCamera for real-time planar reflections ──
   202→      const dpr = Math.min(window.devicePixelRatio, 2);
   203→      const cubeRTSize = Math.max(128, Math.round(512 * dpr));
   204→
   205→      this._floorCubeRT = new THREE.WebGLCubeRenderTarget(cubeRTSize, {
   206→        generateMipmaps: true,
   207→        minFilter: THREE.LinearMipmapLinearFilter,
   208→        magFilter: THREE.LinearFilter,
   209→      });
   210→
   211→      this._floorCubeCamera = new THREE.CubeCamera(0.1, 100, this._floorCubeRT);
   212→      this._floorCubeCamera.position.set(0, 0.01, 0); // slightly above floor
   213→      this.scene.add(this._floorCubeCamera);
   214→
   215→      // Use CubeCamera texture as envMap; roughness controls blur via mip levels
   216→      groundMat.envMap = this._floorCubeRT.texture;
   217→      groundMat.envMapIntensity = merged.reflectionSharpness;
   218→    } else {
   219→      // No real-time reflections — use scene environment map if available
   220→      if (this.scene.environment) {
   221→        groundMat.envMap = this.scene.environment;
   222→      }
   223→      groundMat.envMapIntensity = 0.5;
   224→    }
   225→
   226→    this.ground = new THREE.Mesh(groundGeo, groundMat);
   227→    this.ground.rotation.x = -Math.PI / 2;
   228→    this.ground.position.y = -0.005;
   229→    this.ground.receiveShadow = true;
   230→    this.ground.name = '__floor__';
   231→    this.scene.add(this.ground);
   232→
   233→    // ── Fade overlay: fades ground edges into background ──
   234→    if (merged.fadeRadius > 0) {
   235→      const overlayGeo = new THREE.PlaneGeometry(GROUND_SIZE, GROUND_SIZE);
   236→      const overlayMat = new THREE.ShaderMaterial({
   237→        transparent: true,
   238→        depthWrite: false,
   239→        uniforms: {
   240→          uFadeRadius: { value: merged.fadeRadius },
   241→          uGroundSize: { value: GROUND_SIZE / 2 },
   242→          uBgColor: { value: new THREE.Color('#0d0d1a') },
   243→        },
   244→        vertexShader: /* glsl */ `
   245→          varying vec2 vWorldPos;
   246→          void main() {
   247→            vec4 worldPos = modelMatrix * vec4(position, 1.0);
   248→            vWorldPos = worldPos.xz;
   249→            gl_Position = projectionMatrix * viewMatrix * worldPos;
   250→          }
   251→        `,
   252→        fragmentShader: /* glsl */ `
   253→          uniform float uFadeRadius;
   254→          uniform float uGroundSize;
   255→          uniform vec3 uBgColor;
   256→          varying vec2 vWorldPos;
   257→          void main() {
   258→            float dist = length(vWorldPos);
   259→            float fade = 1.0 - smoothstep(uFadeRadius, uGroundSize, dist);
   260→            gl_FragColor = vec4(uBgColor, 1.0 - fade);
   261→          }
   262→        `,
   263→      });
   264→      this.groundOverlay = new THREE.Mesh(overlayGeo, overlayMat);
   265→      this.groundOverlay.rotation.x = -Math.PI / 2;
   266→      this.groundOverlay.position.y = -0.003;
   267→      this.scene.add(this.groundOverlay);
   268→    }
   269→  }

   298→  animateCameraTo(
   299→    position: [number, number, number],
   300→    target: [number, number, number],
   301→    duration: number = 600
   302→  ): void {
   303→    const startPos = this.camera.position.clone();
   304→    const startTarget = this.controls.target.clone();
   305→    const endPos = new THREE.Vector3(...position);
   306→    const endTarget = new THREE.Vector3(...target);
   307→    const startTime = performance.now();
   308→
   309→    const animate = () => {
   310→      const elapsed = performance.now() - startTime;
   311→      const t = Math.min(elapsed / duration, 1);
   312→      const ease = t < 0.5 ? 4 * t * t * t : 1 - Math.pow(-2 * t + 2, 3) / 2;
   313→
   314→      this.camera.position.lerpVectors(startPos, endPos, ease);
   315→      this.controls.target.lerpVectors(startTarget, endTarget, ease);
   316→      this.controls.update();
   317→
   318→      if (t < 1) {
   319→        requestAnimationFrame(animate);
   320→      }
   321→    };
   322→    requestAnimationFrame(animate);
   323→  }

   325→  setOnFrame(cb: (delta: number) => void): void {
   326→    this._onFrame = cb;
   327→  }
   328→
   329→  startRenderLoop(): void {
   330→    const loop = () => {
   331→      this._animationId = requestAnimationFrame(loop);
   332→      const delta = this._clock.getDelta();
   333→
   334→      // Turntable rotation
   335→      if (this._turntableActive) {
   336→        const model = this._findModel();
   337→        if (model) {
   338→          model.rotation.y += delta * this._turntableSpeed * 0.5;
   339→        }
   340→      }
   341→
   342→      // Update CubeCamera for PBR floor reflections (hide floor to avoid self-reflection)
   343→      if (this._floorCubeCamera && this.ground && this._groundSettings?.reflections) {
   344→        this.ground.visible = false;
   345→        this._floorCubeCamera.update(this.renderer, this.scene);
   346→        this.ground.visible = true;
   347→      }
   348→
   349→      if (this._onFrame) this._onFrame(delta);
   350→      this.controls.update();
   351→      this.renderer.render(this.scene, this.camera);
   352→    };
   353→    loop();
   354→  }
   355→
   356→  _findModel(): THREE.Object3D | null {
   357→    for (const child of this.scene.children) {
   358→      if (
   359→        child.type === 'Group' &&
   360→        !(child instanceof THREE.GridHelper) &&
   361→        !child.userData.isHelper &&
   362→        child !== this.ground &&
   363→        child !== this._floorCubeCamera
   364→      ) {
   365→        return child;
   366→      }
   367→    }
   368→    return null;
   369→  }
   370→
   371→  stopRenderLoop(): void {
   372→    cancelAnimationFrame(this._animationId);
   373→  }

   422→// ── Vignette shader (custom) ────────────────────────────────────────────────
   423→const VignetteShader = {
   424→  uniforms: {
   425→    tDiffuse: { value: null as THREE.Texture | null },
   426→    intensity: { value: 0.4 },
   427→    offset: { value: 1.1 },
   428→    darkness: { value: 1.2 },
   429→  },
   430→  vertexShader: /* glsl */ `
   431→    varying vec2 vUv;
   432→    void main() {
   433→      vUv = uv;
   434→      gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
   435→    }
   436→  `,
   437→  fragmentShader: /* glsl */ `
   438→    uniform sampler2D tDiffuse;
   439→    uniform float intensity;
   440→    uniform float offset;
   441→    uniform float darkness;
   442→    varying vec2 vUv;
   443→    void main() {
   444→      vec4 texel = texture2D(tDiffuse, vUv);
   445→      vec2 uv = (vUv - vec2(0.5)) * vec2(offset);
   446→      float vig = clamp(1.0 - dot(uv, uv), 0.0, 1.0);
   447→      float f = mix(1.0, vig, intensity * darkness);
   448→      gl_FragColor = vec4(texel.rgb * f, texel.a);
   449→    }
   450→  `,
   451→};
   452→
   453→// ── Color Grading shader (brightness / contrast / saturation) ───────────────
   454→const ColorGradingShader = {
   455→  uniforms: {
   456→    tDiffuse: { value: null as THREE.Texture | null },
   457→    brightness: { value: 0.0 },
   458→    contrast: { value: 0.0 },
   459→    saturation: { value: 0.0 },
   460→  },
   461→  vertexShader: /* glsl */ `
   462→    varying vec2 vUv;
   463→    void main() {
   464→      vUv = uv;
   465→      gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
   466→    }
   467→  `,
   468→  fragmentShader: /* glsl */ `
   469→    uniform sampler2D tDiffuse;
   470→    uniform float brightness;
   471→    uniform float contrast;
   472→    uniform float saturation;
   473→    varying vec2 vUv;
   474→
   475→    void main() {
   476→      vec4 texel = texture2D(tDiffuse, vUv);
   477→      vec3 c = texel.rgb;
   478→
   479→      // Brightness
   480→      c += brightness * 0.5;
   481→
   482→      // Contrast
   483→      c = (c - 0.5) * (1.0 + contrast) + 0.5;
   484→
   485→      // Saturation
   486→      float lum = dot(c, vec3(0.2126, 0.7152, 0.0722));
   487→      c = mix(vec3(lum), c, 1.0 + saturation);
   488→
   489→      gl_FragColor = vec4(clamp(c, 0.0, 1.0), texel.a);
   490→    }
   491→  `,
   492→};

   494→// ── Pipeline settings interface ─────────────────────────────────────────────
   495→export interface PipelineConfig {
   496→  bloom: { enabled: boolean; intensity: number; threshold: number; radius: number };
   497→  ao: { enabled: boolean; radius: number; intensity: number };
   498→  vignette: { enabled: boolean; intensity: number };
   499→  colorGrading: { enabled: boolean; brightness: number; contrast: number; saturation: number };
   500→  antialiasing: 'none' | 'fxaa' | 'smaa' | 'taa';
   501→  shadowQuality: 'none' | 'low' | 'medium' | 'high';
   502→  quality: 'low' | 'medium' | 'high' | 'ultra';
   503→  tonemapping: 'aces' | 'reinhard' | 'linear';
   504→  exposure: number;
   505→}
   506→
   507→export class RenderPipeline {
   508→  private _sm: SceneManager;
   509→  private _composer: EffectComposer | null = null;
   510→  private _renderPass: RenderPass | null = null;
   511→  private _bloomPass: UnrealBloomPass | null = null;
   512→  private _smaaPass: SMAAPass | null = null;
   513→  private _fxaaPass: ShaderPass | null = null;
   514→  private _ssaoPass: SSAOPass | null = null;
   515→  private _vignettePass: ShaderPass | null = null;
   516→  private _colorGradingPass: ShaderPass | null = null;
   517→  private _config: PipelineConfig;
   518→  private _needsRebuild = false;
   519→
   520→  constructor(sceneManager: SceneManager) {
   521→    this._sm = sceneManager;
   522→    this._config = {
   523→      bloom: { enabled: true, intensity: 0.3, threshold: 0.8, radius: 0.5 },
   524→      ao: { enabled: false, radius: 0.5, intensity: 0.5 },
   525→      vignette: { enabled: false, intensity: 0.4 },
   526→      colorGrading: { enabled: false, brightness: 0, contrast: 0, saturation: 0 },
   527→      antialiasing: 'smaa',
   528→      shadowQuality: 'high',
   529→      quality: 'high',
   530→      tonemapping: 'aces',
   531→      exposure: 1.0,
   532→    };
   533→  }
   534→
   535→  /** Build (or rebuild) the EffectComposer from scratch. */
   536→  build(): void {
   537→    this.dispose();
   538→    const r = this._sm.renderer;
   539→    const w = r.domElement.width || r.getSize(new THREE.Vector2()).x || 2;
   540→    const h = r.domElement.height || r.getSize(new THREE.Vector2()).y || 2;
   541→
   542→    this._composer = new EffectComposer(r);
   543→    this._renderPass = new RenderPass(this._sm.scene, this._sm.camera);
   544→    this._composer.addPass(this._renderPass);
   545→
   546→    // Bloom
   547→    if (this._config.bloom.enabled) {
   548→      this._bloomPass = new UnrealBloomPass(
   549→        new THREE.Vector2(w / 2, h / 2),
   550→        this._config.bloom.intensity,
   551→        this._config.bloom.radius,
   552→        this._config.bloom.threshold,
   553→      );
   554→      this._composer.addPass(this._bloomPass);
   555→    }
   556→
   557→    // Anti-aliasing
   558→    if (this._config.antialiasing === 'smaa') {
   559→      // @ts-expect-error — SMAAPass types are incomplete in @types/three
   560→      this._smaaPass = new SMAAPass(w, h);
   561→      this._composer.addPass(this._smaaPass);
   562→    } else if (this._config.antialiasing === 'fxaa') {
   563→      const fxaaMat = new THREE.ShaderMaterial({ ...FXAAShader, uniforms: { ...FXAAShader.uniforms, resolution: { value: new THREE.Vector2(1 / w, 1 / h) } } });
   564→      this._fxaaPass = new ShaderPass(fxaaMat);
   565→      this._composer.addPass(this._fxaaPass);
   566→    }
   567→    // 'taa' — handled via renderer settings + jitter; we keep SMAA as fallback
   568→    // 'none' — no AA pass
   569→
   570→    // SSAO
   571→    if (this._config.ao.enabled) {
   572→      this._ssaoPass = new SSAOPass(
   573→        this._sm.scene,
   574→        this._sm.camera,
   575→        w,
   576→        h,
   577→      );
   578→      this._applyAOParams(this._config.ao.radius, this._config.ao.intensity);
   579→      this._ssaoPass.output = SSAOPass.OUTPUT.Default;
   580→      this._composer.addPass(this._ssaoPass);
   581→    }
   582→
   583→    // Vignette
   584→    if (this._config.vignette.enabled) {
   585→      const vMat = new THREE.ShaderMaterial(VignetteShader);
   586→      vMat.uniforms['intensity'].value = this._config.vignette.intensity;
   587→      this._vignettePass = new ShaderPass(vMat);
   588→      this._composer.addPass(this._vignettePass);
   589→    }
   590→
   591→    // Color grading
   592→    if (this._config.colorGrading.enabled) {
   593→      const cgMat = new THREE.ShaderMaterial(ColorGradingShader);
   594→      cgMat.uniforms['brightness'].value = this._config.colorGrading.brightness;
   595→      cgMat.uniforms['contrast'].value = this._config.colorGrading.contrast;
   596→      cgMat.uniforms['saturation'].value = this._config.colorGrading.saturation;
   597→      this._colorGradingPass = new ShaderPass(cgMat);
   598→      this._composer.addPass(this._colorGradingPass);
   599→    }
   600→
   601→    // Apply tone mapping & exposure to the renderer
   602→    this._applyToneMapping(this._config.tonemapping);
   603→    this._sm.renderer.toneMappingExposure = this._config.exposure;
   604→
   605→    // Shadow quality
   606→    this._applyShadowQuality(this._config.shadowQuality);
   607→  }
   608→
   609→  /** Render one frame through the composer (or fallback direct render). */
   610→  render(): void {
   611→    if (this._composer) {
   612→      this._composer.render();
   613→    } else {
   614→      this._sm.renderer.render(this._sm.scene, this._sm.camera);
   615→    }
   616→  }
   617→
   618→  /** Resize the composer and internal passes. */
   619→  resize(): void {
   620→    if (!this._composer) return;
   621→    const r = this._sm.renderer;
   622→    const size = r.getSize(new THREE.Vector2());
   623→    const w = size.x || 2;
   624→    const h = size.y || 2;
   625→    this._composer.setSize(w, h);
   626→
   627→    if (this._bloomPass) {
   628→      this._bloomPass.resolution.set(w / 2, h / 2);
   629→    }
   630→    if (this._ssaoPass) {
   631→      this._ssaoPass.setSize(w, h);
   632→    }
   633→    if (this._smaaPass) {
   634→      this._smaaPass.setSize(w, h);
   635→    }
   636→    if (this._fxaaPass) {
   637→      const mat = this._fxaaPass.material as THREE.ShaderMaterial;
   638→      const res = mat.uniforms['resolution'];
   639→      if (res && res.value instanceof THREE.Vector2) {
   640→        res.value.set(1 / w, 1 / h);
   641→      }
   642→    }
   643→  }
   644→
   645→  /** Update the full pipeline config. Rebuilds composer if structural changes detected. */
   646→  updateConfig(cfg: Partial<PipelineConfig>): void {
   647→    const prev = { ...this._config };
   648→    this._config = { ...this._config, ...cfg };
   649→
   650→    const structuralChange =
   651→      cfg.bloom?.enabled !== undefined && cfg.bloom.enabled !== prev.bloom.enabled ||
   652→      cfg.antialiasing !== undefined && cfg.antialiasing !== prev.antialiasing ||
   653→      cfg.vignette?.enabled !== undefined && cfg.vignette.enabled !== prev.vignette.enabled ||
   654→      cfg.ao?.enabled !== undefined && cfg.ao.enabled !== prev.ao.enabled ||
   655→      cfg.colorGrading?.enabled !== undefined && cfg.colorGrading.enabled !== prev.colorGrading.enabled;
   656→
   657→    if (structuralChange) {
   658→      this.build();
   659→      return;
   660→    }
   661→
   662→    // In-place parameter updates (no rebuild needed)
   663→    if (this._bloomPass && cfg.bloom) {
   664→      this._bloomPass.strength = cfg.bloom.intensity ?? this._bloomPass.strength;
   665→      this._bloomPass.radius = cfg.bloom.radius ?? this._bloomPass.radius;
   666→      this._bloomPass.threshold = cfg.bloom.threshold ?? this._bloomPass.threshold;
   667→    }
   668→
   669→    if (this._ssaoPass && cfg.ao) {
   670→      this._applyAOParams(
   671→        cfg.ao.radius ?? this._config.ao.radius,
   672→        cfg.ao.intensity ?? this._config.ao.intensity,
   673→      );
   674→    }
   675→
   676→    if (this._vignettePass && cfg.vignette) {
   677→      const mat = this._vignettePass.material as THREE.ShaderMaterial;
   678→      mat.uniforms['intensity'].value = cfg.vignette.intensity ?? 0.4;
   679→    }
   680→
   681→    if (this._colorGradingPass && cfg.colorGrading) {
   682→      const mat = this._colorGradingPass.material as THREE.ShaderMaterial;
   683→      if (cfg.colorGrading.brightness !== undefined) mat.uniforms['brightness'].value = cfg.colorGrading.brightness;
   684→      if (cfg.colorGrading.contrast !== undefined) mat.uniforms['contrast'].value = cfg.colorGrading.contrast;
   685→      if (cfg.colorGrading.saturation !== undefined) mat.uniforms['saturation'].value = cfg.colorGrading.saturation;
   686→    }
   687→
   688→    if (cfg.tonemapping !== undefined) {
   689→      this._applyToneMapping(cfg.tonemapping);
   690→    }
   691→    if (cfg.exposure !== undefined) {
   692→      this._sm.renderer.toneMappingExposure = cfg.exposure;
   693→    }
   694→    if (cfg.shadowQuality !== undefined) {
   695→      this._applyShadowQuality(cfg.shadowQuality);
   696→    }
   697→    if (cfg.quality !== undefined) {
   698→      this._applyQuality(cfg.quality);
   699→    }
   700→  }
   701→
   702→  // ── Convenience setters (called from Viewport sync) ───────────────────────
   703→
   704→  setEngine(engine: 'pbr' | 'pathtracer'): void {
   705→    if (engine === 'pbr') {
   706→      this._applyToneMapping(this._config.tonemapping);
   707→    } else {
   708→      this._sm.renderer.toneMapping = THREE.NoToneMapping;

   710→  }
   711→
   712→  setToneMapping(mapping: 'aces' | 'reinhard' | 'linear'): void {

   714→    this._applyToneMapping(mapping);
   715→  }
   716→

   719→      bloom: { enabled, intensity, threshold, radius },
   720→    });
   721→  }

   723→  setAO(enabled: boolean, radius: number, intensity: number): void {

   725→      ao: { enabled, radius, intensity },
   726→    });
   727→  }
   728→
   729→  setAntialiasing(mode: 'none' | 'fxaa' | 'smaa' | 'taa'): void {

   731→  }
   732→
   733→  setShadowQuality(quality: 'none' | 'low' | 'medium' | 'high'): void {

   735→    this._applyShadowQuality(quality);
   736→  }
   737→

   741→  }

   743→  setExposure(exposure: number): void {

   745→    this._sm.renderer.toneMappingExposure = exposure;
   746→  }

   748→  setVignette(enabled: boolean, intensity: number): void {

   756→  /** Get the EffectComposer (for screenshot / export via composer). */
   757→  getComposer(): EffectComposer | null {

   759→  }

   765→  }
   766→
   767→  /** Dispose all passes and the composer. */
   768→  dispose(): void {
   769→    if (this._composer) {
   770→      const passes = this._composer.passes.slice().reverse();
   771→      for (const pass of passes) {
   772→        if ('dispose' in pass && typeof (pass as { dispose: () => void }).dispose === 'function') {
   773→          (pass as { dispose: () => void }).dispose();
   774→        }
   775→      }
   776→      this._composer.dispose();
   777→    }
   778→    this._composer = null;
   779→    this._renderPass = null;
   780→    this._bloomPass = null;
   781→    this._ssaoPass = null;
   782→    this._smaaPass = null;
   783→    this._fxaaPass = null;
   784→    this._vignettePass = null;
   785→    this._colorGradingPass = null;
   786→  }

   788→  // ── Private helpers ───────────────────────────────────────────────────────
   789→
   790→  /** Apply SSAO parameters — radius controls kernel spread, intensity controls maxDistance scaling. */
   791→  private _applyAOParams(radius: number, intensity: number): void {
   792→    if (!this._ssaoPass) return;
   793→    this._ssaoPass.kernelRadius = radius;
   794→    this._ssaoPass.minDistance = 0.005;
   795→    // Scale maxDistance with intensity: higher intensity → AO visible at greater depth range
   796→    // Tuned for car-scale scenes (objects ~2–5m across)
   797→    this._ssaoPass.maxDistance = 0.05 + intensity * 0.5;
   798→  }
   799→
   800→  private _applyToneMapping(mapping: 'aces' | 'reinhard' | 'linear'): void {
   801→    const r = this._sm.renderer;
   802→    switch (mapping) {
   803→      case 'aces':
   804→        r.toneMapping = THREE.ACESFilmicToneMapping;
   805→        break;
   806→      case 'reinhard':
   807→        r.toneMapping = THREE.ReinhardToneMapping;
   808→        break;
   809→      case 'linear':
   810→        r.toneMapping = THREE.LinearToneMapping;
   811→        break;
   812→    }
   813→    r.toneMappingExposure = this._config.exposure;
   814→  }
   815→
   816→  private _applyShadowQuality(quality: string): void {
   817→    const r = this._sm.renderer;
   818→    switch (quality) {
   819→      case 'none':
   820→        r.shadowMap.enabled = false;
   821→        break;
   822→      case 'low':
   823→        r.shadowMap.enabled = true;
   824→        r.shadowMap.type = THREE.BasicShadowMap;
   825→        break;
   826→      case 'medium':
   827→        r.shadowMap.enabled = true;
   828→        r.shadowMap.type = THREE.PCFShadowMap;
   829→        break;
   830→      case 'high':
   831→      default:
   832→        r.shadowMap.enabled = true;
   833→        r.shadowMap.type = THREE.PCFSoftShadowMap;
   834→        break;
   835→    }
   836→
   837→    // Update shadow map sizes on all lights
   838→    this._sm.scene.traverse((child) => {
   839→      const light = child as THREE.DirectionalLight | THREE.SpotLight | THREE.PointLight;
   840→      if (light.isLight && light.shadow) {
   841→        const shadow = light.shadow;
   842→        switch (quality) {
   843→          case 'low':
   844→            shadow.mapSize.set(512, 512);
   845→            break;
   846→          case 'medium':
   847→            shadow.mapSize.set(1024, 1024);
   848→            break;
   849→          case 'high':
   850→          default:
   851→            shadow.mapSize.set(2048, 2048);
   852→            break;
   853→        }
   854→        if (shadow.map) {
   855→          shadow.map.dispose();
   856→          shadow.map = null;
   857→        }
   858→      }
   859→    });
   860→  }
   861→
   862→  private _applyQuality(quality: 'low' | 'medium' | 'high' | 'ultra'): void {
   863→    const r = this._sm.renderer;
   864→    switch (quality) {
   865→      case 'low':
   866→        r.setPixelRatio(1);
   867→        break;
   868→      case 'medium':
   869→        r.setPixelRatio(Math.min(window.devicePixelRatio, 1.5));
   870→        break;
   871→      case 'high':
   872→        r.setPixelRatio(Math.min(window.devicePixelRatio, 2));
   873→        break;
   874→      case 'ultra':
   875→        r.setPixelRatio(window.devicePixelRatio);
   876→        break;
   877→    }
   878→    this._sm.resize();
   879→    if (this._composer) {
   880→      this.resize();
   881→    }
   882→  }
   883→}

   885→interface LightSyncEntry {
   886→  id: string;
   887→  type: string;
   888→  color: string;
   889→  brightness: number;
   890→  opacity: number;
   891→  visible: boolean;
   892→  solo: boolean;
   893→  falloff: string;
   894→  gearVisible: boolean;
   895→  transform: {
   896→    position: { x: number; y: number; z: number };
   897→    spherical: { lat: number; lng: number; radius: number; height: number };
   898→    rotation: { x: number; y: number; z: number; enabled: boolean };
   899→  };
   900→  // Spotlight-specific
   901→  spotAngle?: number;
   902→  spotPenumbra?: number;
   903→  spotDecay?: number;
   904→  // Area light specific
   905→  areaWidth?: number;
   906→  areaHeight?: number;
   907→}
   908→
   909→interface LightObjectEntry {
   910→  object: THREE.Object3D;
   911→  lightType: string;
   912→  target?: THREE.Object3D;
   913→}
   914→
   915→export class LightManager {
   916→  private _scene: THREE.Scene;
   917→  private _entries: Map<string, LightObjectEntry> = new Map();
   918→  private _helpers: Map<string, THREE.Object3D> = new Map();
   919→  private _types: Map<string, string> = new Map();
   920→
   921→  constructor(scene: THREE.Scene) {
   922→    this._scene = scene;
   923→    ensureRectAreaLib();
   924→  }
   925→
   926→  syncLights(lights: LightSyncEntry[]): void {
   927→    const activeIds = new Set(lights.map((l) => l.id));
   928→    const hasSolo = lights.some((l) => l.solo);
   929→
   930→    // Remove stale lights
   931→    for (const [id] of this._entries) {
   932→      if (!activeIds.has(id)) {
   933→        this._removeLight(id);
   934→      }
   935→    }
   936→
   937→    // Create or update each light
   938→    for (const ld of lights) {
   939→      const isEffectiveSolo = hasSolo ? ld.solo : true;
   940→      const shouldShow = ld.visible && isEffectiveSolo;
   941→
   942→      const existing = this._entries.get(ld.id);
   943→      const currentType = this._types.get(ld.id);
   944→
   945→      // If type changed, recreate the light
   946→      if (existing && currentType !== ld.type) {
   947→        this._removeLight(ld.id);
   948→      }
   949→
   950→      if (!this._entries.has(ld.id)) {
   951→        this._createLight(ld);
   952→      }
   953→
   954→      this._updateLight(ld, shouldShow);
   955→    }
   956→  }
   957→
   958→  private _createLight(ld: LightSyncEntry): void {
   959→    ensureRectAreaLib();
   960→    const lightObj = this._createLightByType(ld.type);
   961→
   962→    // For spot/directional, add a target object
   963→    let target: THREE.Object3D | undefined;
   964→    if (lightObj instanceof THREE.SpotLight || lightObj instanceof THREE.DirectionalLight) {
   965→      target = new THREE.Object3D();
   966→      target.position.set(0, 0, 0);
   967→      this._scene.add(target);
   968→      lightObj.target = target;
   969→    }
   970→
   971→    this._scene.add(lightObj);
   972→    this._entries.set(ld.id, { object: lightObj, lightType: ld.type, target });
   973→    this._types.set(ld.id, ld.type);
   974→
   975→    // Create helper
   976→    const helper = this._createHelper(ld.type, lightObj as THREE.Light);
   977→    if (helper) {
   978→      this._helpers.set(ld.id, helper);
   979→      this._scene.add(helper);
   980→    }
   981→  }
   982→
   983→  private _updateLight(ld: LightSyncEntry, shouldShow: boolean): void {
   984→    const entry = this._entries.get(ld.id);
   985→    if (!entry) return;
   986→
   987→    const { object: lightObj } = entry;
   988→
   989→    // Compute position from spherical
   990→    const s = ld.transform.spherical;
   991→    const latRad = (s.lat * Math.PI) / 180;
   992→    const lngRad = (s.lng * Math.PI) / 180;
   993→    const px = s.radius * Math.cos(latRad) * Math.cos(lngRad);
   994→    const py = s.height;
   995→    const pz = s.radius * Math.cos(latRad) * Math.sin(lngRad);
   996→
   997→    lightObj.position.set(px, py, pz);
   998→    lightObj.visible = shouldShow;
   999→
  1000→    // Apply rotation if enabled
  1001→    if (ld.transform.rotation.enabled) {
  1002→      lightObj.rotation.set(
  1003→        (ld.transform.rotation.x * Math.PI) / 180,
  1004→        (ld.transform.rotation.y * Math.PI) / 180,
  1005→        (ld.transform.rotation.z * Math.PI) / 180,
  1006→      );
  1007→    }
  1008→
  1009→    if (lightObj instanceof THREE.Light) {
  1010→      lightObj.color.set(ld.color);
  1011→      // Map brightness 0-1000 to intensity 0-10
  1012→      lightObj.intensity = (ld.brightness / 1000) * 10;
  1013→
  1014→      // Apply opacity (only meaningful via distance/decay for physically-based)
  1015→      // We scale intensity by opacity/100 to simulate dimming
  1016→      if (ld.opacity < 100) {
  1017→        lightObj.intensity *= ld.opacity / 100;
  1018→      }
  1019→
  1020→      // Falloff
  1021→      if (lightObj instanceof THREE.PointLight || lightObj instanceof THREE.SpotLight) {
  1022→        switch (ld.falloff) {
  1023→          case 'linear':
  1024→            lightObj.decay = 1;
  1025→            break;
  1026→          case 'none':
  1027→            lightObj.decay = 0;
  1028→            break;
  1029→          case 'custom':
  1030→            lightObj.decay = 1.5;
  1031→            break;
  1032→          case 'quadratic':
  1033→          default:
  1034→            lightObj.decay = 2;
  1035→            break;
  1036→        }
  1037→      }
  1038→
  1039→      // Spotlight-specific
  1040→      if (lightObj instanceof THREE.SpotLight) {
  1041→        if (ld.spotAngle !== undefined) {
  1042→          lightObj.angle = (ld.spotAngle * Math.PI) / 180;
  1043→        }
  1044→        if (ld.spotPenumbra !== undefined) {
  1045→          lightObj.penumbra = ld.spotPenumbra;
  1046→        }
  1047→        if (ld.spotDecay !== undefined) {
  1048→          lightObj.decay = ld.spotDecay;
  1049→        }
  1050→      }
  1051→
  1052→      // Area light dimensions
  1053→      if (lightObj instanceof THREE.RectAreaLight) {
  1054→        if (ld.areaWidth !== undefined) lightObj.width = ld.areaWidth;
  1055→        if (ld.areaHeight !== undefined) lightObj.height = ld.areaHeight;
  1056→      }
  1057→
  1058→      // Shadow config
  1059→      this._configureShadows(lightObj, ld.type);
  1060→    }
  1061→
  1062→    // Update target to look at origin for spot/directional
  1063→    if (entry.target) {
  1064→      entry.target.position.set(0, 0, 0);
  1065→    }
  1066→
  1067→    // Update helper
  1068→    const helper = this._helpers.get(ld.id);
  1069→    if (helper) {
  1070→      helper.visible = shouldShow && ld.gearVisible;
  1071→      helper.position.copy(lightObj.position);
  1072→      helper.rotation.copy(lightObj.rotation);
  1073→      if (helper instanceof THREE.Mesh && lightObj instanceof THREE.Light) {
  1074→        (helper.material as THREE.MeshBasicMaterial).color.copy(lightObj.color);
  1075→      }
  1076→    }
  1077→  }
  1078→
  1079→  private _createLightByType(type: string): THREE.Object3D {
  1080→    const color = 0xffffff;
  1081→    const intensity = 1;
  1082→    switch (type) {
  1083→      case 'directional': {
  1084→        const dl = new THREE.DirectionalLight(color, intensity);
  1085→        dl.castShadow = true;
  1086→        dl.shadow.mapSize.set(2048, 2048);
  1087→        dl.shadow.camera.near = 0.5;
  1088→        dl.shadow.camera.far = 50;
  1089→        dl.shadow.camera.left = -10;
  1090→        dl.shadow.camera.right = 10;
  1091→        dl.shadow.camera.top = 10;
  1092→        dl.shadow.camera.bottom = -10;
  1093→        dl.shadow.bias = -0.001;
  1094→        return dl;
  1095→      }
  1096→      case 'spot': {
  1097→        const sl = new THREE.SpotLight(color, intensity, 50, Math.PI / 4, 0.5, 2);
  1098→        sl.castShadow = true;
  1099→        sl.shadow.mapSize.set(2048, 2048);
  1100→        sl.shadow.bias = -0.001;
  1101→        return sl;
  1102→      }
  1103→      case 'area': {
  1104→        ensureRectAreaLib();
  1105→        return new THREE.RectAreaLight(color, intensity, 2, 2);
  1106→      }
  1107→      case 'overhead': {
  1108→        ensureRectAreaLib();
  1109→        return new THREE.RectAreaLight(color, intensity, 4, 4);
  1110→      }
  1111→      case 'rim': {
  1112→        const sl = new THREE.SpotLight(color, intensity, 50, Math.PI / 6, 0.3, 2);
  1113→        sl.castShadow = true;
  1114→        sl.shadow.mapSize.set(2048, 2048);
  1115→        sl.shadow.bias = -0.001;
  1116→        return sl;
  1117→      }
  1118→      case 'underlight': {
  1119→        const pl = new THREE.PointLight(color, intensity, 50, 2);
  1120→        pl.castShadow = true;
  1121→        pl.shadow.mapSize.set(1024, 1024);
  1122→        return pl;
  1123→      }
  1124→      case 'ies':
  1125→      case 'point':
  1126→      default: {
  1127→        const pl = new THREE.PointLight(color, intensity, 50, 2);
  1128→        pl.castShadow = true;
  1129→        pl.shadow.mapSize.set(1024, 1024);
  1130→        return pl;
  1131→      }
  1132→    }
  1133→  }

  1135→  private _configureShadows(light: THREE.Object3D, _type: string): void {
  1136→    const sl = light as THREE.SpotLight;
  1137→    const dl = light as THREE.DirectionalLight;
  1138→    const pl = light as THREE.PointLight;
  1139→    const shadowCapable = light instanceof THREE.SpotLight
  1140→      || light instanceof THREE.DirectionalLight
  1141→      || light instanceof THREE.PointLight;
  1142→
  1143→    if (!shadowCapable || !sl.shadow) return;
  1144→
  1145→    sl.castShadow = true;
  1146→    sl.shadow.bias = -0.001;
  1147→
  1148→    if (light instanceof THREE.DirectionalLight) {
  1149→      dl.shadow.mapSize.set(2048, 2048);
  1150→      dl.shadow.camera.near = 0.5;
  1151→      dl.shadow.camera.far = 50;
  1152→      dl.shadow.camera.left = -10;
  1153→      dl.shadow.camera.right = 10;
  1154→      dl.shadow.camera.top = 10;
  1155→      dl.shadow.camera.bottom = -10;
  1156→    } else if (light instanceof THREE.SpotLight) {
  1157→      sl.shadow.mapSize.set(2048, 2048);
  1158→      sl.shadow.camera.near = 0.5;
  1159→      sl.shadow.camera.far = 50;
  1160→    } else if (light instanceof THREE.PointLight) {
  1161→      pl.shadow.mapSize.set(1024, 1024);
  1162→      pl.shadow.camera.near = 0.1;
  1163→      pl.shadow.camera.far = 50;

  1167→  private _createHelper(type: string, light: THREE.Light): THREE.Object3D | null {
  1168→    const isArea = type === 'area' || type === 'overhead';
  1169→    let geo: THREE.BufferGeometry;
  1170→    if (isArea) {
  1171→      geo = new THREE.PlaneGeometry(0.3, 0.3);
  1172→    } else if (type === 'directional') {
  1173→      // Use a slightly larger sphere for directional
  1174→      geo = new THREE.SphereGeometry(0.1, 12, 12);
  1175→    } else {
  1176→      geo = new THREE.SphereGeometry(0.06, 10, 10);
  1177→    }
  1178→    const mat = new THREE.MeshBasicMaterial({
  1179→      color: light.color,
  1180→      transparent: true,
  1181→      opacity: 0.7,
  1182→      depthTest: false,
  1183→    });
  1184→    const mesh = new THREE.Mesh(geo, mat);
  1185→    mesh.userData.isHelper = true;
  1186→    mesh.renderOrder = 999;
  1187→    return mesh;
  1188→  }

  1190→  private _removeLight(id: string): void {
  1191→    const entry = this._entries.get(id);
  1192→    if (!entry) return;
  1193→
  1194→    const { object, target } = entry;
  1195→
  1196→    // Dispose shadow maps
  1197→    const shadowLight = object as THREE.SpotLight | THREE.DirectionalLight | THREE.PointLight;
  1198→    if (object instanceof THREE.Light && shadowLight.shadow && shadowLight.shadow.map) {
  1199→      shadowLight.shadow.map.dispose();
  1200→    }
  1201→
  1202→    this._scene.remove(object);
  1203→    this._disposeObject(object);
  1204→    if (target) {
  1205→      this._scene.remove(target);
  1206→    }
  1207→
  1208→    this._entries.delete(id);
  1209→    this._types.delete(id);
  1210→
  1211→    const helper = this._helpers.get(id);
  1212→    if (helper) {
  1213→      this._scene.remove(helper);
  1214→      this._disposeObject(helper);
  1215→      this._helpers.delete(id);
  1216→    }
  1217→  }
  1218→
  1219→  private _disposeObject(obj: THREE.Object3D): void {
  1220→    obj.traverse((child) => {
  1221→      if (child instanceof THREE.Mesh) {
  1222→        child.geometry.dispose();
  1223→        if (child.material instanceof THREE.Material) {
  1224→          child.material.dispose();
  1225→        }
  1226→      }
  1227→    });
  1228→  }
  1229→
  1230→  getLightObject(id: string): THREE.Object3D | undefined {
  1231→    return this._entries.get(id)?.object;
  1232→  }
  1233→
  1234→  getLightCount(): number {
  1235→    return this._entries.size;
  1236→  }
  1237→
  1238→  dispose(): void {
  1239→    const ids = Array.from(this._entries.keys());
  1240→    for (const id of ids) {
  1241→      this._removeLight(id);
  1242→    }
  1243→  }
  1244→}

  1251→  private _progress = 0;
  1252→  private _onProgress: ((progress: number) => void) | null = null;
  1253→  private _onLoaded: ((name: string) => void) | null = null;
  1254→  private _onError: ((error: string) => void) | null = null;
  1255→
  1256→  constructor(scene: THREE.Scene) {
  1257→    this._scene = scene;
  1258→  }
  1259→

  1262→

  1265→  }
  1266→
  1267→  setCallbacks(callbacks: {
  1268→    onProgress?: (progress: number) => void;
  1269→    onLoaded?: (name: string) => void;
  1270→    onError?: (error: string) => void;
  1271→  }): void {
  1272→    this._onProgress = callbacks.onProgress ?? null;
  1273→    this._onLoaded = callbacks.onLoaded ?? null;
  1274→    this._onError = callbacks.onError ?? null;
  1275→  }
  1276→
  1277→  async loadFromFile(file: File): Promise<void> {
  1278→    if (this._loading) return;
  1279→    this._loading = true;
  1280→    this._progress = 0;
  1281→    this._onProgress?.(0);
  1282→
  1283→    // Read file as ArrayBuffer to store raw data for scene persistence
  1284→    let arrayBuffer: ArrayBuffer | null = null;
  1285→    try {
  1286→      arrayBuffer = await file.arrayBuffer();
  1287→      setRawModelData(arrayBuffer, file.name);
  1288→    } catch {
  1289→      // Non-critical — model will load but won't be saveable
  1290→    }
  1291→
  1292→    const url = URL.createObjectURL(file);
  1293→
  1294→    try {
  1295→      const gltf = await new Promise<{ scene: THREE.Group }>((resolve, reject) => {
  1296→        this._loader.load(
  1297→          url,
  1298→          (result) => resolve(result),
  1299→          (event) => {
  1300→            if (event.lengthComputable) {
  1301→              this._progress = (event.loaded / event.total) * 100;
  1302→              this._onProgress?.(this._progress);
  1303→            }
  1304→          },
  1305→          (error) => reject(error)
  1306→        );
  1307→      });
  1308→
  1309→      this.removeCurrentModel();
  1310→
  1311→      this._currentModel = gltf.scene;
  1312→      this._scene.add(gltf.scene);
  1313→
  1314→      // Enable shadows
  1315→      gltf.scene.traverse((child) => {
  1316→        if (child instanceof THREE.Mesh) {
  1317→          child.castShadow = true;
  1318→          child.receiveShadow = true;
  1319→        }
  1320→      });
  1321→
  1322→      // Center and scale
  1323→      const box = new THREE.Box3().setFromObject(gltf.scene);
  1324→      const center = box.getCenter(new THREE.Vector3());
  1325→      const size = box.getSize(new THREE.Vector3());
  1326→      const maxDim = Math.max(size.x, size.y, size.z);
  1327→      if (maxDim > 0) {
  1328→        const scale = 4 / maxDim;
  1329→        gltf.scene.scale.multiplyScalar(scale);
  1330→      }
  1331→      // Recompute after scaling
  1332→      const box2 = new THREE.Box3().setFromObject(gltf.scene);
  1333→      const center2 = box2.getCenter(new THREE.Vector3());
  1334→      gltf.scene.position.sub(center2);
  1335→      gltf.scene.position.y += box2.getSize(new THREE.Vector3()).y / 2;
  1336→
  1337→      this._loading = false;
  1338→      this._progress = 100;
  1339→      this._onProgress?.(100);
  1340→      this._onLoaded?.(file.name.replace(/\.(glb|gltf)$/i, ''));
  1341→    } catch (err) {
  1342→      this._loading = false;
  1343→      this._progress = 0;
  1344→      const msg = err instanceof Error ? err.message : 'Failed to load model';
  1345→      this._onError?.(msg);
  1346→    } finally {
  1347→      URL.revokeObjectURL(url);
  1348→    }
  1349→  }
