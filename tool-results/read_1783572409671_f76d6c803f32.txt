    18→import { useUIStore } from '../../store/uiStore';
    19→
    20→interface ViewportProps {
    21→  sceneManagerRef: React.MutableRefObject<SceneManager | null>;
    22→  onScreenshot: (dataUrl: string) => void;
    23→  onReady?: (renderPipeline: RenderPipeline | null, materialManager: MaterialManager | null) => void;
    24→}
    25→
    26→export const Viewport: React.FC<ViewportProps> = ({ sceneManagerRef, onScreenshot, onReady }) => {
    27→  const containerRef = useRef<HTMLDivElement>(null);
    28→  const resizeObserverRef = useRef<ResizeObserver | null>(null);
    29→  const renderPipelineRef = useRef<RenderPipeline | null>(null);
    30→  const lightManagerRef = useRef<LightManager | null>(null);
    31→  const modelLoaderRef = useRef<ModelLoader | null>(null);
    32→  const envLoaderRef = useRef<EnvironmentLoader | null>(null);
    33→  const materialManagerRef = useRef<MaterialManager | null>(null);
    34→  const fileInputRef = useRef<HTMLInputElement>(null);
    35→  const animAccumulatorRef = useRef(0);
    36→  const mouseDownPos = useRef<{ x: number; y: number } | null>(null);
    37→
    38→  const [isDragging, setIsDragging] = useState(false);
    39→  const [loadProgress, setLoadProgress] = useState(0);
    40→  const [isLoading, setIsLoading] = useState(false);
    41→  const [loadError, setLoadError] = useState<string | null>(null);
    42→
    43→  // Store selectors
    44→  const showGrid = useSceneStore((s) => s.showGrid);
    45→  const turntable = useSceneStore((s) => s.turntable);
    46→  const environment = useSceneStore((s) => s.environment);
    47→  const renderSettings = useSceneStore((s) => s.renderSettings);
    48→  const setModel = useSceneStore((s) => s.setModel);
    49→  const setExposure = useSceneStore((s) => s.setExposure);
    50→  const setBloom = useSceneStore((s) => s.setBloom);
    51→  const pendingModelData = useSceneStore((s) => s._pendingModelDataBase64);
    52→  const pendingModelFileName = useSceneStore((s) => s._pendingModelFileName);
    53→  const clearPendingModelData = useSceneStore((s) => s.clearPendingModelData);
    54→  const pendingHDRIData = useSceneStore((s) => s._pendingHDRIDataBase64);
    55→  const clearPendingHDRIData = useSceneStore((s) => s.clearPendingHDRIData);
    56→  const backplate = useSceneStore((s) => s.environment.backplate);
    57→  const backplateOpacity = useSceneStore((s) => s.environment.backplateOpacity);
    58→
    59→  const lights = useLightsStore((s) => s.lights);
    60→  const updateLight = useLightsStore((s) => s.updateLight);
    61→
    62→  // Animation store selectors (non-reactive — read inside the loop via getState)
    63→  // We only use isPlaying for the dependency to know if animation is active
    64→
    65→  // Initialize the 3D scene on mount
    66→  useEffect(() => {
    67→    if (!containerRef.current) return;
    68→
    69→    const sceneManager = new SceneManager();
    70→    sceneManagerRef.current = sceneManager;
    71→    sceneManager.attach(containerRef.current);
    72→
    73→    const lightManager = new LightManager(sceneManager.scene);
    74→    lightManagerRef.current = lightManager;
    75→
    76→    const modelLoader = new ModelLoader(sceneManager.scene);
    77→    modelLoaderRef.current = modelLoader;
    78→
    79→    // Material Manager
    80→    const materialManager = new MaterialManager();
    81→    materialManagerRef.current = materialManager;
    82→
    83→    // Phase 9: Environment loader
    84→    const envLoader = new EnvironmentLoader();
    85→    envLoaderRef.current = envLoader;
    86→
    87→    modelLoader.setCallbacks({
    88→      onProgress: (p) => setLoadProgress(p),
    89→      onLoaded: (name) => {
    90→        setIsLoading(false);
    91→        setLoadProgress(100);
    92→        setModel('', name);
    93→
    94→        // Extract materials from the loaded model
    95→        setTimeout(() => {
    96→          const model = modelLoader.getCurrentModel?.();
    97→          if (model && materialManager) {
    98→            // Check if we have saved material states (from a scene file restore)
    99→            const savedMaterials = useMaterialEditorStore.getState().materials;
   100→            const hasSavedStates = savedMaterials.length > 0;
   101→
   102→            const matStates = materialManager.extractMaterials(model);
   103→
   104→            if (hasSavedStates) {
   105→              // Merge saved edits over the freshly extracted defaults
   106→              const merged = matStates.map((extracted) => {
   107→                const saved = savedMaterials.find((s) => s.name === extracted.name);
   108→                if (saved) {
   109→                  // Keep the extracted ID (it's linked to the materialMap), but use saved values
   110→                  return { ...extracted, ...saved, id: extracted.id };
   111→                }
   112→                return extracted;
   113→              });
   114→              useMaterialEditorStore.getState().setMaterials(merged);
   115→
   116→              requestAnimationFrame(() => {
   117→                materialManager.rebuildMaterialMap(sceneManager.scene, merged);
   118→                // Apply saved property overrides to Three.js materials
   119→                for (const state of merged) {
   120→                  materialManager.applyMaterialState(state, sceneManager.scene);
   121→                }
   122→              });
   123→            } else {
   124→              useMaterialEditorStore.getState().setMaterials(matStates);
   125→              requestAnimationFrame(() => {
   126→                materialManager.rebuildMaterialMap(sceneManager.scene, matStates);
   127→              });
   128→            }
   129→          }
   130→        }, 100);
   131→
   132→        setTimeout(() => setLoadProgress(0), 1000);
   133→      },
   134→      onError: (err) => {
   135→        setIsLoading(false);
   136→        setLoadProgress(0);
   137→        setLoadError(err);
   138→        setTimeout(() => setLoadError(null), 4000);
   139→      },
   140→    });
   141→
   142→    sceneManager.setGrid(true);
   143→
   144→    const renderPipeline = new RenderPipeline(sceneManager);
   145→    renderPipelineRef.current = renderPipeline;
   146→    renderPipeline.build();
   147→
   148→    // Phase 9: Apply default environment preset
   149→    const defaultPreset = getHDRIPresetById(useSceneStore.getState().environment.presetId);
   150→    if (defaultPreset && defaultPreset.id !== 'none') {
   151→      const envTexture = envLoader.generateFromPreset(defaultPreset, sceneManager.pmremGenerator, useSceneStore.getState().environment.rotation);
   152→      envLoader.setEnvironmentTexture(sceneManager.scene, envTexture, useSceneStore.getState().environment.intensity);
   153→    }
   154→
   155→    // Notify parent that the render pipeline is ready (for export)
   156→    onReady?.(renderPipeline, materialManager);
   157→
   158→    // ── Animation frame accumulator & tick callback ─────────────────────
   159→    const clock = new THREE.Clock();
   160→    animAccumulatorRef.current = 0;
   161→
   162→    const loop = () => {
   163→      const delta = clock.getDelta();
   164→
   165→      // ── Animation tick (frame-accurate playback) ──────────────────────
   166→      const animState = useAnimationStore.getState();
   167→      if (animState.isPlaying && animState.tracks.length > 0) {
   168→        animAccumulatorRef.current += delta;
   169→        const frameDuration = 1 / animState.fps;
   170→
   171→        while (animAccumulatorRef.current >= frameDuration) {
   172→          animAccumulatorRef.current -= frameDuration;
   173→          const wrapped = animState.tick();
   174→
   175→          // Evaluate animation at the new frame
   176→          const evaluated = animationEngine.evaluate(animState.tracks, animState.currentFrame);
   177→
   178→          // Apply evaluated values to stores
   179→          if (evaluated.values.size > 0) {
   180→            AnimationEngine.applyToScene(evaluated, {
   181→              updateLightProperty: (lightId: string, property: string, value: number) => {
   182→                const prop = property as AnimatedProperty;
   183→                const currentLight = useLightsStore.getState().lights.find((l) => l.id === lightId);
   184→                if (!currentLight) return;
   185→                const t = currentLight.transform;
   186→                switch (prop) {
   187→                  case 'light.brightness':
   188→                    updateLight(lightId, { brightness: value });
   189→                    break;
   190→                  case 'light.positionX':
   191→                    updateLight(lightId, { transform: { ...t, position: { ...t.position, x: value } } });
   192→                    break;
   193→                  case 'light.positionY':
   194→                    updateLight(lightId, { transform: { ...t, position: { ...t.position, y: value } } });
   195→                    break;
   196→                  case 'light.positionZ':
   197→                    updateLight(lightId, { transform: { ...t, position: { ...t.position, z: value } } });
   198→                    break;
   199→                  case 'light.rotationX':
   200→                    updateLight(lightId, { transform: { ...t, rotation: { ...t.rotation, x: value, enabled: true } } });
   201→                    break;
   202→                  case 'light.rotationY':
   203→                    updateLight(lightId, { transform: { ...t, rotation: { ...t.rotation, y: value, enabled: true } } });
   204→                    break;
   205→                  case 'light.rotationZ':
   206→                    updateLight(lightId, { transform: { ...t, rotation: { ...t.rotation, z: value, enabled: true } } });
   207→                    break;
   208→                }
   209→              },
   210→              updateCameraProperty: (_property: string, _value: number) => {
   211→                // Camera animation applied to SceneManager directly
   212→                const cam = sceneManager.camera;
   213→                const prop = _property as AnimatedProperty;
   214→                switch (prop) {
   215→                  case 'camera.positionX':
   216→                    cam.position.x = _value;
   217→                    break;
   218→                  case 'camera.positionY':
   219→                    cam.position.y = _value;
   220→                    break;
   221→                  case 'camera.positionZ':
   222→                    cam.position.z = _value;
   223→                    break;
   224→                  case 'camera.fov':
   225→                    cam.fov = _value;
   226→                    cam.updateProjectionMatrix();
   227→                    break;
   228→                  default:
   229→                    break;
   230→                }
   231→                sceneManager.controls.update();
   232→              },
   233→              updateTurntableProperty: (property: string, value: number) => {
   234→                if (property === 'turntable.speed') {
   235→                  sceneManager._turntableSpeed = value;
   236→                } else if (property === 'turntable.rotation') {
   237→                  const model = sceneManager._findModel();
   238→                  if (model) {
   239→                    model.rotation.y = (value * Math.PI) / 180;
   240→                  }
   241→                }
   242→              },
   243→              updateRenderProperty: (property: string, value: number) => {
   244→                if (property === 'render.exposure') {
   245→                  setExposure(value);
   246→                } else if (property === 'render.bloomIntensity') {
   247→                  setBloom({ intensity: value });
   248→                }
   249→              },
   250→            }, animState.tracks);
   251→          }
   252→        }
   253→      } else {
   254→        // Reset accumulator when not playing
   255→        animAccumulatorRef.current = 0;
   256→      }
   257→
   258→      // ── Turntable (non-animated, manual rotation) ─────────────────────
   259→      if (sceneManager._turntableActive) {
   260→        const model = sceneManager._findModel();
   261→        if (model) model.rotation.y += delta * sceneManager._turntableSpeed * 0.5;
   262→      }
   263→
   264→      sceneManager.controls.update();
   265→      renderPipeline.render();
   266→      sceneManager._animationId = requestAnimationFrame(loop);
   267→    };
   268→    sceneManager._animationId = requestAnimationFrame(loop);
   269→
   270→    // Set up resize observer
   271→    const observer = new ResizeObserver(() => {
   272→      sceneManager.resize();
   273→      renderPipeline.resize();
   274→    });
   275→    observer.observe(containerRef.current);
   276→    resizeObserverRef.current = observer;
   277→
   278→    return () => {
   279→      observer.disconnect();
   280→      sceneManager.stopRenderLoop();
   281→      renderPipeline.dispose();
   282→      modelLoader.dispose();
   283→      lightManager.dispose();
   284→      envLoader.dispose();
   285→      materialManager.dispose();
   286→      sceneManager.dispose();
   287→      sceneManagerRef.current = null;
   288→      renderPipelineRef.current = null;
   289→      lightManagerRef.current = null;
   290→      modelLoaderRef.current = null;
   291→      envLoaderRef.current = null;
   292→    };
   293→    // eslint-disable-next-line react-hooks/exhaustive-deps
   294→  }, []);
   295→
   296→  // Sync grid visibility
   297→  useEffect(() => {
   298→    sceneManagerRef.current?.setGrid(showGrid);
   299→  }, [showGrid, sceneManagerRef]);
   300→
   301→  // Backplate: render as scene.background when set
   302→  const backplateTextureRef = useRef<THREE.Texture | null>(null);
   303→
   304→  useEffect(() => {
   305→    const sm = sceneManagerRef.current;
   306→    if (!sm) return;
   307→
   308→    if (backplate) {
   309→      // Load backplate as texture for background
   310→      if (backplateTextureRef.current) {
   311→        backplateTextureRef.current.dispose();
   312→        backplateTextureRef.current = null;
   313→      }
   314→      const loader = new THREE.TextureLoader();
   315→      loader.load(backplate, (tex) => {
   316→        tex.colorSpace = THREE.SRGBColorSpace;
   317→        tex.minFilter = THREE.LinearFilter;
   318→        tex.magFilter = THREE.LinearFilter;
   319→        backplateTextureRef.current = tex;
   320→        if (backplateOpacity >= 0.99) {
   321→          sm.scene.background = tex;
   322→        }
   323→      });
   324→    } else {
   325→      if (backplateTextureRef.current) {
   326→        backplateTextureRef.current.dispose();
   327→        backplateTextureRef.current = null;
   328→      }
   329→      // Restore normal background behavior
   330→      sm.setBackground(
   331→        useSceneStore.getState().environment.background,
   332→        useSceneStore.getState().environment.showBackground,
   333→      );
   334→    }
   335→
   336→    return () => {
   337→      if (backplateTextureRef.current) {
   338→        backplateTextureRef.current.dispose();
   339→        backplateTextureRef.current = null;
   340→      }
   341→    };
   342→  }, [backplate, backplateOpacity, sceneManagerRef]);
   343→
   344→  // Sync environment background (only when no backplate)
   345→  useEffect(() => {
   346→    const sm = sceneManagerRef.current;
   347→    if (!sm || backplate) return; // Skip if backplate is active
   348→    sm.setBackground(environment.background, environment.showBackground);
   349→  }, [environment.background, environment.showBackground, sceneManagerRef, backplate]);
   350→
   351→  // Sync ground settings (reflections, fade, color, PBR)
   352→  useEffect(() => {
   353→    const sm = sceneManagerRef.current;
   354→    if (!sm) return;
   355→    sm.updateGround(renderSettings.ground);
   356→  }, [renderSettings.ground, sceneManagerRef]);
   357→
   358→  // Phase 9: Sync environment preset to 3D scene (built-in presets only)
   359→  useEffect(() => {
   360→    const sm = sceneManagerRef.current;
   361→    const el = envLoaderRef.current;
   362→    if (!sm || !el) return;
   363→
   364→    // Skip custom HDRI — handled by the separate effect below
   365→    if (environment.presetId === '__custom__') return;
   366→
   367→    const preset = getHDRIPresetById(environment.presetId);
   368→    if (!preset) return;
   369→
   370→    const envTexture = el.generateFromPreset(preset, sm.pmremGenerator, environment.rotation);
   371→    el.setEnvironmentTexture(sm.scene, envTexture, environment.intensity);
   372→  }, [environment.presetId, environment.rotation, environment.intensity, sceneManagerRef, envLoaderRef]);
   373→
   374→  // Phase 9: Load custom HDRI file into the 3D scene
   375→  useEffect(() => {
   376→    const sm = sceneManagerRef.current;
   377→    const el = envLoaderRef.current;
   378→    if (!sm || !el) return;
   379→
   380→    if (environment.presetId === '__custom__' && environment.hdri) {
   381→      el.loadHDRI(environment.hdri, sm.pmremGenerator)
   382→        .then((envTexture) => {
   383→          el.setEnvironmentTexture(sm.scene, envTexture, environment.intensity);
   384→        })
   385→        .catch(() => {
   386→          // Fallback to neutral studio on error
   387→          const fallback = getHDRIPresetById('studio-neutral');
   388→          if (fallback) {
   389→            const envTexture = el.generateFromPreset(fallback, sm.pmremGenerator, 0);
   390→            el.setEnvironmentTexture(sm.scene, envTexture, environment.intensity);
   391→          }
   392→        });
   393→    }
   394→  }, [environment.presetId, environment.hdri, environment.intensity, sceneManagerRef, envLoaderRef]);
   395→
   396→  // Restore model from scene file (triggered when _pendingModelDataBase64 is set)
   397→  useEffect(() => {
   398→    if (!pendingModelData) return;
   399→    const ml = modelLoaderRef.current;
   400→    if (!ml) return;
   401→
   402→    setIsLoading(true);
   403→    setLoadError(null);
   404→    setLoadProgress(0);
   405→
   406→    try {
   407→      const arrayBuffer = base64ToArrayBuffer(pendingModelData);
   408→      ml.loadFromBuffer(arrayBuffer, pendingModelFileName);
   409→    } catch {
   410→      setLoadError('Failed to restore model from scene file');
   411→      setIsLoading(false);
   412→    }
   413→
   414→    clearPendingModelData();
   415→  }, [pendingModelData, pendingModelFileName, modelLoaderRef, clearPendingModelData]);
   416→
   417→  // Restore custom HDRI from scene file
   418→  useEffect(() => {
   419→    if (!pendingHDRIData) return;
   420→    const sm = sceneManagerRef.current;
   421→    const el = envLoaderRef.current;
   422→    if (!sm || !el) return;
   423→
   424→    try {
   425→      const arrayBuffer = hdriBase64ToArrayBuffer(pendingHDRIData);
   426→      setRawHDRIData(arrayBuffer, 'custom.hdr');
   427→      const blob = new Blob([arrayBuffer], { type: 'application/octet-stream' });
   428→      const url = URL.createObjectURL(blob);
   429→      el.loadHDRI(url, sm.pmremGenerator)
   430→        .then((envTexture) => {
   431→          el.setEnvironmentTexture(sm.scene, envTexture, useSceneStore.getState().environment.intensity);
   432→        })
   433→        .catch(() => {
   434→          // Fallback to neutral studio
   435→          const fallback = getHDRIPresetById('studio-neutral');
   436→          if (fallback) {
   437→            const tex = el.generateFromPreset(fallback, sm.pmremGenerator, 0);
   438→            el.setEnvironmentTexture(sm.scene, tex, 1);
   439→          }
   440→        });
   441→    } catch {
   442→      // ignore
   443→    }
   444→
   445→    clearPendingHDRIData();
   446→  }, [pendingHDRIData, sceneManagerRef, envLoaderRef, clearPendingHDRIData]);
   447→
   448→  // Sync render settings
   449→  useEffect(() => {
   450→    const rp = renderPipelineRef.current;
   451→    if (!rp) return;
   452→    rp.setEngine(renderSettings.engine);
   453→    rp.setToneMapping(renderSettings.tonemapping);
   454→    rp.setExposure(renderSettings.exposure);
   455→    rp.setQuality(renderSettings.quality);
   456→    rp.setShadowQuality(renderSettings.shadowQuality);
   457→    rp.setBloom(
   458→      renderSettings.bloom.enabled,
   459→      renderSettings.bloom.intensity,
   460→      renderSettings.bloom.threshold,
   461→      renderSettings.bloom.radius
   462→    );
   463→    rp.setAO(
   464→      renderSettings.ao.enabled,
   465→      renderSettings.ao.radius,
   466→      renderSettings.ao.intensity
   467→    );
   468→    rp.setAntialiasing(renderSettings.antialiasing);
   469→    rp.setVignette(
   470→      renderSettings.vignette.enabled,
   471→      renderSettings.vignette.intensity
   472→    );
   473→    rp.setColorGrading(
   474→      renderSettings.colorGrading.enabled,
   475→      renderSettings.colorGrading.brightness,
   476→      renderSettings.colorGrading.contrast,
   477→      renderSettings.colorGrading.saturation
   478→    );
   479→  }, [renderSettings, renderPipelineRef]);
   480→
   481→  // Sync lights to scene
   482→  useEffect(() => {
   483→    lightManagerRef.current?.syncLights(
   484→      lights.map((l) => ({
   485→        id: l.id,
   486→        type: l.type,
   487→        color: l.color,
   488→        brightness: l.brightness,
   489→        opacity: l.opacity,
   490→        visible: l.visible,
   491→        solo: l.solo,
   492→        falloff: l.falloff,
   493→        gearVisible: l.gearVisible,
   494→        transform: l.transform,
   495→        spotAngle: l.spotAngle,
   496→        spotPenumbra: l.spotPenumbra,
   497→        spotDecay: 2,
   498→        areaWidth: l.areaWidth,
   499→        areaHeight: l.areaHeight,
   500→      }))
   501→    );
   502→  }, [lights, lightManagerRef]);
   503→
   504→  // Sync turntable state to SceneManager
   505→  useEffect(() => {
   506→    const sm = sceneManagerRef.current;
   507→    if (!sm) return;
   508→    sm._turntableActive = turntable.active;
   509→    sm._turntableSpeed = turntable.speed;
   510→  }, [turntable.active, turntable.speed, sceneManagerRef]);
   511→
   512→  // Click-to-select material from viewport
   513→  const handleViewportClick = useCallback((event: React.MouseEvent) => {
   514→    // Only work with select tool
   515→    if (useUIStore.getState().activeTool !== 'select') return;
   516→
   517→    const container = containerRef.current;
   518→    const sm = sceneManagerRef.current;
   519→    if (!container || !sm) return;
   520→
   521→    const rect = container.getBoundingClientRect();
   522→    const mouse = new THREE.Vector2(
   523→      ((event.clientX - rect.left) / rect.width) * 2 - 1,
   524→      -((event.clientY - rect.top) / rect.height) * 2 + 1
   525→    );
   526→
   527→    const raycaster = new THREE.Raycaster();
   528→    raycaster.setFromCamera(mouse, sm.camera);
   529→
   530→    const meshes: THREE.Mesh[] = [];
   531→    sm.scene.traverse((obj) => {
   532→      if (obj instanceof THREE.Mesh && !(obj as any).userData?.isHelper) {
   533→        meshes.push(obj);
   534→      }
   535→    });
   536→
   537→    const hits = raycaster.intersectObjects(meshes, false);
   538→    if (hits.length > 0) {
   539→      const mesh = hits[0].object as THREE.Mesh;
   540→      // Get material name from Three.js material
   541→      const material = Array.isArray(mesh.material) ? mesh.material[0] : mesh.material;
   542→      const matName = material.name || mesh.name;
   543→
   544→      // Find matching material in store by name or mesh name
   545→      const matState = useMaterialEditorStore.getState().materials;
   546→      const match = matState.find((m) =>
   547→        m.name === matName || m.meshNames.includes(mesh.name)
   548→      );
   549→
   550→      if (match) {
   551→        useMaterialEditorStore.getState().selectMaterial(match.id);
   552→
   553→        // Open right panel and switch to matEdit tab if needed
   554→        const ui = useUIStore.getState();
   555→        if (!ui.rightPanelOpen) ui.showPanel('rightPanel');
   556→
   557→        // Visual feedback: brief emissive flash on the mesh
   558→        if (material && 'emissive' in material) {
   559→          const mat = material as THREE.MeshStandardMaterial;
   560→          const origEmissive = mat.emissive.clone();
   561→          mat.emissive.set(0x444466);
   562→          setTimeout(() => {
   563→            mat.emissive.copy(origEmissive);
   564→            mat.needsUpdate = true;
   565→          }, 200);
   566→        }
   567→      }
   568→    }
   569→  }, [sceneManagerRef]);
   570→
   571→  // Drag and drop handlers
   572→  const handleDragEnter = useCallback((e: React.DragEvent) => {
   573→    e.preventDefault();
   574→    e.stopPropagation();
   575→    setIsDragging(true);
   576→  }, []);
   577→
   578→  const handleDragLeave = useCallback((e: React.DragEvent) => {
   579→    e.preventDefault();
   580→    e.stopPropagation();
   581→    if (e.currentTarget === e.target) {
   582→      setIsDragging(false);
   583→    }
   584→  }, []);
   585→
   586→  const handleDragOver = useCallback((e: React.DragEvent) => {
   587→    e.preventDefault();
   588→    e.stopPropagation();
   589→    e.dataTransfer.dropEffect = 'copy';
   590→  }, []);
   591→
   592→  const handleDrop = useCallback(
   593→    (e: React.DragEvent) => {
   594→      e.preventDefault();
   595→      e.stopPropagation();
   596→      setIsDragging(false);
   597→
   598→      const files = e.dataTransfer.files;
   599→      if (files.length === 0) return;
   600→
   601→      const file = files[0];
   602→      const ext = file.name.toLowerCase().split('.').pop();
   603→      if (ext === 'glb' || ext === 'gltf') {
   604→        loadModelFile(file);
   605→      }
   606→    },
   607→    // eslint-disable-next-line react-hooks/exhaustive-deps
   608→    [modelLoaderRef]
   609→  );
   610→
   611→  // Load model from file
   612→  const loadModelFile = useCallback(
   613→    (file: File) => {
   614→      if (!modelLoaderRef.current) return;
   615→      setIsLoading(true);
   616→      setLoadError(null);
   617→      setLoadProgress(0);
   618→      modelLoaderRef.current.loadFromFile(file);
   619→    },
   620→    [modelLoaderRef]
   621→  );
   622→
   623→  // File picker
   624→  const handleFileInput = useCallback(
   625→    (e: React.ChangeEvent<HTMLInputElement>) => {
   626→      const file = e.target.files?.[0];
   627→      if (file) loadModelFile(file);
   628→      e.target.value = '';
   629→    },
   630→    [loadModelFile]
   631→  );
   632→
   633→  const handleOpenFilePicker = useCallback(() => {
   634→    fileInputRef.current?.click();
   635→  }, []);
   636→
   637→  // Screenshot handler
   638→  const handleScreenshot = useCallback(() => {
   639→    const dataUrl = renderPipelineRef.current
   640→      ? renderPipelineRef.current.capture()
   641→      : sceneManagerRef.current?.takeScreenshot() ?? '';
   642→    if (dataUrl) onScreenshot(dataUrl);
   643→  }, [sceneManagerRef, renderPipelineRef, onScreenshot]);
   644→
   645→  return (
   646→    <ThreeSceneProvider>
   647→      <div style={{ display: 'flex', flexDirection: 'column', width: '100%', height: '100%' }}>
   648→        <ViewportToolbar
   649→          sceneManagerRef={sceneManagerRef}
   650→          onScreenshot={handleScreenshot}
   651→          onLoadModel={handleOpenFilePicker}
   652→        />
   653→        <div
   654→          ref={containerRef}
   655→          style={{
   656→            position: 'relative',
   657→            flex: 1,
   658→            overflow: 'hidden',
   659→            background: 'var(--bg-deep)',
   660→          }}
   661→          onMouseDown={(e) => { mouseDownPos.current = { x: e.clientX, y: e.clientY }; }}
   662→          onClick={(e) => {
   663→            if (mouseDownPos.current) {
   664→              const dx = e.clientX - mouseDownPos.current.x;
   665→              const dy = e.clientY - mouseDownPos.current.y;
   666→              if (Math.sqrt(dx*dx + dy*dy) < 3) {
   667→                handleViewportClick(e);
   668→              }
   669→            }
   670→            mouseDownPos.current = null;
   671→          }}
   672→          onDragEnter={handleDragEnter}
   673→          onDragLeave={handleDragLeave}
   674→          onDragOver={handleDragOver}
   675→          onDrop={handleDrop}
   676→        >
   677→          {isDragging && (
   678→            <div className="drop-overlay">
   679→              <span>Drop .glb file to load model</span>
   680→            </div>
   681→          )}
   682→
   683→          {isLoading && loadProgress < 100 && (
   684→            <div
   685→              style={{
   686→                position: 'absolute',
   687→                top: 0,
   688→                left: 0,
   689→                right: 0,
   690→                zIndex: 10,
   691→                padding: '6px 12px',
   692→              }}
   693→            >
   694→              <div className="progress-bar">
   695→                <div
   696→                  className="progress-bar-fill"
   697→                  style={{ width: `${loadProgress}%` }}
   698→                />
   699→              </div>
   700→            </div>
   701→          )}
   702→
   703→          {loadError && (
   704→            <div
   705→              style={{
   706→                position: 'absolute',
   707→                bottom: 12,
   708→                left: '50%',
   709→                transform: 'translateX(-50%)',
   710→                zIndex: 10,
   711→                background: 'var(--danger)',
   712→                color: '#fff',
   713→                padding: '4px 12px',
   714→                borderRadius: 'var(--radius-sm)',
   715→                fontSize: 11,
   716→              }}
   717→            >
   718→              {loadError}
   719→            </div>
   720→          )}
   721→
   722→          <input
   723→            ref={fileInputRef}
   724→            type="file"
   725→            accept=".glb,.gltf"
   726→            style={{ display: 'none' }}
   727→            onChange={handleFileInput}
   728→          />
   729→
   730→          <CameraBookmarks sceneManagerRef={sceneManagerRef} />
   731→        </div>
   732→      </div>
   733→    </ThreeSceneProvider>
   734→  );
   735→};
