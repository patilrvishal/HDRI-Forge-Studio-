   159→   159→  animateCameraTo(
   160→   160→    position: [number, number, number],
   161→   161→    target: [number, number, number],
   162→   162→    duration: number = 600
   163→   163→  ): void {
   164→   164→    const startPos = this.camera.position.clone();
   165→   165→    const startTarget = this.controls.target.clone();
   166→   166→    const endPos = new THREE.Vector3(...position);
   167→   167→    const endTarget = new THREE.Vector3(...target);
   168→   168→    const startTime = performance.now();
   169→   169→
   170→   170→    const animate = () => {
   171→   171→      const elapsed = performance.now() - startTime;
   172→   172→      const t = Math.min(elapsed / duration, 1);
   173→   173→      const ease = t < 0.5 ? 4 * t * t * t : 1 - Math.pow(-2 * t + 2, 3) / 2;
   174→   174→
   175→   175→      this.camera.position.lerpVectors(startPos, endPos, ease);
   176→   176→      this.controls.target.lerpVectors(startTarget, endTarget, ease);
   177→   177→      this.controls.update();
   178→   178→
   179→   179→      if (t < 1) {
   180→   180→        requestAnimationFrame(animate);
   181→   181→      }
   182→   182→    };
   183→   183→    requestAnimationFrame(animate);
   184→   184→  }
   185→   185→
   186→   186→  setOnFrame(cb: (delta: number) => void): void {
   187→   187→    this._onFrame = cb;
   188→   188→  }
   189→   189→
   190→   190→  startRenderLoop(): void {
   191→   191→    const loop = () => {
   192→   192→      this._animationId = requestAnimationFrame(loop);
   193→   193→      const delta = this._clock.getDelta();
   194→   194→
   195→   195→      // Turntable rotation
   196→   196→      if (this._turntableActive) {
   197→   197→        const model = this._findModel();
   198→   198→        if (model) {
   199→   199→          model.rotation.y += delta * this._turntableSpeed * 0.5;
   200→   200→        }
   201→   201→      }
   202→   202→
   203→   203→      if (this._onFrame) this._onFrame(delta);
   204→   204→      this.controls.update();
   205→   205→      this.renderer.render(this.scene, this.camera);
   206→   206→    };
   207→   207→    loop();
   208→   208→  }
   209→   209→
   210→   210→  _findModel(): THREE.Object3D | null {
   211→   211→    for (const child of this.scene.children) {
   212→   212→      if (
   213→   213→        child.type === 'Group' &&
   214→   214→        !(child instanceof THREE.GridHelper) &&
   215→   215→        !child.userData.isHelper &&
   216→   216→        child !== this.ground
   217→   217→      ) {
   218→   218→        return child;
   219→   219→      }
   220→   220→    }
   221→   221→    return null;
   222→   222→  }
   223→   223→
   224→   224→  stopRenderLoop(): void {
   225→   225→    cancelAnimationFrame(this._animationId);
   226→   226→  }

   268→   268→      }
   269→   269→    });

   271→   271→    this.renderer.dispose();
   272→   272→  }

   275→   275→// ── Vignette shader (custom) ────────────────────────────────────────────────
   276→   276→const VignetteShader = {
   277→   277→  uniforms: {
   278→   278→    tDiffuse: { value: null as THREE.Texture | null },
   279→   279→    intensity: { value: 0.4 },
   280→   280→    offset: { value: 1.1 },
   281→   281→    darkness: { value: 1.2 },
   282→   282→  },
   283→   283→  vertexShader: /* glsl */ `
   284→   284→    varying vec2 vUv;
   285→   285→    void main() {
   286→   286→      vUv = uv;
   287→   287→      gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
   288→   288→    }
   289→   289→  `,
   290→   290→  fragmentShader: /* glsl */ `
   291→   291→    uniform sampler2D tDiffuse;
   292→   292→    uniform float intensity;
   293→   293→    uniform float offset;
   294→   294→    uniform float darkness;
   295→   295→    varying vec2 vUv;
   296→   296→    void main() {
   297→   297→      vec4 texel = texture2D(tDiffuse, vUv);
   298→   298→      vec2 uv = (vUv - vec2(0.5)) * vec2(offset);
   299→   299→      float vig = clamp(1.0 - dot(uv, uv), 0.0, 1.0);
   300→   300→      float f = mix(1.0, vig, intensity * darkness);
   301→   301→      gl_FragColor = vec4(texel.rgb * f, texel.a);
   302→   302→    }
   303→   303→  `,
   304→   304→};
   305→   305→
   306→   306→// ── Color Grading shader (brightness / contrast / saturation) ───────────────
   307→   307→const ColorGradingShader = {
   308→   308→  uniforms: {
   309→   309→    tDiffuse: { value: null as THREE.Texture | null },
   310→   310→    brightness: { value: 0.0 },
   311→   311→    contrast: { value: 0.0 },
   312→   312→    saturation: { value: 0.0 },
   313→   313→  },
   314→   314→  vertexShader: /* glsl */ `
   315→   315→    varying vec2 vUv;
   316→   316→    void main() {
   317→   317→      vUv = uv;
   318→   318→      gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
   319→   319→    }
   320→   320→  `,
   321→   321→  fragmentShader: /* glsl */ `
   322→   322→    uniform sampler2D tDiffuse;
   323→   323→    uniform float brightness;
   324→   324→    uniform float contrast;
   325→   325→    uniform float saturation;
   326→   326→    varying vec2 vUv;
   327→   327→
   328→   328→    void main() {
   329→   329→      vec4 texel = texture2D(tDiffuse, vUv);
   330→   330→      vec3 c = texel.rgb;
   331→   331→
   332→   332→      // Brightness
   333→   333→      c += brightness * 0.5;
   334→   334→
   335→   335→      // Contrast
   336→   336→      c = (c - 0.5) * (1.0 + contrast) + 0.5;
   337→   337→
   338→   338→      // Saturation
   339→   339→      float lum = dot(c, vec3(0.2126, 0.7152, 0.0722));
   340→   340→      c = mix(vec3(lum), c, 1.0 + saturation);
   341→   341→
   342→   342→      gl_FragColor = vec4(clamp(c, 0.0, 1.0), texel.a);
   343→   343→    }
   344→   344→  `,
   345→   345→};

   347→   347→// ── Pipeline settings interface ─────────────────────────────────────────────
   348→   348→export interface PipelineConfig {
   349→   349→  bloom: { enabled: boolean; intensity: number; threshold: number; radius: number };
   350→   350→  ao: { enabled: boolean; radius: number; intensity: number };
   351→   351→  vignette: { enabled: boolean; intensity: number };
   352→   352→  colorGrading: { enabled: boolean; brightness: number; contrast: number; saturation: number };
   353→   353→  antialiasing: 'none' | 'fxaa' | 'smaa' | 'taa';
   354→   354→  shadowQuality: 'none' | 'low' | 'medium' | 'high';
   355→   355→  quality: 'low' | 'medium' | 'high' | 'ultra';
   356→   356→  tonemapping: 'aces' | 'reinhard' | 'linear';
   357→   357→  exposure: number;
   358→   358→}
   359→   359→
   360→   360→export class RenderPipeline {
   361→   361→  private _sm: SceneManager;
   362→   362→  private _composer: EffectComposer | null = null;
   363→   363→  private _renderPass: RenderPass | null = null;
   364→   364→  private _bloomPass: UnrealBloomPass | null = null;
   365→   365→  private _smaaPass: SMAAPass | null = null;
   366→   366→  private _fxaaPass: ShaderPass | null = null;
   367→   367→  private _ssaoPass: SSAOPass | null = null;
   368→   368→  private _vignettePass: ShaderPass | null = null;
   369→   369→  private _colorGradingPass: ShaderPass | null = null;
   370→   370→  private _config: PipelineConfig;
   371→   371→  private _needsRebuild = false;
   372→   372→
   373→   373→  constructor(sceneManager: SceneManager) {
   374→   374→    this._sm = sceneManager;
   375→   375→    this._config = {
   376→   376→      bloom: { enabled: true, intensity: 0.3, threshold: 0.8, radius: 0.5 },
   377→   377→      ao: { enabled: false, radius: 0.5, intensity: 0.5 },
   378→   378→      vignette: { enabled: false, intensity: 0.4 },
   379→   379→      colorGrading: { enabled: false, brightness: 0, contrast: 0, saturation: 0 },
   380→   380→      antialiasing: 'smaa',
   381→   381→      shadowQuality: 'high',
   382→   382→      quality: 'high',
   383→   383→      tonemapping: 'aces',
   384→   384→      exposure: 1.0,
   385→   385→    };
   386→   386→  }
   387→   387→
   388→   388→  /** Build (or rebuild) the EffectComposer from scratch. */
   389→   389→  build(): void {
   390→   390→    this.dispose();
   391→   391→    const r = this._sm.renderer;
   392→   392→    const w = r.domElement.width || r.getSize(new THREE.Vector2()).x || 2;
   393→   393→    const h = r.domElement.height || r.getSize(new THREE.Vector2()).y || 2;
   394→   394→
   395→   395→    this._composer = new EffectComposer(r);
   396→   396→    this._renderPass = new RenderPass(this._sm.scene, this._sm.camera);
   397→   397→    this._composer.addPass(this._renderPass);
   398→   398→
   399→   399→    // Bloom
   400→   400→    if (this._config.bloom.enabled) {
   401→   401→      this._bloomPass = new UnrealBloomPass(
   402→   402→        new THREE.Vector2(w / 2, h / 2),
   403→   403→        this._config.bloom.intensity,
   404→   404→        this._config.bloom.radius,
   405→   405→        this._config.bloom.threshold,
   406→   406→      );
   407→   407→      this._composer.addPass(this._bloomPass);
   408→   408→    }
   409→   409→
   410→   410→    // Anti-aliasing
   411→   411→    if (this._config.antialiasing === 'smaa') {
   412→   412→      // @ts-expect-error — SMAAPass types are incomplete in @types/three
   413→   413→      this._smaaPass = new SMAAPass(w, h);
   414→   414→      this._composer.addPass(this._smaaPass);
   415→   415→    } else if (this._config.antialiasing === 'fxaa') {
   416→   416→      const fxaaMat = new THREE.ShaderMaterial({ ...FXAAShader, uniforms: { ...FXAAShader.uniforms, resolution: { value: new THREE.Vector2(1 / w, 1 / h) } } });
   417→   417→      this._fxaaPass = new ShaderPass(fxaaMat);
   418→   418→      this._composer.addPass(this._fxaaPass);
   419→   419→    }
   420→   420→    // 'taa' — handled via renderer settings + jitter; we keep SMAA as fallback
   421→   421→    // 'none' — no AA pass
   422→   422→
   423→   423→    // SSAO
   424→   424→    if (this._config.ao.enabled) {
   425→   425→      this._ssaoPass = new SSAOPass(
   426→   426→        this._sm.scene,
   427→   427→        this._sm.camera,
   428→   428→        w,
   429→   429→        h,
   430→   430→      );
   431→   431→      this._applyAOParams(this._config.ao.radius, this._config.ao.intensity);
   432→   432→      this._ssaoPass.output = SSAOPass.OUTPUT.Default;
   433→   433→      this._composer.addPass(this._ssaoPass);
   434→   434→    }
   435→   435→
   436→   436→    // Vignette
   437→   437→    if (this._config.vignette.enabled) {
   438→   438→      const vMat = new THREE.ShaderMaterial(VignetteShader);
   439→   439→      vMat.uniforms['intensity'].value = this._config.vignette.intensity;
   440→   440→      this._vignettePass = new ShaderPass(vMat);
   441→   441→      this._composer.addPass(this._vignettePass);
   442→   442→    }
   443→   443→
   444→   444→    // Color grading
   445→   445→    if (this._config.colorGrading.enabled) {
   446→   446→      const cgMat = new THREE.ShaderMaterial(ColorGradingShader);
   447→   447→      cgMat.uniforms['brightness'].value = this._config.colorGrading.brightness;
   448→   448→      cgMat.uniforms['contrast'].value = this._config.colorGrading.contrast;
   449→   449→      cgMat.uniforms['saturation'].value = this._config.colorGrading.saturation;
   450→   450→      this._colorGradingPass = new ShaderPass(cgMat);
   451→   451→      this._composer.addPass(this._colorGradingPass);
   452→   452→    }
   453→   453→
   454→   454→    // Apply tone mapping & exposure to the renderer
   455→   455→    this._applyToneMapping(this._config.tonemapping);
   456→   456→    this._sm.renderer.toneMappingExposure = this._config.exposure;
   457→   457→
   458→   458→    // Shadow quality
   459→   459→    this._applyShadowQuality(this._config.shadowQuality);
   460→   460→  }
   461→   461→
   462→   462→  /** Render one frame through the composer (or fallback direct render). */
   463→   463→  render(): void {
   464→   464→    if (this._composer) {
   465→   465→      this._composer.render();
   466→   466→    } else {
   467→   467→      this._sm.renderer.render(this._sm.scene, this._sm.camera);
   468→   468→    }
   469→   469→  }
   470→   470→
   471→   471→  /** Resize the composer and internal passes. */
   472→   472→  resize(): void {
   473→   473→    if (!this._composer) return;
   474→   474→    const r = this._sm.renderer;
   475→   475→    const size = r.getSize(new THREE.Vector2());
   476→   476→    const w = size.x || 2;
   477→   477→    const h = size.y || 2;
   478→   478→    this._composer.setSize(w, h);
   479→   479→
   480→   480→    if (this._bloomPass) {
   481→   481→      this._bloomPass.resolution.set(w / 2, h / 2);
   482→   482→    }
   483→   483→    if (this._ssaoPass) {
   484→   484→      this._ssaoPass.setSize(w, h);
   485→   485→    }
   486→   486→    if (this._smaaPass) {
   487→   487→      this._smaaPass.setSize(w, h);
   488→   488→    }
   489→   489→    if (this._fxaaPass) {
   490→   490→      const mat = this._fxaaPass.material as THREE.ShaderMaterial;
   491→   491→      const res = mat.uniforms['resolution'];
   492→   492→      if (res && res.value instanceof THREE.Vector2) {
   493→   493→        res.value.set(1 / w, 1 / h);
   494→   494→      }
   495→   495→    }
   496→   496→  }

   498→   498→  /** Update the full pipeline config. Rebuilds composer if structural changes detected. */
   499→   499→  updateConfig(cfg: Partial<PipelineConfig>): void {
   500→   500→    const prev = { ...this._config };
   501→   501→    this._config = { ...this._config, ...cfg };
   502→   502→
   503→   503→    const structuralChange =
   504→   504→      cfg.bloom?.enabled !== undefined && cfg.bloom.enabled !== prev.bloom.enabled ||
   505→   505→      cfg.antialiasing !== undefined && cfg.antialiasing !== prev.antialiasing ||
   506→   506→      cfg.vignette?.enabled !== undefined && cfg.vignette.enabled !== prev.vignette.enabled ||
   507→   507→      cfg.ao?.enabled !== undefined && cfg.ao.enabled !== prev.ao.enabled ||
   508→   508→      cfg.colorGrading?.enabled !== undefined && cfg.colorGrading.enabled !== prev.colorGrading.enabled;
   509→   509→
   510→   510→    if (structuralChange) {
   511→   511→      this.build();
   512→   512→      return;
   513→   513→    }
   514→   514→
   515→   515→    // In-place parameter updates (no rebuild needed)
   516→   516→    if (this._bloomPass && cfg.bloom) {
   517→   517→      this._bloomPass.strength = cfg.bloom.intensity ?? this._bloomPass.strength;
   518→   518→      this._bloomPass.radius = cfg.bloom.radius ?? this._bloomPass.radius;
   519→   519→      this._bloomPass.threshold = cfg.bloom.threshold ?? this._bloomPass.threshold;
   520→   520→    }
   521→   521→
   522→   522→    if (this._ssaoPass && cfg.ao) {
   523→   523→      this._applyAOParams(
   524→   524→        cfg.ao.radius ?? this._config.ao.radius,
   525→   525→        cfg.ao.intensity ?? this._config.ao.intensity,
   526→   526→      );
   527→   527→    }
   528→   528→
   529→   529→    if (this._vignettePass && cfg.vignette) {
   530→   530→      const mat = this._vignettePass.material as THREE.ShaderMaterial;
   531→   531→      mat.uniforms['intensity'].value = cfg.vignette.intensity ?? 0.4;
   532→   532→    }
   533→   533→
   534→   534→    if (this._colorGradingPass && cfg.colorGrading) {
   535→   535→      const mat = this._colorGradingPass.material as THREE.ShaderMaterial;
   536→   536→      if (cfg.colorGrading.brightness !== undefined) mat.uniforms['brightness'].value = cfg.colorGrading.brightness;
   537→   537→      if (cfg.colorGrading.contrast !== undefined) mat.uniforms['contrast'].value = cfg.colorGrading.contrast;
   538→   538→      if (cfg.colorGrading.saturation !== undefined) mat.uniforms['saturation'].value = cfg.colorGrading.saturation;
   539→   539→    }
   540→   540→
   541→   541→    if (cfg.tonemapping !== undefined) {
   542→   542→      this._applyToneMapping(cfg.tonemapping);
   543→   543→    }
   544→   544→    if (cfg.exposure !== undefined) {
   545→   545→      this._sm.renderer.toneMappingExposure = cfg.exposure;
   546→   546→    }
   547→   547→    if (cfg.shadowQuality !== undefined) {
   548→   548→      this._applyShadowQuality(cfg.shadowQuality);
   549→   549→    }
   550→   550→    if (cfg.quality !== undefined) {
   551→   551→      this._applyQuality(cfg.quality);
   552→   552→    }
   553→   553→  }
   554→   554→

   620→   620→  /** Dispose all passes and the composer. */
   621→   621→  dispose(): void {
   622→   622→    if (this._composer) {
   623→   623→      const passes = this._composer.passes.slice().reverse();
   624→   624→      for (const pass of passes) {
   625→   625→        if ('dispose' in pass && typeof (pass as { dispose: () => void }).dispose === 'function') {
   626→   626→          (pass as { dispose: () => void }).dispose();
   627→   627→        }
   628→   628→      }
   629→   629→      this._composer.dispose();
   630→   630→    }
   631→   631→    this._composer = null;
   632→   632→    this._renderPass = null;
   633→   633→    this._bloomPass = null;
   634→   634→    this._ssaoPass = null;
   635→   635→    this._smaaPass = null;
   636→   636→    this._fxaaPass = null;
   637→   637→    this._vignettePass = null;
   638→   638→    this._colorGradingPass = null;

   641→   641→  // ── Private helpers ───────────────────────────────────────────────────────
   642→   642→
   643→   643→  /** Apply SSAO parameters — radius controls kernel spread, intensity controls maxDistance scaling. */
   644→   644→  private _applyAOParams(radius: number, intensity: number): void {
   645→   645→    if (!this._ssaoPass) return;
   646→   646→    this._ssaoPass.kernelRadius = radius;
   647→   647→    this._ssaoPass.minDistance = 0.001;
   648→   648→    // Scale maxDistance with intensity: higher intensity → AO visible at greater depth range
   649→   649→    this._ssaoPass.maxDistance = 0.02 + intensity * 0.08;
   650→   650→  }
   651→   651→
   652→   652→  private _applyToneMapping(mapping: 'aces' | 'reinhard' | 'linear'): void {
   653→   653→    const r = this._sm.renderer;
   654→   654→    switch (mapping) {
   655→   655→      case 'aces':
   656→   656→        r.toneMapping = THREE.ACESFilmicToneMapping;
   657→   657→        break;
   658→   658→      case 'reinhard':
   659→   659→        r.toneMapping = THREE.ReinhardToneMapping;
   660→   660→        break;
   661→   661→      case 'linear':
   662→   662→        r.toneMapping = THREE.LinearToneMapping;
   663→   663→        break;
   664→   664→    }
   665→   665→    r.toneMappingExposure = this._config.exposure;
   666→   666→  }
   667→   667→
   668→   668→  private _applyShadowQuality(quality: string): void {
   669→   669→    const r = this._sm.renderer;
   670→   670→    switch (quality) {
   671→   671→      case 'none':
   672→   672→        r.shadowMap.enabled = false;
   673→   673→        break;
   674→   674→      case 'low':
   675→   675→        r.shadowMap.enabled = true;
   676→   676→        r.shadowMap.type = THREE.BasicShadowMap;
   677→   677→        break;
   678→   678→      case 'medium':
   679→   679→        r.shadowMap.enabled = true;
   680→   680→        r.shadowMap.type = THREE.PCFShadowMap;
   681→   681→        break;
   682→   682→      case 'high':
   683→   683→      default:
   684→   684→        r.shadowMap.enabled = true;
   685→   685→        r.shadowMap.type = THREE.PCFSoftShadowMap;
   686→   686→        break;
   687→   687→    }
   688→   688→
   689→   689→    // Update shadow map sizes on all lights
   690→   690→    this._sm.scene.traverse((child) => {
   691→   691→      const light = child as THREE.DirectionalLight | THREE.SpotLight | THREE.PointLight;
   692→   692→      if (light.isLight && light.shadow) {
   693→   693→        const shadow = light.shadow;
   694→   694→        switch (quality) {
   695→   695→          case 'low':
   696→   696→            shadow.mapSize.set(512, 512);
   697→   697→            break;
   698→   698→          case 'medium':
   699→   699→            shadow.mapSize.set(1024, 1024);
   700→   700→            break;
   701→   701→          case 'high':
   702→   702→          default:
   703→   703→            shadow.mapSize.set(2048, 2048);
   704→   704→            break;
   705→   705→        }
   706→   706→        if (shadow.map) {
   707→   707→          shadow.map.dispose();
   708→   708→          shadow.map = null;
   709→   709→        }
   710→   710→      }
   711→   711→    });
   712→   712→  }
   713→   713→
   714→   714→  private _applyQuality(quality: 'low' | 'medium' | 'high' | 'ultra'): void {
   715→   715→    const r = this._sm.renderer;
   716→   716→    switch (quality) {
   717→   717→      case 'low':
   718→   718→        r.setPixelRatio(1);
   719→   719→        break;
   720→   720→      case 'medium':
   721→   721→        r.setPixelRatio(Math.min(window.devicePixelRatio, 1.5));
   722→   722→        break;
   723→   723→      case 'high':
   724→   724→        r.setPixelRatio(Math.min(window.devicePixelRatio, 2));
   725→   725→        break;
   726→   726→      case 'ultra':
   727→   727→        r.setPixelRatio(window.devicePixelRatio);
   728→   728→        break;
   729→   729→    }
   730→   730→    this._sm.resize();
   731→   731→    if (this._composer) {
   732→   732→      this.resize();
   733→   733→    }
   734→   734→  }

   737→   737→interface LightSyncEntry {
   738→   738→  id: string;
   739→   739→  type: string;
   740→   740→  color: string;
   741→   741→  brightness: number;
   742→   742→  opacity: number;
   743→   743→  visible: boolean;
   744→   744→  solo: boolean;
   745→   745→  falloff: string;
   746→   746→  gearVisible: boolean;
   747→   747→  transform: {
   748→   748→    position: { x: number; y: number; z: number };
   749→   749→    spherical: { lat: number; lng: number; radius: number; height: number };
   750→   750→    rotation: { x: number; y: number; z: number; enabled: boolean };
   751→   751→  };
   752→   752→  // Spotlight-specific
   753→   753→  spotAngle?: number;
   754→   754→  spotPenumbra?: number;
   755→   755→  spotDecay?: number;
   756→   756→  // Area light specific
   757→   757→  areaWidth?: number;
   758→   758→  areaHeight?: number;
   759→   759→}
   760→   760→
   761→   761→interface LightObjectEntry {
   762→   762→  object: THREE.Object3D;
   763→   763→  lightType: string;
   764→   764→  target?: THREE.Object3D;
   765→   765→}
   766→   766→
   767→   767→export class LightManager {
   768→   768→  private _scene: THREE.Scene;
   769→   769→  private _entries: Map<string, LightObjectEntry> = new Map();
   770→   770→  private _helpers: Map<string, THREE.Object3D> = new Map();
   771→   771→  private _types: Map<string, string> = new Map();
   772→   772→
   773→   773→  constructor(scene: THREE.Scene) {
   774→   774→    this._scene = scene;
   775→   775→    ensureRectAreaLib();
   776→   776→  }
   777→   777→
   778→   778→  syncLights(lights: LightSyncEntry[]): void {
   779→   779→    const activeIds = new Set(lights.map((l) => l.id));
   780→   780→    const hasSolo = lights.some((l) => l.solo);
   781→   781→
   782→   782→    // Remove stale lights
   783→   783→    for (const [id] of this._entries) {
   784→   784→      if (!activeIds.has(id)) {
   785→   785→        this._removeLight(id);
   786→   786→      }
   787→   787→    }
   788→   788→
   789→   789→    // Create or update each light
   790→   790→    for (const ld of lights) {
   791→   791→      const isEffectiveSolo = hasSolo ? ld.solo : true;
   792→   792→      const shouldShow = ld.visible && isEffectiveSolo;
   793→   793→
   794→   794→      const existing = this._entries.get(ld.id);
   795→   795→      const currentType = this._types.get(ld.id);
   796→   796→
   797→   797→      // If type changed, recreate the light
   798→   798→      if (existing && currentType !== ld.type) {
   799→   799→        this._removeLight(ld.id);
   800→   800→      }
   801→   801→
   802→   802→      if (!this._entries.has(ld.id)) {
   803→   803→        this._createLight(ld);
   804→   804→      }
   805→   805→
   806→   806→      this._updateLight(ld, shouldShow);
   807→   807→    }
   808→   808→  }
   809→   809→
   810→   810→  private _createLight(ld: LightSyncEntry): void {
   811→   811→    ensureRectAreaLib();
   812→   812→    const lightObj = this._createLightByType(ld.type);
   813→   813→
   814→   814→    // For spot/directional, add a target object
   815→   815→    let target: THREE.Object3D | undefined;
   816→   816→    if (lightObj instanceof THREE.SpotLight || lightObj instanceof THREE.DirectionalLight) {
   817→   817→      target = new THREE.Object3D();
   818→   818→      target.position.set(0, 0, 0);
   819→   819→      this._scene.add(target);
   820→   820→      lightObj.target = target;
   821→   821→    }
   822→   822→
   823→   823→    this._scene.add(lightObj);
   824→   824→    this._entries.set(ld.id, { object: lightObj, lightType: ld.type, target });
   825→   825→    this._types.set(ld.id, ld.type);
   826→   826→
   827→   827→    // Create helper
   828→   828→    const helper = this._createHelper(ld.type, lightObj as THREE.Light);
   829→   829→    if (helper) {
   830→   830→      this._helpers.set(ld.id, helper);
   831→   831→      this._scene.add(helper);
   832→   832→    }
   833→   833→  }
   834→   834→
   835→   835→  private _updateLight(ld: LightSyncEntry, shouldShow: boolean): void {
   836→   836→    const entry = this._entries.get(ld.id);
   837→   837→    if (!entry) return;
   838→   838→
   839→   839→    const { object: lightObj } = entry;
   840→   840→
   841→   841→    // Compute position from spherical
   842→   842→    const s = ld.transform.spherical;
   843→   843→    const latRad = (s.lat * Math.PI) / 180;
   844→   844→    const lngRad = (s.lng * Math.PI) / 180;
   845→   845→    const px = s.radius * Math.cos(latRad) * Math.cos(lngRad);
   846→   846→    const py = s.height;
   847→   847→    const pz = s.radius * Math.cos(latRad) * Math.sin(lngRad);
   848→   848→
   849→   849→    lightObj.position.set(px, py, pz);
   850→   850→    lightObj.visible = shouldShow;
   851→   851→
   852→   852→    // Apply rotation if enabled
   853→   853→    if (ld.transform.rotation.enabled) {
   854→   854→      lightObj.rotation.set(
   855→   855→        (ld.transform.rotation.x * Math.PI) / 180,
   856→   856→        (ld.transform.rotation.y * Math.PI) / 180,
   857→   857→        (ld.transform.rotation.z * Math.PI) / 180,
   858→   858→      );
   859→   859→    }
   860→   860→
   861→   861→    if (lightObj instanceof THREE.Light) {
   862→   862→      lightObj.color.set(ld.color);
   863→   863→      // Map brightness 0-1000 to intensity 0-10
   864→   864→      lightObj.intensity = (ld.brightness / 1000) * 10;
   865→   865→
   866→   866→      // Apply opacity (only meaningful via distance/decay for physically-based)
   867→   867→      // We scale intensity by opacity/100 to simulate dimming
   868→   868→      if (ld.opacity < 100) {
   869→   869→        lightObj.intensity *= ld.opacity / 100;
   870→   870→      }
   871→   871→
   872→   872→      // Falloff
   873→   873→      if (lightObj instanceof THREE.PointLight || lightObj instanceof THREE.SpotLight) {
   874→   874→        switch (ld.falloff) {
   875→   875→          case 'linear':
   876→   876→            lightObj.decay = 1;
   877→   877→            break;
   878→   878→          case 'none':
   879→   879→            lightObj.decay = 0;
   880→   880→            break;
   881→   881→          case 'custom':
   882→   882→            lightObj.decay = 1.5;
   883→   883→            break;
   884→   884→          case 'quadratic':
   885→   885→          default:
   886→   886→            lightObj.decay = 2;
   887→   887→            break;
   888→   888→        }
   889→   889→      }
   890→   890→
   891→   891→      // Spotlight-specific
   892→   892→      if (lightObj instanceof THREE.SpotLight) {
   893→   893→        if (ld.spotAngle !== undefined) {
   894→   894→          lightObj.angle = (ld.spotAngle * Math.PI) / 180;
   895→   895→        }
   896→   896→        if (ld.spotPenumbra !== undefined) {
   897→   897→          lightObj.penumbra = ld.spotPenumbra;
   898→   898→        }
   899→   899→        if (ld.spotDecay !== undefined) {
   900→   900→          lightObj.decay = ld.spotDecay;
   901→   901→        }
   902→   902→      }
   903→   903→
   904→   904→      // Area light dimensions
   905→   905→      if (lightObj instanceof THREE.RectAreaLight) {
   906→   906→        if (ld.areaWidth !== undefined) lightObj.width = ld.areaWidth;
   907→   907→        if (ld.areaHeight !== undefined) lightObj.height = ld.areaHeight;
   908→   908→      }
   909→   909→
   910→   910→      // Shadow config
   911→   911→      this._configureShadows(lightObj, ld.type);
   912→   912→    }
   913→   913→
   914→   914→    // Update target to look at origin for spot/directional
   915→   915→    if (entry.target) {
   916→   916→      entry.target.position.set(0, 0, 0);
   917→   917→    }
   918→   918→
   919→   919→    // Update helper
   920→   920→    const helper = this._helpers.get(ld.id);
   921→   921→    if (helper) {
   922→   922→      helper.visible = shouldShow && ld.gearVisible;
   923→   923→      helper.position.copy(lightObj.position);
   924→   924→      helper.rotation.copy(lightObj.rotation);
   925→   925→      if (helper instanceof THREE.Mesh && lightObj instanceof THREE.Light) {
   926→   926→        (helper.material as THREE.MeshBasicMaterial).color.copy(lightObj.color);
   927→   927→      }
   928→   928→    }
   929→   929→  }
   930→   930→
   931→   931→  private _createLightByType(type: string): THREE.Object3D {
   932→   932→    const color = 0xffffff;
   933→   933→    const intensity = 1;
   934→   934→    switch (type) {
   935→   935→      case 'directional': {
   936→   936→        const dl = new THREE.DirectionalLight(color, intensity);
   937→   937→        dl.castShadow = true;
   938→   938→        dl.shadow.mapSize.set(2048, 2048);
   939→   939→        dl.shadow.camera.near = 0.5;
   940→   940→        dl.shadow.camera.far = 50;

  1042→  1042→  private _removeLight(id: string): void {
  1043→  1043→    const entry = this._entries.get(id);
  1044→  1044→    if (!entry) return;
  1045→  1045→
  1046→  1046→    const { object, target } = entry;
  1047→  1047→
  1048→  1048→    // Dispose shadow maps
  1049→  1049→    const shadowLight = object as THREE.SpotLight | THREE.DirectionalLight | THREE.PointLight;
  1050→  1050→    if (object instanceof THREE.Light && shadowLight.shadow && shadowLight.shadow.map) {
  1051→  1051→      shadowLight.shadow.map.dispose();
  1052→  1052→    }
  1053→  1053→
  1054→  1054→    this._scene.remove(object);
  1055→  1055→    this._disposeObject(object);
  1056→  1056→    if (target) {
  1057→  1057→      this._scene.remove(target);
  1058→  1058→    }
  1059→  1059→
  1060→  1060→    this._entries.delete(id);
  1061→  1061→    this._types.delete(id);
  1062→  1062→
  1063→  1063→    const helper = this._helpers.get(id);
  1064→  1064→    if (helper) {
  1065→  1065→      this._scene.remove(helper);
  1066→  1066→      this._disposeObject(helper);
  1067→  1067→      this._helpers.delete(id);
  1068→  1068→    }
  1069→  1069→  }

  1079→  1079→    });

  1098→  1098→export class ModelLoader {

  1101→  1101→  private _loader = new GLTFLoader();

  1103→  1103→  private _progress = 0;

  1105→  1105→  private _onLoaded: ((name: string) => void) | null = null;
  1106→  1106→  private _onError: ((error: string) => void) | null = null;
  1107→  1107→
  1108→  1108→  constructor(scene: THREE.Scene) {

  1110→  1110→  }
  1111→  1111→

  1125→  1125→  async loadFromFile(file: File): Promise<void> {
  1126→  1126→    if (this._loading) return;
  1127→  1127→    this._loading = true;
  1128→  1128→    this._progress = 0;
  1129→  1129→    this._onProgress?.(0);
  1130→  1130→
  1131→  1131→    const url = URL.createObjectURL(file);
  1132→  1132→
  1133→  1133→    try {
  1134→  1134→      const gltf = await new Promise<{ scene: THREE.Group }>((resolve, reject) => {
  1135→  1135→        this._loader.load(
  1136→  1136→          url,
  1137→  1137→          (result) => resolve(result),
  1138→  1138→          (event) => {
  1139→  1139→            if (event.lengthComputable) {
  1140→  1140→              this._progress = (event.loaded / event.total) * 100;
  1141→  1141→              this._onProgress?.(this._progress);
  1142→  1142→            }
  1143→  1143→          },
  1144→  1144→          (error) => reject(error)
  1145→  1145→        );
  1146→  1146→      });
  1147→  1147→
  1148→  1148→      this.removeCurrentModel();
  1149→  1149→
  1150→  1150→      this._currentModel = gltf.scene;
  1151→  1151→      this._scene.add(gltf.scene);
  1152→  1152→
  1153→  1153→      // Enable shadows
  1154→  1154→      gltf.scene.traverse((child) => {
  1155→  1155→        if (child instanceof THREE.Mesh) {
  1156→  1156→          child.castShadow = true;
  1157→  1157→          child.receiveShadow = true;
  1158→  1158→        }
  1159→  1159→      });
  1160→  1160→
  1161→  1161→      // Center and scale
  1162→  1162→      const box = new THREE.Box3().setFromObject(gltf.scene);
  1163→  1163→      const center = box.getCenter(new THREE.Vector3());
  1164→  1164→      const size = box.getSize(new THREE.Vector3());
  1165→  1165→      const maxDim = Math.max(size.x, size.y, size.z);
  1166→  1166→      if (maxDim > 0) {
  1167→  1167→        const scale = 4 / maxDim;
  1168→  1168→        gltf.scene.scale.multiplyScalar(scale);
  1169→  1169→      }
  1170→  1170→      // Recompute after scaling
  1171→  1171→      const box2 = new THREE.Box3().setFromObject(gltf.scene);
  1172→  1172→      const center2 = box2.getCenter(new THREE.Vector3());
  1173→  1173→      gltf.scene.position.sub(center2);
  1174→  1174→      gltf.scene.position.y += box2.getSize(new THREE.Vector3()).y / 2;
  1175→  1175→
  1176→  1176→      this._loading = false;
  1177→  1177→      this._progress = 100;
  1178→  1178→      this._onProgress?.(100);
  1179→  1179→      this._onLoaded?.(file.name.replace(/\.(glb|gltf)$/i, ''));
  1180→  1180→    } catch (err) {
  1181→  1181→      this._loading = false;
  1182→  1182→      this._progress = 0;
  1183→  1183→      const msg = err instanceof Error ? err.message : 'Failed to load model';
  1184→  1184→      this._onError?.(msg);
  1185→  1185→    } finally {
  1186→  1186→      URL.revokeObjectURL(url);
  1187→  1187→    }
  1188→  1188→  }

  1190→  1190→  removeCurrentModel(): void {
  1191→  1191→    if (this._currentModel) {
  1192→  1192→      this._scene.remove(this._currentModel);

  1197→  1197→            child.material.forEach((m) => m.dispose());
  1198→  1198→          } else {
  1199→  1199→            child.material.dispose();
  1200→  1200→          }
  1201→  1201→        }
  1202→  1202→      });
  1203→  1203→      this._currentModel = null;
  1204→  1204→    }
  1205→  1205→  }
