   160→   160→  animateCameraTo(
   161→   161→    position: [number, number, number],
   162→   162→    target: [number, number, number],
   163→   163→    duration: number = 600
   164→   164→  ): void {
   165→   165→    const startPos = this.camera.position.clone();
   166→   166→    const startTarget = this.controls.target.clone();
   167→   167→    const endPos = new THREE.Vector3(...position);
   168→   168→    const endTarget = new THREE.Vector3(...target);
   169→   169→    const startTime = performance.now();
   170→   170→
   171→   171→    const animate = () => {
   172→   172→      const elapsed = performance.now() - startTime;
   173→   173→      const t = Math.min(elapsed / duration, 1);
   174→   174→      const ease = t < 0.5 ? 4 * t * t * t : 1 - Math.pow(-2 * t + 2, 3) / 2;
   175→   175→
   176→   176→      this.camera.position.lerpVectors(startPos, endPos, ease);
   177→   177→      this.controls.target.lerpVectors(startTarget, endTarget, ease);
   178→   178→      this.controls.update();
   179→   179→
   180→   180→      if (t < 1) {
   181→   181→        requestAnimationFrame(animate);
   182→   182→      }
   183→   183→    };
   184→   184→    requestAnimationFrame(animate);
   185→   185→  }
   186→   186→
   187→   187→  setOnFrame(cb: (delta: number) => void): void {
   188→   188→    this._onFrame = cb;
   189→   189→  }
   190→   190→
   191→   191→  startRenderLoop(): void {
   192→   192→    const loop = () => {
   193→   193→      this._animationId = requestAnimationFrame(loop);
   194→   194→      const delta = this._clock.getDelta();
   195→   195→
   196→   196→      // Turntable rotation
   197→   197→      if (this._turntableActive) {
   198→   198→        const model = this._findModel();
   199→   199→        if (model) {
   200→   200→          model.rotation.y += delta * this._turntableSpeed * 0.5;
   201→   201→        }
   202→   202→      }
   203→   203→
   204→   204→      if (this._onFrame) this._onFrame(delta);
   205→   205→      this.controls.update();
   206→   206→      this.renderer.render(this.scene, this.camera);
   207→   207→    };
   208→   208→    loop();
   209→   209→  }
   210→   210→
   211→   211→  _findModel(): THREE.Object3D | null {
   212→   212→    for (const child of this.scene.children) {
   213→   213→      if (
   214→   214→        child.type === 'Group' &&
   215→   215→        !(child instanceof THREE.GridHelper) &&
   216→   216→        !child.userData.isHelper &&
   217→   217→        child !== this.ground
   218→   218→      ) {
   219→   219→        return child;
   220→   220→      }
   221→   221→    }
   222→   222→    return null;
   223→   223→  }
   224→   224→
   225→   225→  stopRenderLoop(): void {
   226→   226→    cancelAnimationFrame(this._animationId);
   227→   227→  }

   266→   266→        } else {
   267→   267→          obj.material.dispose();
   268→   268→        }
   269→   269→      }
   270→   270→    });
   271→   271→    this.controls.dispose();
   272→   272→    this.renderer.dispose();
   273→   273→  }

   276→   276→// ── Vignette shader (custom) ────────────────────────────────────────────────
   277→   277→const VignetteShader = {
   278→   278→  uniforms: {
   279→   279→    tDiffuse: { value: null as THREE.Texture | null },
   280→   280→    intensity: { value: 0.4 },
   281→   281→    offset: { value: 1.1 },
   282→   282→    darkness: { value: 1.2 },
   283→   283→  },
   284→   284→  vertexShader: /* glsl */ `
   285→   285→    varying vec2 vUv;
   286→   286→    void main() {
   287→   287→      vUv = uv;
   288→   288→      gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
   289→   289→    }
   290→   290→  `,
   291→   291→  fragmentShader: /* glsl */ `
   292→   292→    uniform sampler2D tDiffuse;
   293→   293→    uniform float intensity;
   294→   294→    uniform float offset;
   295→   295→    uniform float darkness;
   296→   296→    varying vec2 vUv;
   297→   297→    void main() {
   298→   298→      vec4 texel = texture2D(tDiffuse, vUv);
   299→   299→      vec2 uv = (vUv - vec2(0.5)) * vec2(offset);
   300→   300→      float vig = clamp(1.0 - dot(uv, uv), 0.0, 1.0);
   301→   301→      float f = mix(1.0, vig, intensity * darkness);
   302→   302→      gl_FragColor = vec4(texel.rgb * f, texel.a);
   303→   303→    }
   304→   304→  `,
   305→   305→};
   306→   306→
   307→   307→// ── Color Grading shader (brightness / contrast / saturation) ───────────────
   308→   308→const ColorGradingShader = {
   309→   309→  uniforms: {
   310→   310→    tDiffuse: { value: null as THREE.Texture | null },
   311→   311→    brightness: { value: 0.0 },
   312→   312→    contrast: { value: 0.0 },
   313→   313→    saturation: { value: 0.0 },
   314→   314→  },
   315→   315→  vertexShader: /* glsl */ `
   316→   316→    varying vec2 vUv;
   317→   317→    void main() {
   318→   318→      vUv = uv;
   319→   319→      gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
   320→   320→    }
   321→   321→  `,
   322→   322→  fragmentShader: /* glsl */ `
   323→   323→    uniform sampler2D tDiffuse;
   324→   324→    uniform float brightness;
   325→   325→    uniform float contrast;
   326→   326→    uniform float saturation;
   327→   327→    varying vec2 vUv;
   328→   328→
   329→   329→    void main() {
   330→   330→      vec4 texel = texture2D(tDiffuse, vUv);
   331→   331→      vec3 c = texel.rgb;
   332→   332→
   333→   333→      // Brightness
   334→   334→      c += brightness * 0.5;
   335→   335→
   336→   336→      // Contrast
   337→   337→      c = (c - 0.5) * (1.0 + contrast) + 0.5;
   338→   338→
   339→   339→      // Saturation
   340→   340→      float lum = dot(c, vec3(0.2126, 0.7152, 0.0722));
   341→   341→      c = mix(vec3(lum), c, 1.0 + saturation);
   342→   342→
   343→   343→      gl_FragColor = vec4(clamp(c, 0.0, 1.0), texel.a);
   344→   344→    }
   345→   345→  `,
   346→   346→};

   348→   348→// ── Pipeline settings interface ─────────────────────────────────────────────
   349→   349→export interface PipelineConfig {
   350→   350→  bloom: { enabled: boolean; intensity: number; threshold: number; radius: number };
   351→   351→  ao: { enabled: boolean; radius: number; intensity: number };
   352→   352→  vignette: { enabled: boolean; intensity: number };
   353→   353→  colorGrading: { enabled: boolean; brightness: number; contrast: number; saturation: number };
   354→   354→  antialiasing: 'none' | 'fxaa' | 'smaa' | 'taa';
   355→   355→  shadowQuality: 'none' | 'low' | 'medium' | 'high';
   356→   356→  quality: 'low' | 'medium' | 'high' | 'ultra';
   357→   357→  tonemapping: 'aces' | 'reinhard' | 'linear';
   358→   358→  exposure: number;
   359→   359→}
   360→   360→
   361→   361→export class RenderPipeline {
   362→   362→  private _sm: SceneManager;
   363→   363→  private _composer: EffectComposer | null = null;
   364→   364→  private _renderPass: RenderPass | null = null;
   365→   365→  private _bloomPass: UnrealBloomPass | null = null;
   366→   366→  private _smaaPass: SMAAPass | null = null;
   367→   367→  private _fxaaPass: ShaderPass | null = null;
   368→   368→  private _ssaoPass: SSAOPass | null = null;
   369→   369→  private _vignettePass: ShaderPass | null = null;
   370→   370→  private _colorGradingPass: ShaderPass | null = null;
   371→   371→  private _config: PipelineConfig;
   372→   372→  private _needsRebuild = false;
   373→   373→
   374→   374→  constructor(sceneManager: SceneManager) {
   375→   375→    this._sm = sceneManager;
   376→   376→    this._config = {
   377→   377→      bloom: { enabled: true, intensity: 0.3, threshold: 0.8, radius: 0.5 },
   378→   378→      ao: { enabled: false, radius: 0.5, intensity: 0.5 },
   379→   379→      vignette: { enabled: false, intensity: 0.4 },
   380→   380→      colorGrading: { enabled: false, brightness: 0, contrast: 0, saturation: 0 },
   381→   381→      antialiasing: 'smaa',
   382→   382→      shadowQuality: 'high',
   383→   383→      quality: 'high',
   384→   384→      tonemapping: 'aces',
   385→   385→      exposure: 1.0,
   386→   386→    };
   387→   387→  }
   388→   388→
   389→   389→  /** Build (or rebuild) the EffectComposer from scratch. */
   390→   390→  build(): void {
   391→   391→    this.dispose();
   392→   392→    const r = this._sm.renderer;
   393→   393→    const w = r.domElement.width || r.getSize(new THREE.Vector2()).x || 2;
   394→   394→    const h = r.domElement.height || r.getSize(new THREE.Vector2()).y || 2;
   395→   395→
   396→   396→    this._composer = new EffectComposer(r);
   397→   397→    this._renderPass = new RenderPass(this._sm.scene, this._sm.camera);
   398→   398→    this._composer.addPass(this._renderPass);
   399→   399→
   400→   400→    // Bloom
   401→   401→    if (this._config.bloom.enabled) {
   402→   402→      this._bloomPass = new UnrealBloomPass(
   403→   403→        new THREE.Vector2(w / 2, h / 2),
   404→   404→        this._config.bloom.intensity,
   405→   405→        this._config.bloom.radius,
   406→   406→        this._config.bloom.threshold,
   407→   407→      );
   408→   408→      this._composer.addPass(this._bloomPass);
   409→   409→    }
   410→   410→
   411→   411→    // Anti-aliasing
   412→   412→    if (this._config.antialiasing === 'smaa') {
   413→   413→      // @ts-expect-error — SMAAPass types are incomplete in @types/three
   414→   414→      this._smaaPass = new SMAAPass(w, h);
   415→   415→      this._composer.addPass(this._smaaPass);
   416→   416→    } else if (this._config.antialiasing === 'fxaa') {
   417→   417→      const fxaaMat = new THREE.ShaderMaterial({ ...FXAAShader, uniforms: { ...FXAAShader.uniforms, resolution: { value: new THREE.Vector2(1 / w, 1 / h) } } });
   418→   418→      this._fxaaPass = new ShaderPass(fxaaMat);
   419→   419→      this._composer.addPass(this._fxaaPass);
   420→   420→    }
   421→   421→    // 'taa' — handled via renderer settings + jitter; we keep SMAA as fallback
   422→   422→    // 'none' — no AA pass
   423→   423→
   424→   424→    // SSAO
   425→   425→    if (this._config.ao.enabled) {
   426→   426→      this._ssaoPass = new SSAOPass(
   427→   427→        this._sm.scene,
   428→   428→        this._sm.camera,
   429→   429→        w,
   430→   430→        h,
   431→   431→      );
   432→   432→      this._applyAOParams(this._config.ao.radius, this._config.ao.intensity);
   433→   433→      this._ssaoPass.output = SSAOPass.OUTPUT.Default;
   434→   434→      this._composer.addPass(this._ssaoPass);
   435→   435→    }
   436→   436→
   437→   437→    // Vignette
   438→   438→    if (this._config.vignette.enabled) {
   439→   439→      const vMat = new THREE.ShaderMaterial(VignetteShader);
   440→   440→      vMat.uniforms['intensity'].value = this._config.vignette.intensity;
   441→   441→      this._vignettePass = new ShaderPass(vMat);
   442→   442→      this._composer.addPass(this._vignettePass);
   443→   443→    }
   444→   444→
   445→   445→    // Color grading
   446→   446→    if (this._config.colorGrading.enabled) {
   447→   447→      const cgMat = new THREE.ShaderMaterial(ColorGradingShader);
   448→   448→      cgMat.uniforms['brightness'].value = this._config.colorGrading.brightness;
   449→   449→      cgMat.uniforms['contrast'].value = this._config.colorGrading.contrast;
   450→   450→      cgMat.uniforms['saturation'].value = this._config.colorGrading.saturation;
   451→   451→      this._colorGradingPass = new ShaderPass(cgMat);
   452→   452→      this._composer.addPass(this._colorGradingPass);
   453→   453→    }
   454→   454→
   455→   455→    // Apply tone mapping & exposure to the renderer
   456→   456→    this._applyToneMapping(this._config.tonemapping);
   457→   457→    this._sm.renderer.toneMappingExposure = this._config.exposure;
   458→   458→
   459→   459→    // Shadow quality
   460→   460→    this._applyShadowQuality(this._config.shadowQuality);
   461→   461→  }
   462→   462→
   463→   463→  /** Render one frame through the composer (or fallback direct render). */
   464→   464→  render(): void {
   465→   465→    if (this._composer) {
   466→   466→      this._composer.render();
   467→   467→    } else {
   468→   468→      this._sm.renderer.render(this._sm.scene, this._sm.camera);
   469→   469→    }
   470→   470→  }
   471→   471→
   472→   472→  /** Resize the composer and internal passes. */
   473→   473→  resize(): void {
   474→   474→    if (!this._composer) return;
   475→   475→    const r = this._sm.renderer;
   476→   476→    const size = r.getSize(new THREE.Vector2());
   477→   477→    const w = size.x || 2;
   478→   478→    const h = size.y || 2;
   479→   479→    this._composer.setSize(w, h);
   480→   480→
   481→   481→    if (this._bloomPass) {
   482→   482→      this._bloomPass.resolution.set(w / 2, h / 2);
   483→   483→    }
   484→   484→    if (this._ssaoPass) {
   485→   485→      this._ssaoPass.setSize(w, h);
   486→   486→    }
   487→   487→    if (this._smaaPass) {
   488→   488→      this._smaaPass.setSize(w, h);
   489→   489→    }
   490→   490→    if (this._fxaaPass) {
   491→   491→      const mat = this._fxaaPass.material as THREE.ShaderMaterial;
   492→   492→      const res = mat.uniforms['resolution'];
   493→   493→      if (res && res.value instanceof THREE.Vector2) {
   494→   494→        res.value.set(1 / w, 1 / h);
   495→   495→      }
   496→   496→    }
   497→   497→  }
   498→   498→
   499→   499→  /** Update the full pipeline config. Rebuilds composer if structural changes detected. */
   500→   500→  updateConfig(cfg: Partial<PipelineConfig>): void {
   501→   501→    const prev = { ...this._config };
   502→   502→    this._config = { ...this._config, ...cfg };
   503→   503→
   504→   504→    const structuralChange =
   505→   505→      cfg.bloom?.enabled !== undefined && cfg.bloom.enabled !== prev.bloom.enabled ||
   506→   506→      cfg.antialiasing !== undefined && cfg.antialiasing !== prev.antialiasing ||
   507→   507→      cfg.vignette?.enabled !== undefined && cfg.vignette.enabled !== prev.vignette.enabled ||
   508→   508→      cfg.ao?.enabled !== undefined && cfg.ao.enabled !== prev.ao.enabled ||
   509→   509→      cfg.colorGrading?.enabled !== undefined && cfg.colorGrading.enabled !== prev.colorGrading.enabled;
   510→   510→
   511→   511→    if (structuralChange) {
   512→   512→      this.build();
   513→   513→      return;
   514→   514→    }
   515→   515→
   516→   516→    // In-place parameter updates (no rebuild needed)
   517→   517→    if (this._bloomPass && cfg.bloom) {
   518→   518→      this._bloomPass.strength = cfg.bloom.intensity ?? this._bloomPass.strength;
   519→   519→      this._bloomPass.radius = cfg.bloom.radius ?? this._bloomPass.radius;
   520→   520→      this._bloomPass.threshold = cfg.bloom.threshold ?? this._bloomPass.threshold;
   521→   521→    }
   522→   522→
   523→   523→    if (this._ssaoPass && cfg.ao) {
   524→   524→      this._applyAOParams(
   525→   525→        cfg.ao.radius ?? this._config.ao.radius,
   526→   526→        cfg.ao.intensity ?? this._config.ao.intensity,
   527→   527→      );
   528→   528→    }
   529→   529→
   530→   530→    if (this._vignettePass && cfg.vignette) {
   531→   531→      const mat = this._vignettePass.material as THREE.ShaderMaterial;
   532→   532→      mat.uniforms['intensity'].value = cfg.vignette.intensity ?? 0.4;
   533→   533→    }
   534→   534→
   535→   535→    if (this._colorGradingPass && cfg.colorGrading) {
   536→   536→      const mat = this._colorGradingPass.material as THREE.ShaderMaterial;
   537→   537→      if (cfg.colorGrading.brightness !== undefined) mat.uniforms['brightness'].value = cfg.colorGrading.brightness;
   538→   538→      if (cfg.colorGrading.contrast !== undefined) mat.uniforms['contrast'].value = cfg.colorGrading.contrast;
   539→   539→      if (cfg.colorGrading.saturation !== undefined) mat.uniforms['saturation'].value = cfg.colorGrading.saturation;
   540→   540→    }
   541→   541→
   542→   542→    if (cfg.tonemapping !== undefined) {
   543→   543→      this._applyToneMapping(cfg.tonemapping);
   544→   544→    }
   545→   545→    if (cfg.exposure !== undefined) {
   546→   546→      this._sm.renderer.toneMappingExposure = cfg.exposure;
   547→   547→    }
   548→   548→    if (cfg.shadowQuality !== undefined) {
   549→   549→      this._applyShadowQuality(cfg.shadowQuality);
   550→   550→    }
   551→   551→    if (cfg.quality !== undefined) {
   552→   552→      this._applyQuality(cfg.quality);
   553→   553→    }
   554→   554→  }
   555→   555→

   621→   621→  /** Dispose all passes and the composer. */
   622→   622→  dispose(): void {
   623→   623→    if (this._composer) {
   624→   624→      const passes = this._composer.passes.slice().reverse();
   625→   625→      for (const pass of passes) {
   626→   626→        if ('dispose' in pass && typeof (pass as { dispose: () => void }).dispose === 'function') {
   627→   627→          (pass as { dispose: () => void }).dispose();
   628→   628→        }
   629→   629→      }
   630→   630→      this._composer.dispose();
   631→   631→    }
   632→   632→    this._composer = null;
   633→   633→    this._renderPass = null;
   634→   634→    this._bloomPass = null;
   635→   635→    this._ssaoPass = null;
   636→   636→    this._smaaPass = null;
   637→   637→    this._fxaaPass = null;
   638→   638→    this._vignettePass = null;
   639→   639→    this._colorGradingPass = null;
   640→   640→  }

   642→   642→  // ── Private helpers ───────────────────────────────────────────────────────
   643→   643→
   644→   644→  /** Apply SSAO parameters — radius controls kernel spread, intensity controls maxDistance scaling. */
   645→   645→  private _applyAOParams(radius: number, intensity: number): void {
   646→   646→    if (!this._ssaoPass) return;
   647→   647→    this._ssaoPass.kernelRadius = radius;
   648→   648→    this._ssaoPass.minDistance = 0.001;
   649→   649→    // Scale maxDistance with intensity: higher intensity → AO visible at greater depth range
   650→   650→    this._ssaoPass.maxDistance = 0.02 + intensity * 0.08;
   651→   651→  }
   652→   652→
   653→   653→  private _applyToneMapping(mapping: 'aces' | 'reinhard' | 'linear'): void {
   654→   654→    const r = this._sm.renderer;
   655→   655→    switch (mapping) {
   656→   656→      case 'aces':
   657→   657→        r.toneMapping = THREE.ACESFilmicToneMapping;
   658→   658→        break;
   659→   659→      case 'reinhard':
   660→   660→        r.toneMapping = THREE.ReinhardToneMapping;
   661→   661→        break;
   662→   662→      case 'linear':
   663→   663→        r.toneMapping = THREE.LinearToneMapping;
   664→   664→        break;
   665→   665→    }
   666→   666→    r.toneMappingExposure = this._config.exposure;
   667→   667→  }
   668→   668→
   669→   669→  private _applyShadowQuality(quality: string): void {
   670→   670→    const r = this._sm.renderer;
   671→   671→    switch (quality) {
   672→   672→      case 'none':
   673→   673→        r.shadowMap.enabled = false;
   674→   674→        break;
   675→   675→      case 'low':
   676→   676→        r.shadowMap.enabled = true;
   677→   677→        r.shadowMap.type = THREE.BasicShadowMap;
   678→   678→        break;
   679→   679→      case 'medium':
   680→   680→        r.shadowMap.enabled = true;
   681→   681→        r.shadowMap.type = THREE.PCFShadowMap;
   682→   682→        break;
   683→   683→      case 'high':
   684→   684→      default:
   685→   685→        r.shadowMap.enabled = true;
   686→   686→        r.shadowMap.type = THREE.PCFSoftShadowMap;
   687→   687→        break;
   688→   688→    }
   689→   689→
   690→   690→    // Update shadow map sizes on all lights
   691→   691→    this._sm.scene.traverse((child) => {
   692→   692→      const light = child as THREE.DirectionalLight | THREE.SpotLight | THREE.PointLight;
   693→   693→      if (light.isLight && light.shadow) {
   694→   694→        const shadow = light.shadow;
   695→   695→        switch (quality) {
   696→   696→          case 'low':
   697→   697→            shadow.mapSize.set(512, 512);
   698→   698→            break;
   699→   699→          case 'medium':
   700→   700→            shadow.mapSize.set(1024, 1024);
   701→   701→            break;
   702→   702→          case 'high':
   703→   703→          default:
   704→   704→            shadow.mapSize.set(2048, 2048);
   705→   705→            break;
   706→   706→        }
   707→   707→        if (shadow.map) {
   708→   708→          shadow.map.dispose();
   709→   709→          shadow.map = null;
   710→   710→        }
   711→   711→      }
   712→   712→    });
   713→   713→  }
   714→   714→
   715→   715→  private _applyQuality(quality: 'low' | 'medium' | 'high' | 'ultra'): void {
   716→   716→    const r = this._sm.renderer;
   717→   717→    switch (quality) {
   718→   718→      case 'low':
   719→   719→        r.setPixelRatio(1);
   720→   720→        break;
   721→   721→      case 'medium':
   722→   722→        r.setPixelRatio(Math.min(window.devicePixelRatio, 1.5));
   723→   723→        break;
   724→   724→      case 'high':
   725→   725→        r.setPixelRatio(Math.min(window.devicePixelRatio, 2));
   726→   726→        break;
   727→   727→      case 'ultra':
   728→   728→        r.setPixelRatio(window.devicePixelRatio);
   729→   729→        break;
   730→   730→    }
   731→   731→    this._sm.resize();
   732→   732→    if (this._composer) {
   733→   733→      this.resize();
   734→   734→    }
   735→   735→  }

   738→   738→interface LightSyncEntry {
   739→   739→  id: string;
   740→   740→  type: string;
   741→   741→  color: string;
   742→   742→  brightness: number;
   743→   743→  opacity: number;
   744→   744→  visible: boolean;
   745→   745→  solo: boolean;
   746→   746→  falloff: string;
   747→   747→  gearVisible: boolean;
   748→   748→  transform: {
   749→   749→    position: { x: number; y: number; z: number };
   750→   750→    spherical: { lat: number; lng: number; radius: number; height: number };
   751→   751→    rotation: { x: number; y: number; z: number; enabled: boolean };
   752→   752→  };
   753→   753→  // Spotlight-specific
   754→   754→  spotAngle?: number;
   755→   755→  spotPenumbra?: number;
   756→   756→  spotDecay?: number;
   757→   757→  // Area light specific
   758→   758→  areaWidth?: number;
   759→   759→  areaHeight?: number;
   760→   760→}
   761→   761→
   762→   762→interface LightObjectEntry {
   763→   763→  object: THREE.Object3D;
   764→   764→  lightType: string;
   765→   765→  target?: THREE.Object3D;
   766→   766→}
   767→   767→
   768→   768→export class LightManager {
   769→   769→  private _scene: THREE.Scene;
   770→   770→  private _entries: Map<string, LightObjectEntry> = new Map();
   771→   771→  private _helpers: Map<string, THREE.Object3D> = new Map();
   772→   772→  private _types: Map<string, string> = new Map();
   773→   773→
   774→   774→  constructor(scene: THREE.Scene) {
   775→   775→    this._scene = scene;
   776→   776→    ensureRectAreaLib();
   777→   777→  }
   778→   778→
   779→   779→  syncLights(lights: LightSyncEntry[]): void {
   780→   780→    const activeIds = new Set(lights.map((l) => l.id));
   781→   781→    const hasSolo = lights.some((l) => l.solo);
   782→   782→
   783→   783→    // Remove stale lights
   784→   784→    for (const [id] of this._entries) {
   785→   785→      if (!activeIds.has(id)) {
   786→   786→        this._removeLight(id);
   787→   787→      }
   788→   788→    }
   789→   789→
   790→   790→    // Create or update each light
   791→   791→    for (const ld of lights) {
   792→   792→      const isEffectiveSolo = hasSolo ? ld.solo : true;
   793→   793→      const shouldShow = ld.visible && isEffectiveSolo;
   794→   794→
   795→   795→      const existing = this._entries.get(ld.id);
   796→   796→      const currentType = this._types.get(ld.id);
   797→   797→
   798→   798→      // If type changed, recreate the light
   799→   799→      if (existing && currentType !== ld.type) {
   800→   800→        this._removeLight(ld.id);
   801→   801→      }
   802→   802→
   803→   803→      if (!this._entries.has(ld.id)) {
   804→   804→        this._createLight(ld);
   805→   805→      }
   806→   806→
   807→   807→      this._updateLight(ld, shouldShow);
   808→   808→    }
   809→   809→  }
   810→   810→
   811→   811→  private _createLight(ld: LightSyncEntry): void {
   812→   812→    ensureRectAreaLib();
   813→   813→    const lightObj = this._createLightByType(ld.type);
   814→   814→
   815→   815→    // For spot/directional, add a target object
   816→   816→    let target: THREE.Object3D | undefined;
   817→   817→    if (lightObj instanceof THREE.SpotLight || lightObj instanceof THREE.DirectionalLight) {
   818→   818→      target = new THREE.Object3D();
   819→   819→      target.position.set(0, 0, 0);
   820→   820→      this._scene.add(target);
   821→   821→      lightObj.target = target;
   822→   822→    }
   823→   823→
   824→   824→    this._scene.add(lightObj);
   825→   825→    this._entries.set(ld.id, { object: lightObj, lightType: ld.type, target });
   826→   826→    this._types.set(ld.id, ld.type);
   827→   827→
   828→   828→    // Create helper
   829→   829→    const helper = this._createHelper(ld.type, lightObj as THREE.Light);
   830→   830→    if (helper) {
   831→   831→      this._helpers.set(ld.id, helper);
   832→   832→      this._scene.add(helper);
   833→   833→    }
   834→   834→  }
   835→   835→
   836→   836→  private _updateLight(ld: LightSyncEntry, shouldShow: boolean): void {
   837→   837→    const entry = this._entries.get(ld.id);
   838→   838→    if (!entry) return;
   839→   839→
   840→   840→    const { object: lightObj } = entry;
   841→   841→
   842→   842→    // Compute position from spherical
   843→   843→    const s = ld.transform.spherical;
   844→   844→    const latRad = (s.lat * Math.PI) / 180;
   845→   845→    const lngRad = (s.lng * Math.PI) / 180;
   846→   846→    const px = s.radius * Math.cos(latRad) * Math.cos(lngRad);
   847→   847→    const py = s.height;
   848→   848→    const pz = s.radius * Math.cos(latRad) * Math.sin(lngRad);
   849→   849→
   850→   850→    lightObj.position.set(px, py, pz);
   851→   851→    lightObj.visible = shouldShow;
   852→   852→
   853→   853→    // Apply rotation if enabled
   854→   854→    if (ld.transform.rotation.enabled) {
   855→   855→      lightObj.rotation.set(
   856→   856→        (ld.transform.rotation.x * Math.PI) / 180,
   857→   857→        (ld.transform.rotation.y * Math.PI) / 180,
   858→   858→        (ld.transform.rotation.z * Math.PI) / 180,
   859→   859→      );
   860→   860→    }
   861→   861→
   862→   862→    if (lightObj instanceof THREE.Light) {
   863→   863→      lightObj.color.set(ld.color);
   864→   864→      // Map brightness 0-1000 to intensity 0-10
   865→   865→      lightObj.intensity = (ld.brightness / 1000) * 10;
   866→   866→
   867→   867→      // Apply opacity (only meaningful via distance/decay for physically-based)
   868→   868→      // We scale intensity by opacity/100 to simulate dimming
   869→   869→      if (ld.opacity < 100) {
   870→   870→        lightObj.intensity *= ld.opacity / 100;
   871→   871→      }
   872→   872→
   873→   873→      // Falloff
   874→   874→      if (lightObj instanceof THREE.PointLight || lightObj instanceof THREE.SpotLight) {
   875→   875→        switch (ld.falloff) {
   876→   876→          case 'linear':
   877→   877→            lightObj.decay = 1;
   878→   878→            break;
   879→   879→          case 'none':
   880→   880→            lightObj.decay = 0;
   881→   881→            break;
   882→   882→          case 'custom':
   883→   883→            lightObj.decay = 1.5;
   884→   884→            break;
   885→   885→          case 'quadratic':
   886→   886→          default:
   887→   887→            lightObj.decay = 2;
   888→   888→            break;
   889→   889→        }
   890→   890→      }
   891→   891→
   892→   892→      // Spotlight-specific
   893→   893→      if (lightObj instanceof THREE.SpotLight) {
   894→   894→        if (ld.spotAngle !== undefined) {
   895→   895→          lightObj.angle = (ld.spotAngle * Math.PI) / 180;
   896→   896→        }
   897→   897→        if (ld.spotPenumbra !== undefined) {
   898→   898→          lightObj.penumbra = ld.spotPenumbra;
   899→   899→        }
   900→   900→        if (ld.spotDecay !== undefined) {
   901→   901→          lightObj.decay = ld.spotDecay;
   902→   902→        }
   903→   903→      }
   904→   904→
   905→   905→      // Area light dimensions
   906→   906→      if (lightObj instanceof THREE.RectAreaLight) {
   907→   907→        if (ld.areaWidth !== undefined) lightObj.width = ld.areaWidth;
   908→   908→        if (ld.areaHeight !== undefined) lightObj.height = ld.areaHeight;
   909→   909→      }
   910→   910→
   911→   911→      // Shadow config
   912→   912→      this._configureShadows(lightObj, ld.type);
   913→   913→    }
   914→   914→
   915→   915→    // Update target to look at origin for spot/directional
   916→   916→    if (entry.target) {
   917→   917→      entry.target.position.set(0, 0, 0);
   918→   918→    }
   919→   919→
   920→   920→    // Update helper
   921→   921→    const helper = this._helpers.get(ld.id);
   922→   922→    if (helper) {
   923→   923→      helper.visible = shouldShow && ld.gearVisible;
   924→   924→      helper.position.copy(lightObj.position);
   925→   925→      helper.rotation.copy(lightObj.rotation);
   926→   926→      if (helper instanceof THREE.Mesh && lightObj instanceof THREE.Light) {
   927→   927→        (helper.material as THREE.MeshBasicMaterial).color.copy(lightObj.color);
   928→   928→      }
   929→   929→    }
   930→   930→  }
   931→   931→
   932→   932→  private _createLightByType(type: string): THREE.Object3D {
   933→   933→    const color = 0xffffff;
   934→   934→    const intensity = 1;
   935→   935→    switch (type) {
   936→   936→      case 'directional': {
   937→   937→        const dl = new THREE.DirectionalLight(color, intensity);
   938→   938→        dl.castShadow = true;
   939→   939→        dl.shadow.mapSize.set(2048, 2048);
   940→   940→        dl.shadow.camera.near = 0.5;
   941→   941→        dl.shadow.camera.far = 50;

  1020→  1020→  private _createHelper(type: string, light: THREE.Light): THREE.Object3D | null {
  1021→  1021→    const isArea = type === 'area' || type === 'overhead';
  1022→  1022→    let geo: THREE.BufferGeometry;
  1023→  1023→    if (isArea) {
  1024→  1024→      geo = new THREE.PlaneGeometry(0.3, 0.3);
  1025→  1025→    } else if (type === 'directional') {
  1026→  1026→      // Use a slightly larger sphere for directional
  1027→  1027→      geo = new THREE.SphereGeometry(0.1, 12, 12);
  1028→  1028→    } else {
  1029→  1029→      geo = new THREE.SphereGeometry(0.06, 10, 10);
  1030→  1030→    }
  1031→  1031→    const mat = new THREE.MeshBasicMaterial({
  1032→  1032→      color: light.color,
  1033→  1033→      transparent: true,
  1034→  1034→      opacity: 0.7,
  1035→  1035→      depthTest: false,
  1036→  1036→    });
  1037→  1037→    const mesh = new THREE.Mesh(geo, mat);
  1038→  1038→    mesh.userData.isHelper = true;
  1039→  1039→    mesh.renderOrder = 999;
  1040→  1040→    return mesh;
  1041→  1041→  }

  1043→  1043→  private _removeLight(id: string): void {
  1044→  1044→    const entry = this._entries.get(id);
  1045→  1045→    if (!entry) return;
  1046→  1046→
  1047→  1047→    const { object, target } = entry;
  1048→  1048→
  1049→  1049→    // Dispose shadow maps
  1050→  1050→    const shadowLight = object as THREE.SpotLight | THREE.DirectionalLight | THREE.PointLight;
  1051→  1051→    if (object instanceof THREE.Light && shadowLight.shadow && shadowLight.shadow.map) {
  1052→  1052→      shadowLight.shadow.map.dispose();
  1053→  1053→    }
  1054→  1054→
  1055→  1055→    this._scene.remove(object);
  1056→  1056→    this._disposeObject(object);
  1057→  1057→    if (target) {
  1058→  1058→      this._scene.remove(target);
  1059→  1059→    }
  1060→  1060→
  1061→  1061→    this._entries.delete(id);
  1062→  1062→    this._types.delete(id);
  1063→  1063→
  1064→  1064→    const helper = this._helpers.get(id);
  1065→  1065→    if (helper) {
  1066→  1066→      this._scene.remove(helper);
  1067→  1067→      this._disposeObject(helper);
  1068→  1068→      this._helpers.delete(id);
  1069→  1069→    }
  1070→  1070→  }
  1071→  1071→
  1072→  1072→  private _disposeObject(obj: THREE.Object3D): void {
  1073→  1073→    obj.traverse((child) => {
  1074→  1074→      if (child instanceof THREE.Mesh) {
  1075→  1075→        child.geometry.dispose();
  1076→  1076→        if (child.material instanceof THREE.Material) {
  1077→  1077→          child.material.dispose();
  1078→  1078→        }
  1079→  1079→      }
  1080→  1080→    });
  1081→  1081→  }
  1082→  1082→
  1083→  1083→  getLightObject(id: string): THREE.Object3D | undefined {
  1084→  1084→    return this._entries.get(id)?.object;
  1085→  1085→  }
  1086→  1086→
  1087→  1087→  getLightCount(): number {
  1088→  1088→    return this._entries.size;
  1089→  1089→  }
  1090→  1090→
  1091→  1091→  dispose(): void {
  1092→  1092→    const ids = Array.from(this._entries.keys());
  1093→  1093→    for (const id of ids) {
  1094→  1094→      this._removeLight(id);
  1095→  1095→    }
  1096→  1096→  }

  1130→  1130→  async loadFromFile(file: File): Promise<void> {
  1131→  1131→    if (this._loading) return;
  1132→  1132→    this._loading = true;
  1133→  1133→    this._progress = 0;
  1134→  1134→    this._onProgress?.(0);
  1135→  1135→
  1136→  1136→    // Read file as ArrayBuffer to store raw data for scene persistence
  1137→  1137→    let arrayBuffer: ArrayBuffer | null = null;
  1138→  1138→    try {
  1139→  1139→      arrayBuffer = await file.arrayBuffer();
  1140→  1140→      setRawModelData(arrayBuffer, file.name);
  1141→  1141→    } catch {
  1142→  1142→      // Non-critical — model will load but won't be saveable
  1143→  1143→    }
  1144→  1144→
  1145→  1145→    const url = URL.createObjectURL(file);
  1146→  1146→
  1147→  1147→    try {
  1148→  1148→      const gltf = await new Promise<{ scene: THREE.Group }>((resolve, reject) => {
  1149→  1149→        this._loader.load(
  1150→  1150→          url,
  1151→  1151→          (result) => resolve(result),
  1152→  1152→          (event) => {
  1153→  1153→            if (event.lengthComputable) {
  1154→  1154→              this._progress = (event.loaded / event.total) * 100;
  1155→  1155→              this._onProgress?.(this._progress);
  1156→  1156→            }
  1157→  1157→          },
  1158→  1158→          (error) => reject(error)
  1159→  1159→        );
  1160→  1160→      });
  1161→  1161→
  1162→  1162→      this.removeCurrentModel();
  1163→  1163→
  1164→  1164→      this._currentModel = gltf.scene;
  1165→  1165→      this._scene.add(gltf.scene);
  1166→  1166→
  1167→  1167→      // Enable shadows
  1168→  1168→      gltf.scene.traverse((child) => {
  1169→  1169→        if (child instanceof THREE.Mesh) {
  1170→  1170→          child.castShadow = true;
  1171→  1171→          child.receiveShadow = true;
  1172→  1172→        }
  1173→  1173→      });
  1174→  1174→
  1175→  1175→      // Center and scale
  1176→  1176→      const box = new THREE.Box3().setFromObject(gltf.scene);
  1177→  1177→      const center = box.getCenter(new THREE.Vector3());
  1178→  1178→      const size = box.getSize(new THREE.Vector3());
  1179→  1179→      const maxDim = Math.max(size.x, size.y, size.z);
  1180→  1180→      if (maxDim > 0) {
  1181→  1181→        const scale = 4 / maxDim;
  1182→  1182→        gltf.scene.scale.multiplyScalar(scale);
  1183→  1183→      }
  1184→  1184→      // Recompute after scaling
  1185→  1185→      const box2 = new THREE.Box3().setFromObject(gltf.scene);
  1186→  1186→      const center2 = box2.getCenter(new THREE.Vector3());
  1187→  1187→      gltf.scene.position.sub(center2);
  1188→  1188→      gltf.scene.position.y += box2.getSize(new THREE.Vector3()).y / 2;
  1189→  1189→
  1190→  1190→      this._loading = false;
  1191→  1191→      this._progress = 100;
  1192→  1192→      this._onProgress?.(100);
  1193→  1193→      this._onLoaded?.(file.name.replace(/\.(glb|gltf)$/i, ''));
  1194→  1194→    } catch (err) {
  1195→  1195→      this._loading = false;
  1196→  1196→      this._progress = 0;
  1197→  1197→      const msg = err instanceof Error ? err.message : 'Failed to load model';
  1198→  1198→      this._onError?.(msg);
  1199→  1199→    } finally {
  1200→  1200→      URL.revokeObjectURL(url);
  1201→  1201→    }
  1202→  1202→  }
