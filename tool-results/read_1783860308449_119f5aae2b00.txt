   133→   133→  /** Remove existing ground objects from the scene and dispose their resources. */
   134→   134→  private _disposeGround(): void {
   135→   135→    if (this.groundOverlay) {
   136→   136→      this.scene.remove(this.groundOverlay);
   137→   137→      this.groundOverlay.geometry.dispose();
   138→   138→      (this.groundOverlay.material as THREE.Material).dispose();
   139→   139→      this.groundOverlay = null;
   140→   140→    }
   141→   141→    if (this.ground) {
   142→   142→      this.scene.remove(this.ground);
   143→   143→      this.ground.geometry.dispose();
   144→   144→      if (Array.isArray(this.ground.material)) {
   145→   145→        this.ground.material.forEach((m) => m.dispose());
   146→   146→      } else {
   147→   147→        this.ground.material.dispose();
   148→   148→      }
   149→   149→      this.ground = null;
   150→   150→    }
   151→   151→    if (this._floorCubeCamera) {
   152→   152→      this._floorCubeCamera.dispose();
   153→   153→      this._floorCubeCamera = null;
   154→   154→    }
   155→   155→    if (this._floorCubeRT) {
   156→   156→      this._floorCubeRT.dispose();
   157→   157→      this._floorCubeRT = null;
   158→   158→    }
   159→   159→    this._floorMaterial = null;
   160→   160→  }
   161→   161→
   162→   162→  /**
   163→   163→   * Rebuild the ground plane according to the given settings.
   164→   164→   * Always uses MeshStandardMaterial (PBR) so roughness/metalness work correctly.
   165→   165→   * - reflections=true  → CubeCamera real-time reflections via envMap
   166→   166→   * - reflections=false → Scene environment only (or none)
   167→   167→   */
   168→   168→  updateGround(settings?: GroundSettings | null): void {
   169→   169→    if (!settings) return;
   170→   170→    // Merge with defaults so old scene files missing new fields don't crash
   171→   171→    const merged: GroundSettings = {
   172→   172→      visible: true,
   173→   173→      reflections: true,
   174→   174→      reflectionSharpness: 0.85,
   175→   175→      color: '#111118',
   176→   176→      roughness: 0.15,
   177→   177→      metalness: 0.95,
   178→   178→      fadeRadius: 8.0,
   179→   179→      ...settings,
   180→   180→    };
   181→   181→    this._groundSettings = merged;
   182→   182→    this._disposeGround();
   183→   183→
   184→   184→    if (!merged.visible) return;
   185→   185→
   186→   186→    const GROUND_SIZE = 40;
   187→   187→    const groundGeo = new THREE.PlaneGeometry(GROUND_SIZE, GROUND_SIZE);
   188→   188→    const color = new THREE.Color(merged.color);
   189→   189→
   190→   190→    // ── PBR ground material (always MeshStandardMaterial) ──
   191→   191→    const groundMat = new THREE.MeshStandardMaterial({
   192→   192→      color: color.getHex(),
   193→   193→      metalness: merged.metalness,
   194→   194→      roughness: merged.roughness,
   195→   195→      envMapIntensity: 1.0,
   196→   196→      side: THREE.FrontSide,
   197→   197→    });
   198→   198→    this._floorMaterial = groundMat;
   199→   199→
   200→   200→    if (merged.reflections) {
   201→   201→      // ── CubeCamera for real-time planar reflections ──
   202→   202→      const dpr = Math.min(window.devicePixelRatio, 2);
   203→   203→      const cubeRTSize = Math.max(128, Math.round(512 * dpr));
   204→   204→
   205→   205→      this._floorCubeRT = new THREE.WebGLCubeRenderTarget(cubeRTSize, {
   206→   206→        generateMipmaps: true,
   207→   207→        minFilter: THREE.LinearMipmapLinearFilter,
   208→   208→        magFilter: THREE.LinearFilter,
   209→   209→      });
   210→   210→
   211→   211→      this._floorCubeCamera = new THREE.CubeCamera(0.1, 100, this._floorCubeRT);
   212→   212→      this._floorCubeCamera.position.set(0, 0.01, 0); // slightly above floor
   213→   213→      this.scene.add(this._floorCubeCamera);
   214→   214→
   215→   215→      // Use CubeCamera texture as envMap; roughness controls blur via mip levels
   216→   216→      groundMat.envMap = this._floorCubeRT.texture;
   217→   217→      groundMat.envMapIntensity = merged.reflectionSharpness;
   218→   218→    } else {
   219→   219→      // No real-time reflections — use scene environment map if available
   220→   220→      if (this.scene.environment) {
   221→   221→        groundMat.envMap = this.scene.environment;
   222→   222→      }
   223→   223→      groundMat.envMapIntensity = 0.5;
   224→   224→    }
   225→   225→
   226→   226→    this.ground = new THREE.Mesh(groundGeo, groundMat);
   227→   227→    this.ground.rotation.x = -Math.PI / 2;
   228→   228→    this.ground.position.y = -0.005;
   229→   229→    this.ground.receiveShadow = true;
   230→   230→    this.ground.name = '__floor__';
   231→   231→    this.scene.add(this.ground);
   232→   232→
   233→   233→    // ── Fade overlay: fades ground edges into background ──
   234→   234→    if (merged.fadeRadius > 0) {
   235→   235→      const overlayGeo = new THREE.PlaneGeometry(GROUND_SIZE, GROUND_SIZE);
   236→   236→      const overlayMat = new THREE.ShaderMaterial({
   237→   237→        transparent: true,
   238→   238→        depthWrite: false,
   239→   239→        uniforms: {
   240→   240→          uFadeRadius: { value: merged.fadeRadius },
   241→   241→          uGroundSize: { value: GROUND_SIZE / 2 },
   242→   242→          uBgColor: { value: new THREE.Color('#0d0d1a') },
   243→   243→        },
   244→   244→        vertexShader: /* glsl */ `
   245→   245→          varying vec2 vWorldPos;
   246→   246→          void main() {
   247→   247→            vec4 worldPos = modelMatrix * vec4(position, 1.0);
   248→   248→            vWorldPos = worldPos.xz;
   249→   249→            gl_Position = projectionMatrix * viewMatrix * worldPos;
   250→   250→          }
   251→   251→        `,
   252→   252→        fragmentShader: /* glsl */ `
   253→   253→          uniform float uFadeRadius;
   254→   254→          uniform float uGroundSize;
   255→   255→          uniform vec3 uBgColor;
   256→   256→          varying vec2 vWorldPos;
   257→   257→          void main() {
   258→   258→            float dist = length(vWorldPos);
   259→   259→            float fade = 1.0 - smoothstep(uFadeRadius, uGroundSize, dist);
   260→   260→            gl_FragColor = vec4(uBgColor, 1.0 - fade);
   261→   261→          }
   262→   262→        `,
   263→   263→      });
   264→   264→      this.groundOverlay = new THREE.Mesh(overlayGeo, overlayMat);
   265→   265→      this.groundOverlay.rotation.x = -Math.PI / 2;
   266→   266→      this.groundOverlay.position.y = -0.003;
   267→   267→      this.scene.add(this.groundOverlay);
   268→   268→    }
   269→   269→  }

   298→   298→  animateCameraTo(
   299→   299→    position: [number, number, number],
   300→   300→    target: [number, number, number],
   301→   301→    duration: number = 600
   302→   302→  ): void {
   303→   303→    const startPos = this.camera.position.clone();
   304→   304→    const startTarget = this.controls.target.clone();
   305→   305→    const endPos = new THREE.Vector3(...position);
   306→   306→    const endTarget = new THREE.Vector3(...target);
   307→   307→    const startTime = performance.now();
   308→   308→
   309→   309→    const animate = () => {
   310→   310→      const elapsed = performance.now() - startTime;
   311→   311→      const t = Math.min(elapsed / duration, 1);
   312→   312→      const ease = t < 0.5 ? 4 * t * t * t : 1 - Math.pow(-2 * t + 2, 3) / 2;
   313→   313→
   314→   314→      this.camera.position.lerpVectors(startPos, endPos, ease);
   315→   315→      this.controls.target.lerpVectors(startTarget, endTarget, ease);
   316→   316→      this.controls.update();
   317→   317→
   318→   318→      if (t < 1) {
   319→   319→        requestAnimationFrame(animate);
   320→   320→      }
   321→   321→    };
   322→   322→    requestAnimationFrame(animate);
   323→   323→  }

   329→   329→  startRenderLoop(): void {
   330→   330→    const loop = () => {
   331→   331→      this._animationId = requestAnimationFrame(loop);
   332→   332→      const delta = this._clock.getDelta();
   333→   333→
   334→   334→      // Turntable rotation
   335→   335→      if (this._turntableActive) {
   336→   336→        const model = this._findModel();
   337→   337→        if (model) {
   338→   338→          model.rotation.y += delta * this._turntableSpeed * 0.5;
   339→   339→        }
   340→   340→      }
   341→   341→
   342→   342→      // Update CubeCamera for PBR floor reflections (hide floor to avoid self-reflection)
   343→   343→      if (this._floorCubeCamera && this.ground && this._groundSettings?.reflections) {
   344→   344→        this.ground.visible = false;
   345→   345→        this._floorCubeCamera.update(this.renderer, this.scene);
   346→   346→        this.ground.visible = true;
   347→   347→      }
   348→   348→
   349→   349→      if (this._onFrame) this._onFrame(delta);
   350→   350→      this.controls.update();
   351→   351→      this.renderer.render(this.scene, this.camera);
   352→   352→    };
   353→   353→    loop();
   354→   354→  }

   356→   356→  _findModel(): THREE.Object3D | null {

   362→   362→        child !== this.ground &&
   363→   363→        child !== this._floorCubeCamera
   364→   364→      ) {
   365→   365→        return child;
   366→   366→      }
   367→   367→    }
   368→   368→    return null;
   369→   369→  }

   407→   407→    this.scene.traverse((obj) => {

   411→   411→          obj.material.forEach((m) => m.dispose());
   412→   412→        } else {
   413→   413→          obj.material.dispose();
   414→   414→        }
   415→   415→      }
   416→   416→    });
   417→   417→    this.controls.dispose();
   418→   418→    this.renderer.dispose();
   419→   419→  }

   422→   422→// ── Vignette shader (custom) ────────────────────────────────────────────────
   423→   423→const VignetteShader = {
   424→   424→  uniforms: {
   425→   425→    tDiffuse: { value: null as THREE.Texture | null },
   426→   426→    intensity: { value: 0.4 },
   427→   427→    offset: { value: 1.1 },
   428→   428→    darkness: { value: 1.2 },
   429→   429→  },
   430→   430→  vertexShader: /* glsl */ `
   431→   431→    varying vec2 vUv;
   432→   432→    void main() {
   433→   433→      vUv = uv;
   434→   434→      gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
   435→   435→    }
   436→   436→  `,
   437→   437→  fragmentShader: /* glsl */ `
   438→   438→    uniform sampler2D tDiffuse;
   439→   439→    uniform float intensity;
   440→   440→    uniform float offset;
   441→   441→    uniform float darkness;
   442→   442→    varying vec2 vUv;
   443→   443→    void main() {
   444→   444→      vec4 texel = texture2D(tDiffuse, vUv);
   445→   445→      vec2 uv = (vUv - vec2(0.5)) * vec2(offset);
   446→   446→      float vig = clamp(1.0 - dot(uv, uv), 0.0, 1.0);
   447→   447→      float f = mix(1.0, vig, intensity * darkness);
   448→   448→      gl_FragColor = vec4(texel.rgb * f, texel.a);
   449→   449→    }
   450→   450→  `,
   451→   451→};
   452→   452→
   453→   453→// ── Color Grading shader (brightness / contrast / saturation) ───────────────
   454→   454→const ColorGradingShader = {
   455→   455→  uniforms: {
   456→   456→    tDiffuse: { value: null as THREE.Texture | null },
   457→   457→    brightness: { value: 0.0 },
   458→   458→    contrast: { value: 0.0 },
   459→   459→    saturation: { value: 0.0 },
   460→   460→  },
   461→   461→  vertexShader: /* glsl */ `
   462→   462→    varying vec2 vUv;

   464→   464→      vUv = uv;
   465→   465→      gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);

   467→   467→  `,

   469→   469→    uniform sampler2D tDiffuse;
   470→   470→    uniform float brightness;
   471→   471→    uniform float contrast;
   472→   472→    uniform float saturation;
   473→   473→    varying vec2 vUv;
   474→   474→
   475→   475→    void main() {
   476→   476→      vec4 texel = texture2D(tDiffuse, vUv);
   477→   477→      vec3 c = texel.rgb;
   478→   478→
   479→   479→      // Brightness
   480→   480→      c += brightness * 0.5;
   481→   481→
   482→   482→      // Contrast
   483→   483→      c = (c - 0.5) * (1.0 + contrast) + 0.5;
   484→   484→
   485→   485→      // Saturation
   486→   486→      float lum = dot(c, vec3(0.2126, 0.7152, 0.0722));
   487→   487→      c = mix(vec3(lum), c, 1.0 + saturation);
   488→   488→
   489→   489→      gl_FragColor = vec4(clamp(c, 0.0, 1.0), texel.a);
   490→   490→    }
   491→   491→  `,
   492→   492→};
   493→   493→
   494→   494→// ── Pipeline settings interface ─────────────────────────────────────────────
   495→   495→export interface PipelineConfig {
   496→   496→  bloom: { enabled: boolean; intensity: number; threshold: number; radius: number };
   497→   497→  ao: { enabled: boolean; radius: number; intensity: number };
   498→   498→  vignette: { enabled: boolean; intensity: number };
   499→   499→  colorGrading: { enabled: boolean; brightness: number; contrast: number; saturation: number };
   500→   500→  antialiasing: 'none' | 'fxaa' | 'smaa' | 'taa';
   501→   501→  shadowQuality: 'none' | 'low' | 'medium' | 'high';
   502→   502→  quality: 'low' | 'medium' | 'high' | 'ultra';
   503→   503→  tonemapping: 'aces' | 'reinhard' | 'linear';
   504→   504→  exposure: number;
   505→   505→}
   506→   506→
   507→   507→export class RenderPipeline {
   508→   508→  private _sm: SceneManager;
   509→   509→  private _composer: EffectComposer | null = null;
   510→   510→  private _renderPass: RenderPass | null = null;
   511→   511→  private _bloomPass: UnrealBloomPass | null = null;
   512→   512→  private _smaaPass: SMAAPass | null = null;
   513→   513→  private _fxaaPass: ShaderPass | null = null;
   514→   514→  private _ssaoPass: SSAOPass | null = null;
   515→   515→  private _vignettePass: ShaderPass | null = null;
   516→   516→  private _colorGradingPass: ShaderPass | null = null;
   517→   517→  private _config: PipelineConfig;
   518→   518→  private _needsRebuild = false;
   519→   519→
   520→   520→  constructor(sceneManager: SceneManager) {
   521→   521→    this._sm = sceneManager;
   522→   522→    this._config = {
   523→   523→      bloom: { enabled: true, intensity: 0.3, threshold: 0.8, radius: 0.5 },
   524→   524→      ao: { enabled: false, radius: 0.5, intensity: 0.5 },
   525→   525→      vignette: { enabled: false, intensity: 0.4 },
   526→   526→      colorGrading: { enabled: false, brightness: 0, contrast: 0, saturation: 0 },
   527→   527→      antialiasing: 'smaa',
   528→   528→      shadowQuality: 'high',
   529→   529→      quality: 'high',
   530→   530→      tonemapping: 'aces',
   531→   531→      exposure: 1.0,
   532→   532→    };
   533→   533→  }
   534→   534→
   535→   535→  /** Build (or rebuild) the EffectComposer from scratch. */
   536→   536→  build(): void {
   537→   537→    this.dispose();
   538→   538→    const r = this._sm.renderer;
   539→   539→    const w = r.domElement.width || r.getSize(new THREE.Vector2()).x || 2;
   540→   540→    const h = r.domElement.height || r.getSize(new THREE.Vector2()).y || 2;
   541→   541→
   542→   542→    this._composer = new EffectComposer(r);
   543→   543→    this._renderPass = new RenderPass(this._sm.scene, this._sm.camera);
   544→   544→    this._composer.addPass(this._renderPass);
   545→   545→
   546→   546→    // Bloom
   547→   547→    if (this._config.bloom.enabled) {
   548→   548→      this._bloomPass = new UnrealBloomPass(
   549→   549→        new THREE.Vector2(w / 2, h / 2),
   550→   550→        this._config.bloom.intensity,
   551→   551→        this._config.bloom.radius,
   552→   552→        this._config.bloom.threshold,
   553→   553→      );
   554→   554→      this._composer.addPass(this._bloomPass);
   555→   555→    }
   556→   556→
   557→   557→    // Anti-aliasing
   558→   558→    if (this._config.antialiasing === 'smaa') {
   559→   559→      // @ts-expect-error — SMAAPass types are incomplete in @types/three
   560→   560→      this._smaaPass = new SMAAPass(w, h);
   561→   561→      this._composer.addPass(this._smaaPass);
   562→   562→    } else if (this._config.antialiasing === 'fxaa') {
   563→   563→      const fxaaMat = new THREE.ShaderMaterial({ ...FXAAShader, uniforms: { ...FXAAShader.uniforms, resolution: { value: new THREE.Vector2(1 / w, 1 / h) } } });
   564→   564→      this._fxaaPass = new ShaderPass(fxaaMat);
   565→   565→      this._composer.addPass(this._fxaaPass);
   566→   566→    }
   567→   567→    // 'taa' — handled via renderer settings + jitter; we keep SMAA as fallback
   568→   568→    // 'none' — no AA pass
   569→   569→
   570→   570→    // SSAO
   571→   571→    if (this._config.ao.enabled) {
   572→   572→      this._ssaoPass = new SSAOPass(
   573→   573→        this._sm.scene,
   574→   574→        this._sm.camera,
   575→   575→        w,
   576→   576→        h,
   577→   577→      );
   578→   578→      this._applyAOParams(this._config.ao.radius, this._config.ao.intensity);
   579→   579→      this._ssaoPass.output = SSAOPass.OUTPUT.Default;
   580→   580→      this._composer.addPass(this._ssaoPass);
   581→   581→    }
   582→   582→
   583→   583→    // Vignette
   584→   584→    if (this._config.vignette.enabled) {
   585→   585→      const vMat = new THREE.ShaderMaterial(VignetteShader);
   586→   586→      vMat.uniforms['intensity'].value = this._config.vignette.intensity;
   587→   587→      this._vignettePass = new ShaderPass(vMat);
   588→   588→      this._composer.addPass(this._vignettePass);
   589→   589→    }
   590→   590→
   591→   591→    // Color grading
   592→   592→    if (this._config.colorGrading.enabled) {
   593→   593→      const cgMat = new THREE.ShaderMaterial(ColorGradingShader);
   594→   594→      cgMat.uniforms['brightness'].value = this._config.colorGrading.brightness;
   595→   595→      cgMat.uniforms['contrast'].value = this._config.colorGrading.contrast;
   596→   596→      cgMat.uniforms['saturation'].value = this._config.colorGrading.saturation;
   597→   597→      this._colorGradingPass = new ShaderPass(cgMat);
   598→   598→      this._composer.addPass(this._colorGradingPass);
   599→   599→    }
   600→   600→
   601→   601→    // Apply tone mapping & exposure to the renderer
   602→   602→    this._applyToneMapping(this._config.tonemapping);
   603→   603→    this._sm.renderer.toneMappingExposure = this._config.exposure;
   604→   604→
   605→   605→    // Shadow quality
   606→   606→    this._applyShadowQuality(this._config.shadowQuality);
   607→   607→  }
   608→   608→
   609→   609→  /** Render one frame through the composer (or fallback direct render). */
   610→   610→  render(): void {
   611→   611→    if (this._composer) {
   612→   612→      this._composer.render();
   613→   613→    } else {
   614→   614→      this._sm.renderer.render(this._sm.scene, this._sm.camera);
   615→   615→    }
   616→   616→  }
   617→   617→
   618→   618→  /** Resize the composer and internal passes. */
   619→   619→  resize(): void {
   620→   620→    if (!this._composer) return;
   621→   621→    const r = this._sm.renderer;
   622→   622→    const size = r.getSize(new THREE.Vector2());
   623→   623→    const w = size.x || 2;
   624→   624→    const h = size.y || 2;
   625→   625→    this._composer.setSize(w, h);
   626→   626→
   627→   627→    if (this._bloomPass) {
   628→   628→      this._bloomPass.resolution.set(w / 2, h / 2);
   629→   629→    }
   630→   630→    if (this._ssaoPass) {
   631→   631→      this._ssaoPass.setSize(w, h);
   632→   632→    }
   633→   633→    if (this._smaaPass) {
   634→   634→      this._smaaPass.setSize(w, h);
   635→   635→    }
   636→   636→    if (this._fxaaPass) {
   637→   637→      const mat = this._fxaaPass.material as THREE.ShaderMaterial;
   638→   638→      const res = mat.uniforms['resolution'];
   639→   639→      if (res && res.value instanceof THREE.Vector2) {
   640→   640→        res.value.set(1 / w, 1 / h);
   641→   641→      }
   642→   642→    }
   643→   643→  }
   644→   644→
   645→   645→  /** Update the full pipeline config. Rebuilds composer if structural changes detected. */
   646→   646→  updateConfig(cfg: Partial<PipelineConfig>): void {
   647→   647→    const prev = { ...this._config };
   648→   648→    this._config = { ...this._config, ...cfg };
   649→   649→
   650→   650→    const structuralChange =
   651→   651→      cfg.bloom?.enabled !== undefined && cfg.bloom.enabled !== prev.bloom.enabled ||
   652→   652→      cfg.antialiasing !== undefined && cfg.antialiasing !== prev.antialiasing ||
   653→   653→      cfg.vignette?.enabled !== undefined && cfg.vignette.enabled !== prev.vignette.enabled ||
   654→   654→      cfg.ao?.enabled !== undefined && cfg.ao.enabled !== prev.ao.enabled ||
   655→   655→      cfg.colorGrading?.enabled !== undefined && cfg.colorGrading.enabled !== prev.colorGrading.enabled;
   656→   656→
   657→   657→    if (structuralChange) {
   658→   658→      this.build();
   659→   659→      return;
   660→   660→    }
   661→   661→
   662→   662→    // In-place parameter updates (no rebuild needed)
   663→   663→    if (this._bloomPass && cfg.bloom) {
   664→   664→      this._bloomPass.strength = cfg.bloom.intensity ?? this._bloomPass.strength;
   665→   665→      this._bloomPass.radius = cfg.bloom.radius ?? this._bloomPass.radius;
   666→   666→      this._bloomPass.threshold = cfg.bloom.threshold ?? this._bloomPass.threshold;
   667→   667→    }
   668→   668→
   669→   669→    if (this._ssaoPass && cfg.ao) {
   670→   670→      this._applyAOParams(
   671→   671→        cfg.ao.radius ?? this._config.ao.radius,
   672→   672→        cfg.ao.intensity ?? this._config.ao.intensity,
   673→   673→      );
   674→   674→    }
   675→   675→
   676→   676→    if (this._vignettePass && cfg.vignette) {
   677→   677→      const mat = this._vignettePass.material as THREE.ShaderMaterial;
   678→   678→      mat.uniforms['intensity'].value = cfg.vignette.intensity ?? 0.4;
   679→   679→    }
   680→   680→
   681→   681→    if (this._colorGradingPass && cfg.colorGrading) {
   682→   682→      const mat = this._colorGradingPass.material as THREE.ShaderMaterial;
   683→   683→      if (cfg.colorGrading.brightness !== undefined) mat.uniforms['brightness'].value = cfg.colorGrading.brightness;
   684→   684→      if (cfg.colorGrading.contrast !== undefined) mat.uniforms['contrast'].value = cfg.colorGrading.contrast;
   685→   685→      if (cfg.colorGrading.saturation !== undefined) mat.uniforms['saturation'].value = cfg.colorGrading.saturation;
   686→   686→    }
   687→   687→
   688→   688→    if (cfg.tonemapping !== undefined) {
   689→   689→      this._applyToneMapping(cfg.tonemapping);
   690→   690→    }
   691→   691→    if (cfg.exposure !== undefined) {
   692→   692→      this._sm.renderer.toneMappingExposure = cfg.exposure;
   693→   693→    }
   694→   694→    if (cfg.shadowQuality !== undefined) {
   695→   695→      this._applyShadowQuality(cfg.shadowQuality);
   696→   696→    }
   697→   697→    if (cfg.quality !== undefined) {
   698→   698→      this._applyQuality(cfg.quality);
   699→   699→    }
   700→   700→  }
   701→   701→
   702→   702→  // ── Convenience setters (called from Viewport sync) ───────────────────────
   703→   703→

   767→   767→  /** Dispose all passes and the composer. */
   768→   768→  dispose(): void {
   769→   769→    if (this._composer) {
   770→   770→      const passes = this._composer.passes.slice().reverse();
   771→   771→      for (const pass of passes) {
   772→   772→        if ('dispose' in pass && typeof (pass as { dispose: () => void }).dispose === 'function') {
   773→   773→          (pass as { dispose: () => void }).dispose();
   774→   774→        }
   775→   775→      }
   776→   776→      this._composer.dispose();
   777→   777→    }
   778→   778→    this._composer = null;
   779→   779→    this._renderPass = null;
   780→   780→    this._bloomPass = null;
   781→   781→    this._ssaoPass = null;
   782→   782→    this._smaaPass = null;
   783→   783→    this._fxaaPass = null;
   784→   784→    this._vignettePass = null;
   785→   785→    this._colorGradingPass = null;
   786→   786→  }

   788→   788→  // ── Private helpers ───────────────────────────────────────────────────────
   789→   789→
   790→   790→  /** Apply SSAO parameters — radius controls kernel spread, intensity controls maxDistance scaling. */
   791→   791→  private _applyAOParams(radius: number, intensity: number): void {
   792→   792→    if (!this._ssaoPass) return;
   793→   793→    this._ssaoPass.kernelRadius = radius;
   794→   794→    this._ssaoPass.minDistance = 0.005;
   795→   795→    // Scale maxDistance with intensity: higher intensity → AO visible at greater depth range
   796→   796→    // Tuned for car-scale scenes (objects ~2–5m across)
   797→   797→    this._ssaoPass.maxDistance = 0.05 + intensity * 0.5;
   798→   798→  }
   799→   799→
   800→   800→  private _applyToneMapping(mapping: 'aces' | 'reinhard' | 'linear'): void {
   801→   801→    const r = this._sm.renderer;
   802→   802→    switch (mapping) {
   803→   803→      case 'aces':
   804→   804→        r.toneMapping = THREE.ACESFilmicToneMapping;
   805→   805→        break;
   806→   806→      case 'reinhard':
   807→   807→        r.toneMapping = THREE.ReinhardToneMapping;
   808→   808→        break;
   809→   809→      case 'linear':
   810→   810→        r.toneMapping = THREE.LinearToneMapping;
   811→   811→        break;
   812→   812→    }
   813→   813→    r.toneMappingExposure = this._config.exposure;
   814→   814→  }
   815→   815→
   816→   816→  private _applyShadowQuality(quality: string): void {
   817→   817→    const r = this._sm.renderer;
   818→   818→    switch (quality) {
   819→   819→      case 'none':
   820→   820→        r.shadowMap.enabled = false;
   821→   821→        break;
   822→   822→      case 'low':
   823→   823→        r.shadowMap.enabled = true;
   824→   824→        r.shadowMap.type = THREE.BasicShadowMap;
   825→   825→        break;
   826→   826→      case 'medium':
   827→   827→        r.shadowMap.enabled = true;
   828→   828→        r.shadowMap.type = THREE.PCFShadowMap;
   829→   829→        break;
   830→   830→      case 'high':
   831→   831→      default:
   832→   832→        r.shadowMap.enabled = true;
   833→   833→        r.shadowMap.type = THREE.PCFSoftShadowMap;
   834→   834→        break;
   835→   835→    }
   836→   836→
   837→   837→    // Update shadow map sizes on all lights
   838→   838→    this._sm.scene.traverse((child) => {
   839→   839→      const light = child as THREE.DirectionalLight | THREE.SpotLight | THREE.PointLight;
   840→   840→      if (light.isLight && light.shadow) {
   841→   841→        const shadow = light.shadow;
   842→   842→        switch (quality) {
   843→   843→          case 'low':
   844→   844→            shadow.mapSize.set(512, 512);
   845→   845→            break;
   846→   846→          case 'medium':
   847→   847→            shadow.mapSize.set(1024, 1024);
   848→   848→            break;
   849→   849→          case 'high':
   850→   850→          default:
   851→   851→            shadow.mapSize.set(2048, 2048);
   852→   852→            break;
   853→   853→        }
   854→   854→        if (shadow.map) {
   855→   855→          shadow.map.dispose();
   856→   856→          shadow.map = null;
   857→   857→        }
   858→   858→      }
   859→   859→    });
   860→   860→  }
   861→   861→
   862→   862→  private _applyQuality(quality: 'low' | 'medium' | 'high' | 'ultra'): void {
   863→   863→    const r = this._sm.renderer;
   864→   864→    switch (quality) {
   865→   865→      case 'low':
   866→   866→        r.setPixelRatio(1);
   867→   867→        break;
   868→   868→      case 'medium':
   869→   869→        r.setPixelRatio(Math.min(window.devicePixelRatio, 1.5));
   870→   870→        break;
   871→   871→      case 'high':
   872→   872→        r.setPixelRatio(Math.min(window.devicePixelRatio, 2));
   873→   873→        break;
   874→   874→      case 'ultra':
   875→   875→        r.setPixelRatio(window.devicePixelRatio);
   876→   876→        break;
   877→   877→    }
   878→   878→    this._sm.resize();
   879→   879→    if (this._composer) {
   880→   880→      this.resize();
   881→   881→    }
   882→   882→  }
   883→   883→}

   914→   914→
   915→   915→export class LightManager {
   916→   916→  private _scene: THREE.Scene;
   917→   917→  private _entries: Map<string, LightObjectEntry> = new Map();
   918→   918→  private _helpers: Map<string, THREE.Object3D> = new Map();
   919→   919→  private _types: Map<string, string> = new Map();
   920→   920→
   921→   921→  constructor(scene: THREE.Scene) {
   922→   922→    this._scene = scene;
   923→   923→    ensureRectAreaLib();
   924→   924→  }
   925→   925→
   926→   926→  syncLights(lights: LightSyncEntry[]): void {
   927→   927→    const activeIds = new Set(lights.map((l) => l.id));
   928→   928→    const hasSolo = lights.some((l) => l.solo);
   929→   929→
   930→   930→    // Remove stale lights
   931→   931→    for (const [id] of this._entries) {
   932→   932→      if (!activeIds.has(id)) {
   933→   933→        this._removeLight(id);
   934→   934→      }
   935→   935→    }
   936→   936→
   937→   937→    // Create or update each light
   938→   938→    for (const ld of lights) {
   939→   939→      const isEffectiveSolo = hasSolo ? ld.solo : true;
   940→   940→      const shouldShow = ld.visible && isEffectiveSolo;
   941→   941→
   942→   942→      const existing = this._entries.get(ld.id);
   943→   943→      const currentType = this._types.get(ld.id);
   944→   944→
   945→   945→      // If type changed, recreate the light
   946→   946→      if (existing && currentType !== ld.type) {
   947→   947→        this._removeLight(ld.id);
   948→   948→      }
   949→   949→
   950→   950→      if (!this._entries.has(ld.id)) {
   951→   951→        this._createLight(ld);
   952→   952→      }
   953→   953→
   954→   954→      this._updateLight(ld, shouldShow);
   955→   955→    }
   956→   956→  }

   983→   983→  private _updateLight(ld: LightSyncEntry, shouldShow: boolean): void {
   984→   984→    const entry = this._entries.get(ld.id);
   985→   985→    if (!entry) return;
   986→   986→
   987→   987→    const { object: lightObj } = entry;
   988→   988→
   989→   989→    // Compute position from spherical
   990→   990→    const s = ld.transform.spherical;
   991→   991→    const latRad = (s.lat * Math.PI) / 180;
   992→   992→    const lngRad = (s.lng * Math.PI) / 180;
   993→   993→    const px = s.radius * Math.cos(latRad) * Math.cos(lngRad);
   994→   994→    const py = s.height;
   995→   995→    const pz = s.radius * Math.cos(latRad) * Math.sin(lngRad);
   996→   996→
   997→   997→    lightObj.position.set(px, py, pz);
   998→   998→    lightObj.visible = shouldShow;
   999→   999→
  1000→  1000→    // Apply rotation if enabled
  1001→  1001→    if (ld.transform.rotation.enabled) {
  1002→  1002→      lightObj.rotation.set(
  1003→  1003→        (ld.transform.rotation.x * Math.PI) / 180,
  1004→  1004→        (ld.transform.rotation.y * Math.PI) / 180,
  1005→  1005→        (ld.transform.rotation.z * Math.PI) / 180,
  1006→  1006→      );
  1007→  1007→    }
  1008→  1008→
  1009→  1009→    if (lightObj instanceof THREE.Light) {
  1010→  1010→      lightObj.color.set(ld.color);
  1011→  1011→      // Map brightness 0-1000 to intensity 0-10
  1012→  1012→      lightObj.intensity = (ld.brightness / 1000) * 10;
  1013→  1013→
  1014→  1014→      // Apply opacity (only meaningful via distance/decay for physically-based)
  1015→  1015→      // We scale intensity by opacity/100 to simulate dimming
  1016→  1016→      if (ld.opacity < 100) {
  1017→  1017→        lightObj.intensity *= ld.opacity / 100;
  1018→  1018→      }
  1019→  1019→
  1020→  1020→      // Falloff
  1021→  1021→      if (lightObj instanceof THREE.PointLight || lightObj instanceof THREE.SpotLight) {
  1022→  1022→        switch (ld.falloff) {
  1023→  1023→          case 'linear':
  1024→  1024→            lightObj.decay = 1;
  1025→  1025→            break;
  1026→  1026→          case 'none':
  1027→  1027→            lightObj.decay = 0;
  1028→  1028→            break;
  1029→  1029→          case 'custom':
  1030→  1030→            lightObj.decay = 1.5;
  1031→  1031→            break;
  1032→  1032→          case 'quadratic':
  1033→  1033→          default:
  1034→  1034→            lightObj.decay = 2;
  1035→  1035→            break;
  1036→  1036→        }
  1037→  1037→      }
  1038→  1038→
  1039→  1039→      // Spotlight-specific
  1040→  1040→      if (lightObj instanceof THREE.SpotLight) {
  1041→  1041→        if (ld.spotAngle !== undefined) {
  1042→  1042→          lightObj.angle = (ld.spotAngle * Math.PI) / 180;
  1043→  1043→        }
  1044→  1044→        if (ld.spotPenumbra !== undefined) {
  1045→  1045→          lightObj.penumbra = ld.spotPenumbra;
  1046→  1046→        }
  1047→  1047→        if (ld.spotDecay !== undefined) {
  1048→  1048→          lightObj.decay = ld.spotDecay;
  1049→  1049→        }
  1050→  1050→      }
  1051→  1051→
  1052→  1052→      // Area light dimensions
  1053→  1053→      if (lightObj instanceof THREE.RectAreaLight) {
  1054→  1054→        if (ld.areaWidth !== undefined) lightObj.width = ld.areaWidth;
  1055→  1055→        if (ld.areaHeight !== undefined) lightObj.height = ld.areaHeight;
  1056→  1056→      }
  1057→  1057→
  1058→  1058→      // Shadow config
  1059→  1059→      this._configureShadows(lightObj, ld.type);
  1060→  1060→    }
  1061→  1061→
  1062→  1062→    // Update target to look at origin for spot/directional
  1063→  1063→    if (entry.target) {
  1064→  1064→      entry.target.position.set(0, 0, 0);
  1065→  1065→    }
  1066→  1066→
  1067→  1067→    // Update helper
  1068→  1068→    const helper = this._helpers.get(ld.id);
  1069→  1069→    if (helper) {
  1070→  1070→      helper.visible = shouldShow && ld.gearVisible;
  1071→  1071→      helper.position.copy(lightObj.position);
  1072→  1072→      helper.rotation.copy(lightObj.rotation);
  1073→  1073→      if (helper instanceof THREE.Mesh && lightObj instanceof THREE.Light) {
  1074→  1074→        (helper.material as THREE.MeshBasicMaterial).color.copy(lightObj.color);
  1075→  1075→      }
  1076→  1076→    }
  1077→  1077→  }

  1190→  1190→  private _removeLight(id: string): void {

  1192→  1192→    if (!entry) return;
  1193→  1193→
  1194→  1194→    const { object, target } = entry;
  1195→  1195→
  1196→  1196→    // Dispose shadow maps

  1198→  1198→    if (object instanceof THREE.Light && shadowLight.shadow && shadowLight.shadow.map) {

  1201→  1201→
  1202→  1202→    this._scene.remove(object);
  1203→  1203→    this._disposeObject(object);

  1205→  1205→      this._scene.remove(target);
  1206→  1206→    }
  1207→  1207→
  1208→  1208→    this._entries.delete(id);
  1209→  1209→    this._types.delete(id);
  1210→  1210→
  1211→  1211→    const helper = this._helpers.get(id);
  1212→  1212→    if (helper) {
  1213→  1213→      this._scene.remove(helper);
  1214→  1214→      this._disposeObject(helper);
  1215→  1215→      this._helpers.delete(id);
  1216→  1216→    }
  1217→  1217→  }

  1224→  1224→          child.material.dispose();
  1225→  1225→        }
  1226→  1226→      }
  1227→  1227→    });
  1228→  1228→  }

  1349→  1349→  }
