    39→function getNodeType(obj: THREE.Object3D): HierarchyNode['type'] {
    40→  if (obj.userData.isLightHelper || obj.userData.isProxy || obj.userData.isGrid) return 'helper';
    41→  if (obj instanceof THREE.Light) return 'light';
    42→  if (obj instanceof THREE.Camera) return 'camera';
    43→  if (obj instanceof THREE.Mesh) return 'mesh';
    44→  if (obj instanceof THREE.Group && obj.children.length > 0) return 'group';
    45→  return 'other';
    46→}
    47→
    48→function isCollection(obj: THREE.Object3D): boolean {
    49→  return obj instanceof THREE.Group &&
    50→    !obj.userData.isGrid &&
    51→    !obj.userData.isProxy &&
    52→    obj.userData._isCollection === true;
    53→}
    54→
    55→function buildTree(scene: THREE.Scene): HierarchyNode[] {
    56→  const nodes: HierarchyNode[] = [];
    57→  for (const child of scene.children) {
    58→    if (child.userData.isGrid || child.userData.isProxy) continue;
    59→    nodes.push(buildNode(child));
    60→  }
    61→  return nodes;
    62→}
    63→
    64→function buildNode(obj: THREE.Object3D): HierarchyNode {
    65→  const children: HierarchyNode[] = [];
    66→  for (const child of obj.children) {
    67→    if (child.userData.isGrid || child.userData.isProxy) continue;
    68→    children.push(buildNode(child));
    69→  }
    70→  return {
    71→    name: obj.name || `${obj.type}_${obj.id}`,
    72→    type: getNodeType(obj),
    73→    object: obj,
    74→    children,
    75→    isCollection: isCollection(obj),
    76→  };
    77→}
    78→
    79→function hasDescendantMatch(node: HierarchyNode, query: string): boolean {
    80→  const q = query.toLowerCase();
    81→  if (node.name.toLowerCase().includes(q)) return true;
    82→  return node.children.some((c) => hasDescendantMatch(c, q));
    83→}

    85→function findObjectByUuid(scene: THREE.Scene, uuid: string): THREE.Object3D | null {
    86→  let result: THREE.Object3D | null = null;
    87→  scene.traverse((child) => {
    88→    if (child.uuid === uuid) result = child;
    89→  });
    90→  return result;
    91→}
    92→
    93→function getAllDescendantIds(obj: THREE.Object3D): string[] {
    94→  const ids: string[] = [obj.uuid];
    95→  obj.traverse((child) => {
    96→    if (child.uuid !== obj.uuid) ids.push(child.uuid);
    97→  });
    98→  return ids;
    99→}
   100→
   101→function getAllCollections(scene: THREE.Scene): THREE.Group[] {
   102→  const groups: THREE.Group[] = [];
   103→  scene.traverse((child) => {
   104→    if (isCollection(child)) groups.push(child as THREE.Group);
   105→  });
   106→  return groups;
   107→}
   108→
   109→function countCollections(scene: THREE.Scene): number {
   110→  let count = 0;
   111→  for (const child of scene.children) {
   112→    if (isCollection(child)) count++;
   113→  }
   114→  scene.traverse((child) => {
   115→    if (child !== scene && isCollection(child)) count++;
   116→  });
   117→  return count;
   118→}
   119→
   120→/** Check if targetId is a descendant of parentId (or the same object) */
   121→function isDescendantOf(parentObj: THREE.Object3D, checkObj: THREE.Object3D): boolean {
   122→  if (parentObj.uuid === checkObj.uuid) return true;
   123→  let found = false;
   124→  parentObj.traverse((child) => {
   125→    if (child.uuid === checkObj.uuid) found = true;
   126→  });
   127→  return found;
   128→}

   651→/* ═══════════════════════════════════════════════════════════════════
   652→   Tree Node (Blender-style with drag-and-drop support)
   653→   ═══════════════════════════════════════════════════════════════════ */
   654→
   655→const TreeNode: React.FC<{
   656→  node: HierarchyNode;
   657→  depth: number;
   658→  sceneRef: React.MutableRefObject<THREE.Scene | null>;
   659→  refresh: () => void;
   660→  onContextMenu: (e: React.MouseEvent, uuid: string) => void;
   661→  onDragStart: (uuid: string, name: string) => void;
   662→  onDragEnd: () => void;
   663→  onDropOnCollection: (collectionUuid: string) => void;
   664→  dropTargetId: string | null;
   665→}> = ({ node, depth, sceneRef, refresh, onContextMenu, onDragStart, onDragEnd, onDropOnCollection, dropTargetId }) => {
   666→  const selectedId = useSceneHierarchyStore((s) => s.selectedId);
   667→  const expandedIds = useSceneHierarchyStore((s) => s.expandedIds);
   668→  const hiddenIds = useSceneHierarchyStore((s) => s.hiddenIds);
   669→  const filterType = useSceneHierarchyStore((s) => s.filterType);
   670→  const searchQuery = useSceneHierarchyStore((s) => s.searchQuery);
   671→  const select = useSceneHierarchyStore((s) => s.select);
   672→  const toggleExpanded = useSceneHierarchyStore((s) => s.toggleExpanded);
   673→  const toggleVisibility = useSceneHierarchyStore((s) => s.toggleVisibility);
   674→
   675→  const [renaming, setRenaming] = useState(false);
   676→  const [renameValue, setRenameValue] = useState(node.object.name);
   677→  const renameRef = useRef<HTMLInputElement>(null);
   678→  const rowRef = useRef<HTMLDivElement>(null);
   679→
   680→  const hasChildren = node.children.length > 0;
   681→  const isSelected = selectedId === node.object.uuid;
   682→  const isHidden = hiddenIds.includes(node.object.uuid);
   683→  const isExpanded = expandedIds.includes(node.object.uuid);
   684→  const isColl = node.isCollection;
   685→  const isDropTarget = dropTargetId === node.object.uuid;
   686→
   687→  // Filter by type
   688→  if (filterType !== 'all') {
   689→    const typeMap: Record<string, string[]> = {
   690→      mesh: ['mesh'],
   691→      light: ['light'],
   692→      camera: ['camera'],
   693→      group: ['group'],
   694→    };
   695→    const allowed = typeMap[filterType] || [];
   696→    if (allowed.length > 0 && !allowed.includes(node.type)) {
   697→      // Collections (which have type='group') should always show in 'group' filter
   698→      const isGroupMatch = isColl && filterType === 'group';
   699→      if (!isGroupMatch) {
   700→        let hasMatch = false;
   701→        const checkDescendants = (n: HierarchyNode) => {
   702→          if (allowed.includes(n.type) || (n.isCollection && filterType === 'group')) hasMatch = true;
   703→          n.children.forEach(checkDescendants);
   704→        };
   705→        checkDescendants(node);
   706→        if (!hasMatch) return null;
   707→      }
   708→    }
   709→  }
   710→
   711→  // Filter by search
   712→  const matchesSearch = !searchQuery || node.name.toLowerCase().includes(searchQuery.toLowerCase());
   713→  const childMatchesSearch = searchQuery ? hasDescendantMatch(node, searchQuery) : false;
   714→  if (searchQuery && !matchesSearch && !childMatchesSearch) return null;
   715→
   716→  const handleSelect = useCallback(() => {
   717→    select(node.object.uuid);
   718→    // Material integration for mesh
   719→    if (node.type === 'mesh' && node.object instanceof THREE.Mesh) {
   720→      const matStore = useMaterialEditorStore.getState();
   721→      const mat = node.object.material;
   722→      if (mat) {
   723→        const matName = Array.isArray(mat) ? mat[0]?.name : (mat as THREE.Material).name;
   724→        if (matName) {
   725→          const match = matStore.materials.find((m) => m.name === matName);
   726→          if (match) {
   727→            matStore.selectMaterial(match.id);
   728→            useUIStore.getState().setRightPanelTab('matEdit');
   729→            useUIStore.getState().showPanel('rightPanel');
   730→          }
   731→        }
   732→      }
   733→    }
   734→  }, [node, select]);
   735→
   736→  const handleToggleVisibility = useCallback(
   737→    (e: React.MouseEvent) => {
   738→      e.stopPropagation();
   739→      toggleVisibility(node.object.uuid);
   740→      const newHidden = useSceneHierarchyStore.getState().hiddenIds.includes(node.object.uuid);
   741→      node.object.visible = !newHidden;
   742→      node.object.traverse((c) => {
   743→        c.visible = !newHidden;
   744→      });
   745→      refresh();
   746→    },
   747→    [node, toggleVisibility, refresh],
   748→  );
   749→
   750→  const handleContextMenu = useCallback(
   751→    (e: React.MouseEvent) => {
   752→      e.preventDefault();
   753→      e.stopPropagation();
   754→      onContextMenu(e, node.object.uuid);
   755→    },
   756→    [node.object.uuid, onContextMenu],
   757→  );
   758→
   759→  const handleDoubleClickName = useCallback(() => {
   760→    setRenaming(true);
   761→    setRenameValue(node.object.name);
   762→    setTimeout(() => renameRef.current?.select(), 0);
   763→  }, [node.object.name]);
   764→
   765→  const handleRenameCommit = useCallback(() => {
   766→    setRenaming(false);
   767→    if (renameValue.trim()) {
   768→      node.object.name = renameValue.trim();
   769→      refresh();
   770→    }
   771→  }, [renameValue, node.object, refresh]);
   772→
   773→  const handleRenameKeyDown = useCallback(
   774→    (e: React.KeyboardEvent) => {
   775→      if (e.key === 'Enter') handleRenameCommit();
   776→      if (e.key === 'Escape') {
   777→        setRenaming(false);
   778→        setRenameValue(node.object.name);
   779→      }
   780→    },
   781→    [handleRenameCommit, node.object.name],
   782→  );
   783→
   784→  // Drag handlers
   785→  const handleNodeDragStart = useCallback((e: React.DragEvent) => {
   786→    e.dataTransfer.effectAllowed = 'move';
   787→    e.dataTransfer.setData('text/plain', node.object.uuid);
   788→    onDragStart(node.object.uuid, node.name);
   789→    if (rowRef.current) {
   790→      rowRef.current.style.opacity = '0.4';
   791→    }
   792→  }, [node.object.uuid, node.name, onDragStart]);
   793→
   794→  const handleNodeDragEnd = useCallback(() => {
   795→    onDragEnd();
   796→    if (rowRef.current) {
   797→      rowRef.current.style.opacity = '1';
   798→    }
   799→  }, [onDragEnd]);
   800→
   801→  const handleNodeDragOver = useCallback((e: React.DragEvent) => {
   802→    // Only allow drop on collection nodes
   803→    if (!isColl) return;
   804→    e.preventDefault();
   805→    e.dataTransfer.dropEffect = 'move';
   806→  }, [isColl]);
   807→
   808→  const handleNodeDragEnter = useCallback((e: React.DragEvent) => {
   809→    if (!isColl) return;
   810→    e.preventDefault();
   811→    onDropOnCollection(node.object.uuid);
   812→  }, [isColl, node.object.uuid, onDropOnCollection]);
   813→
   814→  const handleNodeDragLeave = useCallback((e: React.DragEvent) => {
   815→    if (!isColl) return;
   816→    onDropOnCollection('');
   817→  }, [isColl, onDropOnCollection]);
   818→
   819→  const handleNodeDrop = useCallback((e: React.DragEvent) => {
   820→    e.preventDefault();
   821→    e.stopPropagation();
   822→    if (!isColl || !sceneRef.current) return;
   823→    const dragUuid = e.dataTransfer.getData('text/plain');
   824→    if (!dragUuid || dragUuid === node.object.uuid) {
   825→      onDropOnCollection('');
   826→      return;
   827→    }
   828→    const scene = sceneRef.current;
   829→    const dragObj = findObjectByUuid(scene, dragUuid);
   830→    if (!dragObj) {
   831→      onDropOnCollection('');
   832→      return;
   833→    }
   834→    // Prevent moving a collection into its own descendant
   835→    if (isDescendantOf(dragObj, node.object)) {
   836→      onDropOnCollection('');
   837→      return;
   838→    }
   839→    dragObj.parent?.remove(dragObj);
   840→    node.object.add(dragObj);
   841→    onDropOnCollection('');
   842→    refresh();
   843→  }, [isColl, node.object, sceneRef, onDropOnCollection, refresh]);
   844→
   845→  // Filter children
   846→  const filteredChildren = useMemo(() => {
   847→    return node.children.filter((child) => {
   848→      if (filterType !== 'all') {
   849→        const typeMap: Record<string, string[]> = {
   850→          mesh: ['mesh'],
   851→          light: ['light'],
   852→          camera: ['camera'],
   853→          group: ['group'],
   854→        };
   855→        const allowed = typeMap[filterType] || [];
   856→        if (allowed.length > 0) {
   857→          const isChildGroupMatch = child.isCollection && filterType === 'group';
   858→          if (allowed.includes(child.type) || isChildGroupMatch) return true;
   859→          let hasMatch = false;
   860→          const check = (n: HierarchyNode) => {
   861→            if (allowed.includes(n.type) || (n.isCollection && filterType === 'group')) hasMatch = true;
   862→            n.children.forEach(check);
   863→          };
   864→          check(child);
   865→          return hasMatch;
   866→        }
   867→      }
   868→      if (searchQuery) {
   869→        const q = searchQuery.toLowerCase();
   870→        return child.name.toLowerCase().includes(q) || hasDescendantMatch(child, searchQuery);
   871→      }
   872→      return true;
   873→    });
   874→  }, [node.children, filterType, searchQuery]);
   875→
   876→  return (
   877→    <div>
   878→      <div
   879→        ref={rowRef}
   880→        draggable={!renaming}
   881→        style={{
   882→          display: 'flex',
   883→          alignItems: 'center',
   884→          gap: 1,
   885→          paddingLeft: 2 + depth * 16,
   886→          paddingRight: 2,
   887→          height: 24,
   888→          cursor: 'pointer',
   889→          background: isDropTarget
   890→            ? 'rgba(251, 191, 36, 0.15)'
   891→            : isSelected
   892→              ? 'rgba(167, 139, 250, 0.15)'
   893→              : 'transparent',
   894→          borderLeft: isSelected ? '2px solid var(--accent)' : '2px solid transparent',
   895→          borderBottom: isDropTarget ? '1px solid rgba(251, 191, 36, 0.5)' : '1px solid transparent',
   896→          borderTop: isDropTarget ? '1px solid rgba(251, 191, 36, 0.5)' : '1px solid transparent',
   897→          transition: 'background 0.1s',
   898→          userSelect: 'none',
   899→        }}
   900→        onClick={handleSelect}
   901→        onContextMenu={handleContextMenu}
   902→        onDragStart={handleNodeDragStart}
   903→        onDragEnd={handleNodeDragEnd}
   904→        onDragOver={handleNodeDragOver}
   905→        onDragEnter={handleNodeDragEnter}
   906→        onDragLeave={handleNodeDragLeave}
   907→        onDrop={handleNodeDrop}
   908→        onMouseEnter={(e) => {
   909→          if (!isSelected && !isDropTarget) {
   910→            (e.currentTarget as HTMLElement).style.background = 'rgba(255,255,255,0.03)';
   911→          }
   912→        }}
   913→        onMouseLeave={(e) => {
   914→          if (!isSelected) {
   915→            (e.currentTarget as HTMLElement).style.background = 'transparent';
   916→          } else if (!isDropTarget) {
   917→            (e.currentTarget as HTMLElement).style.background = 'rgba(167, 139, 250, 0.15)';
   918→          }
   919→        }}
   920→      >
   921→        {/* Chevron */}
   922→        <span
   923→          style={{
   924→            width: 14,
   925→            height: 14,
   926→            display: 'flex',
   927→            alignItems: 'center',
   928→            justifyContent: 'center',
   929→            fontSize: 6,
   930→            color: 'var(--text-dim)',
   931→            visibility: hasChildren ? 'visible' : 'hidden',
   932→            transform: isExpanded ? 'rotate(90deg)' : 'rotate(0deg)',
   933→            transition: 'transform 0.15s',
   934→            flexShrink: 0,
   935→          }}
   936→          onClick={(e) => {
   937→            e.stopPropagation();
   938→            toggleExpanded(node.object.uuid);
   939→          }}
   940→        >
   941→          {'\u25B6'}
   942→        </span>
   943→
   944→        {/* Type icon */}
   945→        <TypeIcon type={node.type} isCollection={isColl} />
   946→
   947→        {/* Name / Rename input */}
   948→        {renaming ? (
   949→          <input
   950→            ref={renameRef}
   951→            value={renameValue}
   952→            onChange={(e) => setRenameValue(e.target.value)}
   953→            onBlur={handleRenameCommit}
   954→            onKeyDown={handleRenameKeyDown}
   955→            onClick={(e) => e.stopPropagation()}
   956→            style={{
   957→              flex: 1,
   958→              fontSize: 10,
   959→              background: 'var(--bg-input)',
   960→              border: '1px solid var(--accent)',
   961→              borderRadius: 2,
   962→              color: 'var(--text)',
   963→              padding: '0 3px',
   964→              outline: 'none',
   965→              minWidth: 0,
   966→              fontFamily: 'var(--font-ui)',
   967→              height: 16,
   968→            }}
   969→            autoFocus
   970→          />
   971→        ) : (
   972→          <span
   973→            style={{
   974→              flex: 1,
   975→              fontSize: 10,
   976→              color: isHidden ? 'var(--text-dim)' : (isColl ? '#fbbf24' : 'var(--text-sec)'),
   977→              opacity: isHidden ? 0.5 : 1,
   978→              overflow: 'hidden',
   979→              textOverflow: 'ellipsis',
   980→              whiteSpace: 'nowrap',
   981→              fontStyle: isHidden ? 'italic' : 'normal',
   982→              minWidth: 0,
   983→              fontWeight: isColl ? 500 : 400,
   984→              lineHeight: '24px',
   985→            }}
   986→            title={node.name}
   987→            onDoubleClick={(e) => {
   988→              e.stopPropagation();
   989→              handleDoubleClickName();
   990→            }}
   991→          >
   992→            {node.name}
   993→          </span>
   994→        )}
   995→
   996→        {/* ── Blender-style row action icons ── */}
   997→
   998→        {/* Eye (viewport visibility) */}
   999→        <RowIconButton
  1000→          active={!isHidden}
  1001→          title={isHidden ? 'Show in Viewport' : 'Hide in Viewport'}
  1002→          onClick={handleToggleVisibility}
  1003→        >
  1004→          {!isHidden ? (
  1005→            <svg width="12" height="12" viewBox="0 0 16 16" fill="none" stroke="currentColor" strokeWidth="1.2">
  1006→              <path d="M1 8s3-5 7-5 7 5 7 5-3 5-7 5-7-5-7-5z" />
  1007→              <circle cx="8" cy="8" r="2" />
  1008→            </svg>
  1009→          ) : (
  1010→            <svg width="12" height="12" viewBox="0 0 16 16" fill="none" stroke="currentColor" strokeWidth="1.2">
  1011→              <path d="M1 8s3-5 7-5 7 5 7 5-3 5-7 5-7-5-7-5z" />
  1012→              <line x1="2" y1="2" x2="14" y2="14" />
  1013→            </svg>
  1014→          )}
  1015→        </RowIconButton>
  1016→      </div>
  1017→
  1018→      {/* Children */}
  1019→      {hasChildren && isExpanded && (
  1020→        <div>
  1021→          {filteredChildren.map((child) => (
  1022→            <TreeNode
  1023→              key={child.object.uuid}
  1024→              node={child}
  1025→              depth={depth + 1}
  1026→              sceneRef={sceneRef}
  1027→              refresh={refresh}
  1028→              onContextMenu={onContextMenu}
  1029→              onDragStart={onDragStart}
  1030→              onDragEnd={onDragEnd}
  1031→              onDropOnCollection={onDropOnCollection}
  1032→              dropTargetId={dropTargetId}
  1033→            />
  1034→          ))}
  1035→        </div>
  1036→      )}
  1037→    </div>
  1038→  );
  1039→};

  1042→   Vec3 Input

  1044→
  1045→const AXIS_COLORS = ['#f87171', '#4ade80', '#60a5fa'];

  1047→
  1048→const Vec3Input: React.FC<{

  1051→  onChange: (idx: number, value: number) => void;

  1058→    {[0, 1, 2].map((idx) => (
