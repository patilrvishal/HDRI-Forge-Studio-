    39→function getNodeType(obj: THREE.Object3D): HierarchyNode['type'] {
    40→  if (obj.userData.isLightHelper || obj.userData.isProxy || obj.userData.isGrid) return 'helper';
    41→  if (obj instanceof THREE.Light) return 'light';
    42→  if (obj instanceof THREE.Camera) return 'camera';
    43→  if (obj instanceof THREE.Mesh) return 'mesh';
    44→  if (obj instanceof THREE.Group && obj.children.length > 0) return 'group';
    45→  return 'other';
    46→}
    47→
    48→function isCollection(obj: THREE.Object3D): boolean {
    49→  return obj instanceof THREE.Group &&
    50→    !obj.userData.isGrid &&
    51→    !obj.userData.isProxy &&
    52→    obj.userData._isCollection === true;
    53→}
    54→
    55→function buildTree(scene: THREE.Scene): HierarchyNode[] {
    56→  const nodes: HierarchyNode[] = [];
    57→  for (const child of scene.children) {
    58→    if (child.userData.isGrid || child.userData.isProxy) continue;
    59→    nodes.push(buildNode(child));
    60→  }
    61→  return nodes;
    62→}
    63→
    64→function buildNode(obj: THREE.Object3D): HierarchyNode {
    65→  const children: HierarchyNode[] = [];
    66→  for (const child of obj.children) {
    67→    if (child.userData.isGrid || child.userData.isProxy) continue;
    68→    children.push(buildNode(child));
    69→  }
    70→  return {
    71→    name: obj.name || `${obj.type}_${obj.id}`,
    72→    type: getNodeType(obj),
    73→    object: obj,
    74→    children,
    75→    isCollection: isCollection(obj),
    76→  };
    77→}
    78→
    79→function hasDescendantMatch(node: HierarchyNode, query: string): boolean {
    80→  const q = query.toLowerCase();
    81→  if (node.name.toLowerCase().includes(q)) return true;
    82→  return node.children.some((c) => hasDescendantMatch(c, q));
    83→}

    85→function findObjectByUuid(scene: THREE.Scene, uuid: string): THREE.Object3D | null {
    86→  let result: THREE.Object3D | null = null;
    87→  scene.traverse((child) => {
    88→    if (child.uuid === uuid) result = child;
    89→  });
    90→  return result;
    91→}
    92→
    93→function getAllDescendantIds(obj: THREE.Object3D): string[] {
    94→  const ids: string[] = [obj.uuid];
    95→  obj.traverse((child) => {
    96→    if (child.uuid !== obj.uuid) ids.push(child.uuid);
    97→  });
    98→  return ids;
    99→}
   100→
   101→function getAllCollections(scene: THREE.Scene): THREE.Group[] {
   102→  const groups: THREE.Group[] = [];
   103→  scene.traverse((child) => {
   104→    if (isCollection(child)) groups.push(child as THREE.Group);
   105→  });
   106→  return groups;
   107→}
   108→
   109→function countCollections(scene: THREE.Scene): number {
   110→  let count = 0;
   111→  for (const child of scene.children) {
   112→    if (isCollection(child)) count++;
   113→  }
   114→  scene.traverse((child) => {
   115→    if (child !== scene && isCollection(child)) count++;
   116→  });
   117→  return count;
   118→}
   119→
   120→/** Check if targetId is a descendant of parentId (or the same object) */
   121→function isDescendantOf(parentObj: THREE.Object3D, checkObj: THREE.Object3D): boolean {
   122→  if (parentObj.uuid === checkObj.uuid) return true;
   123→  let found = false;
   124→  parentObj.traverse((child) => {
   125→    if (child.uuid === checkObj.uuid) found = true;
   126→  });
   127→  return found;
   128→}

   651→/* ═══════════════════════════════════════════════════════════════════
   652→   Tree Node (Blender-style with drag-and-drop support)
   653→   ═══════════════════════════════════════════════════════════════════ */
   654→
   655→const TreeNode: React.FC<{
   656→  node: HierarchyNode;
   657→  depth: number;
   658→  sceneRef: React.MutableRefObject<THREE.Scene | null>;
   659→  refresh: () => void;
   660→  onContextMenu: (e: React.MouseEvent, uuid: string) => void;
   661→  onDragStart: (uuid: string, name: string) => void;
   662→  onDragEnd: () => void;
   663→  onDropOnCollection: (collectionUuid: string) => void;
   664→  dropTargetId: string | null;
   665→}> = ({ node, depth, sceneRef, refresh, onContextMenu, onDragStart, onDragEnd, onDropOnCollection, dropTargetId }) => {
   666→  const selectedId = useSceneHierarchyStore((s) => s.selectedId);
   667→  const expandedIds = useSceneHierarchyStore((s) => s.expandedIds);
   668→  const hiddenIds = useSceneHierarchyStore((s) => s.hiddenIds);
   669→  const filterType = useSceneHierarchyStore((s) => s.filterType);
   670→  const searchQuery = useSceneHierarchyStore((s) => s.searchQuery);
   671→  const select = useSceneHierarchyStore((s) => s.select);
   672→  const toggleExpanded = useSceneHierarchyStore((s) => s.toggleExpanded);
   673→  const toggleVisibility = useSceneHierarchyStore((s) => s.toggleVisibility);
   674→
   675→  const [renaming, setRenaming] = useState(false);
   676→  const [renameValue, setRenameValue] = useState(node.object.name);
   677→  const renameRef = useRef<HTMLInputElement>(null);
   678→  const rowRef = useRef<HTMLDivElement>(null);
   679→
   680→  const hasChildren = node.children.length > 0;
   681→  const isSelected = selectedId === node.object.uuid;
   682→  const isHidden = hiddenIds.includes(node.object.uuid);
   683→  const isExpanded = expandedIds.includes(node.object.uuid);
   684→  const isColl = node.isCollection;
   685→  const isDropTarget = dropTargetId === node.object.uuid;
   686→
   687→  // Filter by type
   688→  if (filterType !== 'all') {
   689→    const typeMap: Record<string, string[]> = {
   690→      mesh: ['mesh'],
   691→      light: ['light'],
   692→      camera: ['camera'],
   693→      group: ['group'],
   694→    };
   695→    const allowed = typeMap[filterType] || [];
   696→    if (allowed.length > 0 && !allowed.includes(node.type)) {
   697→      // Collections (which have type='group') should always show in 'group' filter
   698→      const isGroupMatch = isColl && filterType === 'group';
   699→      if (!isGroupMatch) {
   700→        let hasMatch = false;
   701→        const checkDescendants = (n: HierarchyNode) => {
   702→          if (allowed.includes(n.type) || (n.isCollection && filterType === 'group')) hasMatch = true;
   703→          n.children.forEach(checkDescendants);
   704→        };
   705→        checkDescendants(node);
   706→        if (!hasMatch) return null;
   707→      }
   708→    }
   709→  }
   710→
   711→  // Filter by search
   712→  const matchesSearch = !searchQuery || node.name.toLowerCase().includes(searchQuery.toLowerCase());
   713→  const childMatchesSearch = searchQuery ? hasDescendantMatch(node, searchQuery) : false;
   714→  if (searchQuery && !matchesSearch && !childMatchesSearch) return null;
   715→
   716→  const handleSelect = useCallback(() => {
   717→    select(node.object.uuid);
   718→    // Always show the right panel on Properties tab when selecting from hierarchy
   719→    useUIStore.getState().setRightPanelTab('properties');
   720→    useUIStore.getState().showPanel('rightPanel');
   721→
   722→    // Material integration for mesh — also switch to matEdit if a named material is found
   723→    if (node.type === 'mesh' && node.object instanceof THREE.Mesh) {
   724→      const matStore = useMaterialEditorStore.getState();
   725→      const mat = node.object.material;
   726→      if (mat) {
   727→        const matName = Array.isArray(mat) ? mat[0]?.name : (mat as THREE.Material).name;
   728→        if (matName) {
   729→          const match = matStore.materials.find((m) => m.name === matName);
   730→          if (match) {
   731→            matStore.selectMaterial(match.id);
   732→          }
   733→        }
   734→      }
   735→    }
   736→  }, [node, select]);
   737→
   738→  const handleToggleVisibility = useCallback(
   739→    (e: React.MouseEvent) => {
   740→      e.stopPropagation();
   741→      toggleVisibility(node.object.uuid);
   742→      const newHidden = useSceneHierarchyStore.getState().hiddenIds.includes(node.object.uuid);
   743→      node.object.visible = !newHidden;
   744→      node.object.traverse((c) => {
   745→        c.visible = !newHidden;
   746→      });
   747→      refresh();
   748→    },
   749→    [node, toggleVisibility, refresh],
   750→  );
   751→
   752→  const handleContextMenu = useCallback(
   753→    (e: React.MouseEvent) => {
   754→      e.preventDefault();
   755→      e.stopPropagation();
   756→      onContextMenu(e, node.object.uuid);
   757→    },
   758→    [node.object.uuid, onContextMenu],
   759→  );
   760→
   761→  const handleDoubleClickName = useCallback(() => {
   762→    setRenaming(true);
   763→    setRenameValue(node.object.name);
   764→    setTimeout(() => renameRef.current?.select(), 0);
   765→  }, [node.object.name]);
   766→
   767→  const handleRenameCommit = useCallback(() => {
   768→    setRenaming(false);
   769→    if (renameValue.trim()) {
   770→      node.object.name = renameValue.trim();
   771→      refresh();
   772→    }
   773→  }, [renameValue, node.object, refresh]);
   774→
   775→  const handleRenameKeyDown = useCallback(
   776→    (e: React.KeyboardEvent) => {
   777→      if (e.key === 'Enter') handleRenameCommit();
   778→      if (e.key === 'Escape') {
   779→        setRenaming(false);
   780→        setRenameValue(node.object.name);
   781→      }
   782→    },
   783→    [handleRenameCommit, node.object.name],
   784→  );
   785→
   786→  // Drag handlers
   787→  const handleNodeDragStart = useCallback((e: React.DragEvent) => {
   788→    e.dataTransfer.effectAllowed = 'move';
   789→    e.dataTransfer.setData('text/plain', node.object.uuid);
   790→    onDragStart(node.object.uuid, node.name);
   791→    if (rowRef.current) {
   792→      rowRef.current.style.opacity = '0.4';
   793→    }
   794→  }, [node.object.uuid, node.name, onDragStart]);
   795→
   796→  const handleNodeDragEnd = useCallback(() => {
   797→    onDragEnd();
   798→    if (rowRef.current) {
   799→      rowRef.current.style.opacity = '1';
   800→    }
   801→  }, [onDragEnd]);
   802→
   803→  const handleNodeDragOver = useCallback((e: React.DragEvent) => {
   804→    // Only allow drop on collection nodes
   805→    if (!isColl) return;
   806→    e.preventDefault();
   807→    e.dataTransfer.dropEffect = 'move';
   808→  }, [isColl]);
   809→
   810→  const handleNodeDragEnter = useCallback((e: React.DragEvent) => {
   811→    if (!isColl) return;
   812→    e.preventDefault();
   813→    onDropOnCollection(node.object.uuid);
   814→  }, [isColl, node.object.uuid, onDropOnCollection]);
   815→
   816→  const handleNodeDragLeave = useCallback((e: React.DragEvent) => {
   817→    if (!isColl) return;
   818→    onDropOnCollection('');
   819→  }, [isColl, onDropOnCollection]);
   820→
   821→  const handleNodeDrop = useCallback((e: React.DragEvent) => {
   822→    e.preventDefault();
   823→    e.stopPropagation();
   824→    if (!isColl || !sceneRef.current) return;
   825→    const dragUuid = e.dataTransfer.getData('text/plain');
   826→    if (!dragUuid || dragUuid === node.object.uuid) {
   827→      onDropOnCollection('');
   828→      return;
   829→    }
   830→    const scene = sceneRef.current;
   831→    const dragObj = findObjectByUuid(scene, dragUuid);
   832→    if (!dragObj) {
   833→      onDropOnCollection('');
   834→      return;
   835→    }
   836→    // Prevent moving a collection into its own descendant
   837→    if (isDescendantOf(dragObj, node.object)) {
   838→      onDropOnCollection('');
   839→      return;
   840→    }
   841→    dragObj.parent?.remove(dragObj);
   842→    node.object.add(dragObj);
   843→    onDropOnCollection('');
   844→    refresh();
   845→  }, [isColl, node.object, sceneRef, onDropOnCollection, refresh]);
   846→
   847→  // Filter children
   848→  const filteredChildren = useMemo(() => {
   849→    return node.children.filter((child) => {
   850→      if (filterType !== 'all') {
   851→        const typeMap: Record<string, string[]> = {
   852→          mesh: ['mesh'],
   853→          light: ['light'],
   854→          camera: ['camera'],
   855→          group: ['group'],
   856→        };
   857→        const allowed = typeMap[filterType] || [];
   858→        if (allowed.length > 0) {
   859→          const isChildGroupMatch = child.isCollection && filterType === 'group';
   860→          if (allowed.includes(child.type) || isChildGroupMatch) return true;
   861→          let hasMatch = false;
   862→          const check = (n: HierarchyNode) => {
   863→            if (allowed.includes(n.type) || (n.isCollection && filterType === 'group')) hasMatch = true;
   864→            n.children.forEach(check);
   865→          };
   866→          check(child);
   867→          return hasMatch;
   868→        }
   869→      }
   870→      if (searchQuery) {
   871→        const q = searchQuery.toLowerCase();
   872→        return child.name.toLowerCase().includes(q) || hasDescendantMatch(child, searchQuery);
   873→      }
   874→      return true;
   875→    });
   876→  }, [node.children, filterType, searchQuery]);
   877→
   878→  return (
   879→    <div>
   880→      <div
   881→        ref={rowRef}
   882→        draggable={!renaming}
   883→        style={{
   884→          display: 'flex',
   885→          alignItems: 'center',
   886→          gap: 1,
   887→          paddingLeft: 2 + depth * 16,
   888→          paddingRight: 2,
   889→          height: 24,
   890→          cursor: 'pointer',
   891→          background: isDropTarget
   892→            ? 'rgba(251, 191, 36, 0.15)'
   893→            : isSelected
   894→              ? 'rgba(167, 139, 250, 0.15)'
   895→              : 'transparent',
   896→          borderLeft: isSelected ? '2px solid var(--accent)' : '2px solid transparent',
   897→          borderBottom: isDropTarget ? '1px solid rgba(251, 191, 36, 0.5)' : '1px solid transparent',
   898→          borderTop: isDropTarget ? '1px solid rgba(251, 191, 36, 0.5)' : '1px solid transparent',
   899→          transition: 'background 0.1s',
   900→          userSelect: 'none',
   901→        }}
   902→        onClick={handleSelect}
   903→        onContextMenu={handleContextMenu}
   904→        onDragStart={handleNodeDragStart}
   905→        onDragEnd={handleNodeDragEnd}
   906→        onDragOver={handleNodeDragOver}
   907→        onDragEnter={handleNodeDragEnter}
   908→        onDragLeave={handleNodeDragLeave}
   909→        onDrop={handleNodeDrop}
   910→        onMouseEnter={(e) => {
   911→          if (!isSelected && !isDropTarget) {
   912→            (e.currentTarget as HTMLElement).style.background = 'rgba(255,255,255,0.03)';
   913→          }
   914→        }}
   915→        onMouseLeave={(e) => {
   916→          if (!isSelected) {
   917→            (e.currentTarget as HTMLElement).style.background = 'transparent';
   918→          } else if (!isDropTarget) {
   919→            (e.currentTarget as HTMLElement).style.background = 'rgba(167, 139, 250, 0.15)';
   920→          }
   921→        }}
   922→      >
   923→        {/* Chevron */}
   924→        <span
   925→          style={{
   926→            width: 14,
   927→            height: 14,
   928→            display: 'flex',
   929→            alignItems: 'center',
   930→            justifyContent: 'center',
   931→            fontSize: 6,
   932→            color: 'var(--text-dim)',
   933→            visibility: hasChildren ? 'visible' : 'hidden',
   934→            transform: isExpanded ? 'rotate(90deg)' : 'rotate(0deg)',
   935→            transition: 'transform 0.15s',
   936→            flexShrink: 0,
   937→          }}
   938→          onClick={(e) => {
   939→            e.stopPropagation();
   940→            toggleExpanded(node.object.uuid);
   941→          }}
   942→        >
   943→          {'\u25B6'}
   944→        </span>
   945→
   946→        {/* Type icon */}
   947→        <TypeIcon type={node.type} isCollection={isColl} />
   948→
   949→        {/* Name / Rename input */}
   950→        {renaming ? (
   951→          <input
   952→            ref={renameRef}
   953→            value={renameValue}
   954→            onChange={(e) => setRenameValue(e.target.value)}
   955→            onBlur={handleRenameCommit}
   956→            onKeyDown={handleRenameKeyDown}
   957→            onClick={(e) => e.stopPropagation()}
   958→            style={{
   959→              flex: 1,
   960→              fontSize: 10,
   961→              background: 'var(--bg-input)',
   962→              border: '1px solid var(--accent)',
   963→              borderRadius: 2,
   964→              color: 'var(--text)',
   965→              padding: '0 3px',
   966→              outline: 'none',
   967→              minWidth: 0,
   968→              fontFamily: 'var(--font-ui)',
   969→              height: 16,
   970→            }}
   971→            autoFocus
   972→          />
   973→        ) : (
   974→          <span
   975→            style={{
   976→              flex: 1,
   977→              fontSize: 10,
   978→              color: isHidden ? 'var(--text-dim)' : (isColl ? '#fbbf24' : 'var(--text-sec)'),
   979→              opacity: isHidden ? 0.5 : 1,
   980→              overflow: 'hidden',
   981→              textOverflow: 'ellipsis',
   982→              whiteSpace: 'nowrap',
   983→              fontStyle: isHidden ? 'italic' : 'normal',
   984→              minWidth: 0,
   985→              fontWeight: isColl ? 500 : 400,
   986→              lineHeight: '24px',
   987→            }}
   988→            title={node.name}
   989→            onDoubleClick={(e) => {
   990→              e.stopPropagation();
   991→              handleDoubleClickName();
   992→            }}
   993→          >
   994→            {node.name}
   995→          </span>
   996→        )}
   997→
   998→        {/* ── Blender-style row action icons ── */}
   999→
  1000→        {/* Eye (viewport visibility) */}
  1001→        <RowIconButton
  1002→          active={!isHidden}
  1003→          title={isHidden ? 'Show in Viewport' : 'Hide in Viewport'}
  1004→          onClick={handleToggleVisibility}
  1005→        >
  1006→          {!isHidden ? (
  1007→            <svg width="12" height="12" viewBox="0 0 16 16" fill="none" stroke="currentColor" strokeWidth="1.2">
  1008→              <path d="M1 8s3-5 7-5 7 5 7 5-3 5-7 5-7-5-7-5z" />
  1009→              <circle cx="8" cy="8" r="2" />
  1010→            </svg>
  1011→          ) : (
  1012→            <svg width="12" height="12" viewBox="0 0 16 16" fill="none" stroke="currentColor" strokeWidth="1.2">
  1013→              <path d="M1 8s3-5 7-5 7 5 7 5-3 5-7 5-7-5-7-5z" />
  1014→              <line x1="2" y1="2" x2="14" y2="14" />
  1015→            </svg>
  1016→          )}
  1017→        </RowIconButton>
  1018→      </div>
  1019→
  1020→      {/* Children */}
  1021→      {hasChildren && isExpanded && (
  1022→        <div>
  1023→          {filteredChildren.map((child) => (
  1024→            <TreeNode
  1025→              key={child.object.uuid}
  1026→              node={child}
  1027→              depth={depth + 1}
  1028→              sceneRef={sceneRef}
  1029→              refresh={refresh}
  1030→              onContextMenu={onContextMenu}
  1031→              onDragStart={onDragStart}
  1032→              onDragEnd={onDragEnd}
  1033→              onDropOnCollection={onDropOnCollection}
  1034→              dropTargetId={dropTargetId}
  1035→            />
  1036→          ))}
  1037→        </div>
  1038→      )}
  1039→    </div>
  1040→  );
  1041→};

  1044→   Vec3 Input
  1045→   ═══════════════════════════════════════════════════════════════════ */
  1046→
  1047→const AXIS_COLORS = ['#f87171', '#4ade80', '#60a5fa'];
  1048→const AXIS_LABELS = ['X', 'Y', 'Z'];
  1049→
  1050→const Vec3Input: React.FC<{
  1051→  label: string;
  1052→  values: [number, number, number];
  1053→  onChange: (idx: number, value: number) => void;
  1054→  decimals?: number;
  1055→}> = ({ label, values, onChange, decimals = 3 }) => (
  1056→  <div style={{ display: 'flex', alignItems: 'center', gap: 3 }}>
  1057→    <span style={{ fontSize: 9, color: 'var(--text-dim)', width: 40, flexShrink: 0, fontFamily: 'var(--font-mono)' }}>

  1059→    </span>
