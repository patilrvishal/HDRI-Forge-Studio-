    32→    32→    32→
    33→    33→    33→function getNodeType(obj: THREE.Object3D): HierarchyNode['type'] {
    34→    34→    34→  if (obj.userData.isLightHelper || obj.userData.isProxy || obj.userData.isGrid) return 'helper';
    35→    35→    35→  if (obj instanceof THREE.Light) return 'light';
    36→    36→    36→  if (obj instanceof THREE.Camera) return 'camera';
    37→    37→    37→  if (obj instanceof THREE.Mesh) return 'mesh';
    38→    38→    38→  if (obj instanceof THREE.Group && obj.children.length > 0) return 'group';
    39→    39→    39→  return 'other';
    40→    40→    40→}

    42→    42→    42→function buildTree(scene: THREE.Scene): HierarchyNode[] {
    43→    43→    43→  const nodes: HierarchyNode[] = [];
    44→    44→    44→  for (const child of scene.children) {
    45→    45→    45→    if (child.userData.isGrid || child.userData.isProxy) continue;
    46→    46→    46→    nodes.push(buildNode(child));
    47→    47→    47→  }
    48→    48→    48→  return nodes;
    49→    49→    49→}
    50→    50→    50→
    51→    51→    51→function buildNode(obj: THREE.Object3D): HierarchyNode {
    52→    52→    52→  const children: HierarchyNode[] = [];
    53→    53→    53→  for (const child of obj.children) {
    54→    54→    54→    if (child.userData.isGrid || child.userData.isProxy) continue;
    55→    55→    55→    children.push(buildNode(child));
    56→    56→    56→  }
    57→    57→    57→  return {
    58→    58→    58→    name: obj.name || `${obj.type}_${obj.id}`,
    59→    59→    59→    type: getNodeType(obj),
    60→    60→    60→    object: obj,
    61→    61→    61→    children,
    62→    62→    62→  };
    63→    63→    63→}

    65→    65→    65→function hasDescendantMatch(node: HierarchyNode, query: string): boolean {
    66→    66→    66→  const q = query.toLowerCase();
    67→    67→    67→  if (node.name.toLowerCase().includes(q)) return true;
    68→    68→    68→  return node.children.some((c) => hasDescendantMatch(c, q));
    69→    69→    69→}
