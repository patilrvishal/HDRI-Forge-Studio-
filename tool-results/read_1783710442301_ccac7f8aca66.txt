   119→   119→const MaterialEditorPanel: React.FC<MaterialEditorPanelProps> = ({ materialManagerRef, sceneRef }) => {
   120→   120→  const materials = useMaterialEditorStore((s) => s.materials);
   121→   121→  const selectedId = useMaterialEditorStore((s) => s.selectedMaterialId);
   122→   122→  const selectMaterial = useMaterialEditorStore((s) => s.selectMaterial);
   123→   123→  const updateMaterial = useMaterialEditorStore((s) => s.updateMaterial);
   124→   124→  const updateTextureSlot = useMaterialEditorStore((s) => s.updateTextureSlot);
   125→   125→  const removeTextureSlot = useMaterialEditorStore((s) => s.removeTextureSlot);
   126→   126→
   127→   127→  const [searchOpen, setSearchOpen] = useState(false);
   128→   128→  const [search, setSearch] = useState('');
   129→   129→  const [focusedNumInput, setFocusedNumInput] = useState<string | null>(null);
   130→   130→  const [collapsedSections, setCollapsedSections] = useState<Set<string>>(() => {
   131→   131→    const set = new Set<string>();
   132→   132→    for (const s of SECTIONS) {
   133→   133→      if (!s.defaultExpanded) set.add(s.name);
   134→   134→    }
   135→   135→    return set;
   136→   136→  });
   137→   137→  const textureInputRefs = useRef<Record<string, HTMLInputElement | null>>({});
   138→   138→  const searchInputRef = useRef<HTMLInputElement>(null);
   139→   139→
   140→   140→  const selected = materials.find((m) => m.id === selectedId) ?? null;
   141→   141→
   142→   142→  // Focus search input when opened
   143→   143→  useEffect(() => {
   144→   144→    if (searchOpen && searchInputRef.current) {
   145→   145→      searchInputRef.current.focus();
   146→   146→    }
   147→   147→  }, [searchOpen]);
   148→   148→
   149→   149→  // Filter materials by search
   150→   150→  const filteredMaterials = useMemo(() => {
   151→   151→    if (!search.trim()) return materials;
   152→   152→    const q = search.toLowerCase();
   153→   153→    return materials.filter(
   154→   154→      (m) => m.name.toLowerCase().includes(q) || m.meshNames.some((n) => n.toLowerCase().includes(q)),
   155→   155→    );
   156→   156→  }, [materials, search]);
   157→   157→
   158→   158→  // Physical keys that auto-set isPhysical
   159→   159→  const physicalKeys = useMemo(() => new Set<string>([
   160→   160→    'clearcoat', 'clearcoatRoughness', 'transmission', 'transmissionRoughness',
   161→   161→    'thickness', 'ior', 'sheen', 'sheenRoughness', 'sheenColor',
   162→   162→    'iridescence', 'iridescenceIOR', 'iridescenceThicknessRange',
   163→   163→    'attenuationColor', 'attenuationDistance', 'specularIntensity', 'specularColor',
   164→   164→  ]), []);
   165→   165→
   166→   166→  const handlePropChange = useCallback(
   167→   167→    (key: keyof PBRMaterialState, value: number | string | boolean, iridTupleIndex?: 0 | 1) => {
   168→   168→      if (!selectedId) return;
   169→   169→
   170→   170→      const updates: Partial<PBRMaterialState> = {};
   171→   171→
   172→   172→      // Handle iridescence thickness range tuple specially
   173→   173→      if (key === 'iridescenceThicknessRange' && iridTupleIndex !== undefined) {
   174→   174→        const cur = useMaterialEditorStore.getState().materials.find((m) => m.id === selectedId);
   175→   175→        const existing: [number, number] = cur?.iridescenceThicknessRange ?? [100, 400];
   176→   176→        if (iridTupleIndex === 0) {
   177→   177→          updates.iridescenceThicknessRange = [value as number, existing[1]];
   178→   178→        } else {
   179→   179→          updates.iridescenceThicknessRange = [existing[0], value as number];
   180→   180→        }
   181→   181→      } else {
   182→   182→        (updates as any)[key] = value;
   183→   183→      }
   184→   184→
   185→   185→      // Auto-set isPhysical for physical keys
   186→   186→      if (physicalKeys.has(key as string)) {
   187→   187→        updates.isPhysical = true;
   188→   188→      }
   189→   189→      // Auto-enable transparent when transmission > 0
   190→   190→      if (key === 'transmission' && typeof value === 'number' && value > 0) {
   191→   191→        updates.transparent = true;
   192→   192→      }
   193→   193→
   194→   194→      updateMaterial(selectedId, updates);
   195→   195→
   196→   196→      const mm = materialManagerRef.current;
   197→   197→      const scene = sceneRef.current;
   198→   198→      if (mm && scene) {
   199→   199→        const updated = useMaterialEditorStore.getState().materials.find((m) => m.id === selectedId);
   200→   200→        if (updated) mm.applyMaterialState(updated, scene);
   201→   201→      }
   202→   202→    },
   203→   203→    [selectedId, updateMaterial, materialManagerRef, sceneRef, physicalKeys],
   204→   204→  );
   205→   205→
   206→   206→  const handleTextureUpload = useCallback((slotKey: TextureSlotKey) => {
   207→   207→    textureInputRefs.current[slotKey]?.click();
   208→   208→  }, []);
   209→   209→
   210→   210→  const handleTextureFileChange = useCallback(
   211→   211→    (slotKey: TextureSlotKey, e: React.ChangeEvent<HTMLInputElement>) => {
   212→   212→      const file = e.target.files?.[0];
   213→   213→      if (!file || !selectedId) return;
   214→   214→
   215→   215→      const reader = new FileReader();
   216→   216→      reader.onload = () => {
   217→   217→        const dataUrl = reader.result as string;
   218→   218→        updateTextureSlot(selectedId, slotKey, dataUrl, file.name);
   219→   219→
   220→   220→        const mm = materialManagerRef.current;
   221→   221→        const scene = sceneRef.current;
   222→   222→        if (mm && scene) {
   223→   223→          const updated = useMaterialEditorStore.getState().materials.find((m) => m.id === selectedId);
   224→   224→          if (updated) mm.applyMaterialState(updated, scene);
   225→   225→        }
   226→   226→      };
   227→   227→      reader.readAsDataURL(file);
   228→   228→      e.target.value = '';
   229→   229→    },
   230→   230→    [selectedId, updateTextureSlot, materialManagerRef, sceneRef],
   231→   231→  );
   232→   232→
   233→   233→  const handleRemoveTexture = useCallback(
   234→   234→    (slotKey: TextureSlotKey) => {
   235→   235→      if (!selectedId) return;
   236→   236→      removeTextureSlot(selectedId, slotKey);
   237→   237→    },
   238→   238→    [selectedId, removeTextureSlot],
   239→   239→  );
   240→   240→
   241→   241→  const toggleSection = useCallback((section: string) => {
   242→   242→    setCollapsedSections((prev) => {
   243→   243→      const next = new Set(prev);
   244→   244→      if (next.has(section)) next.delete(section);
   245→   245→      else next.add(section);
   246→   246→      return next;
   247→   247→    });
   248→   248→  }, []);
   249→   249→
   250→   250→  const handleSearchSelect = useCallback(
   251→   251→    (id: string) => {

   253→   253→      setSearch('');
   254→   254→      setSearchOpen(false);
   255→   255→    },
   256→   256→    [selectMaterial],
   257→   257→  );
   258→   258→
   259→   259→  // Group props by section
   260→   260→  const propsBySection = useMemo(() => {
   261→   261→    const map: Record<string, PropRow[]> = {};
   262→   262→    for (const row of PROP_ROWS) {
   263→   263→      if (!map[row.section]) map[row.section] = [];
   264→   264→      map[row.section].push(row);
   265→   265→    }
   266→   266→    return map;
   267→   267→  }, []);
   268→   268→
   269→   269→  // Determine visible sections based on physical mode
   270→   270→  const visibleSections = useMemo(() => {
   271→   271→    if (!selected) return SECTIONS;
   272→   272→    const phys = selected.isPhysical;
   273→   273→    return SECTIONS.filter((s) => {
   274→   274→      if (s.name === 'Texture Maps') return true;
   275→   275→      if (s.physicalOnly) return phys;
   276→   276→      return true;
   277→   277→    });
   278→   278→  }, [selected]);
   279→   279→
   280→   280→  /** Format numeric value for display: 3 decimal places, ∞ for Infinity, integer for irid */
   281→   281→  const formatNumValue = useCallback((rawVal: number | undefined, row: PropRow): string => {
   282→   282→    if (rawVal == null || Number.isNaN(rawVal)) return '0.000';
   283→   283→    if (row.infinityKey && (rawVal === Infinity || rawVal >= 20)) return '\u221e';
   284→   284→    if (row.integerOnly) return Math.round(rawVal).toString();
   285→   285→    return rawVal.toFixed(3);
   286→   286→  }, []);
   287→   287→
   288→   288→  /** Get safe numeric value for slider */
   289→   289→  const getSafeNum = (rawVal: number | undefined, min: number): number => {

   291→   291→    return min;
   292→   292→  };
   293→   293→

   297→   297→      const v = parseFloat(inputValue);
   298→   298→      if (isNaN(v) || v == null) {
   299→   299→        handlePropChange(row.key, row.min, row.iridTupleIndex);
   300→   300→        return;
   301→   301→      }
   302→   302→      const clamped = Math.max(row.min, Math.min(row.max, v));
   303→   303→      handlePropChange(row.key, clamped, row.iridTupleIndex);
   304→   304→    },
   305→   305→    [handlePropChange],
   306→   306→  );
   307→   307→

   310→   310→    (row: PropRow, inputValue: string) => {

   312→   312→      if (!isNaN(v)) {
   313→   313→        handlePropChange(row.key, v, row.iridTupleIndex);
   314→   314→      }
   315→   315→    },
   316→   316→    [handlePropChange],
   317→   317→  );
   318→   318→

   346→   346→    // For iridescence thickness, extract the tuple element

   348→   348→    if (row.iridTupleIndex !== undefined) {
   349→   349→      rawVal = (selected.iridescenceThicknessRange?.[row.iridTupleIndex]) ?? row.min;
   350→   350→    } else {
   351→   351→      rawVal = selected[row.key] as number;
   352→   352→    }
   353→   353→
   354→   354→    const safeVal = getSafeNum(rawVal, row.min);
   355→   355→    const displayVal = formatNumValue(rawVal, row);
   356→   356→
   357→   357→    // For infinity key, slider value is clamped to 20
   358→   358→    const sliderVal = row.infinityKey && (rawVal === Infinity || rawVal >= 20) ? row.max : safeVal;
   359→   359→
   360→   360→    return (
   361→   361→      <div key={inputId} className="mat-param-row">
   362→   362→        <label className="mat-param-label">{row.label}</label>
   363→   363→        <input

   365→   365→          min={row.min}
   366→   366→          max={row.max}
   367→   367→          step={row.step}
   368→   368→          value={sliderVal}
   369→   369→          onChange={(e) => handlePropChange(row.key, parseFloat(e.target.value), row.iridTupleIndex)}

   371→   371→        />
   372→   372→        <input
   373→   373→          type="number"
   374→   374→          min={row.min}
   375→   375→          max={row.max}
   376→   376→          step={row.step}
   377→   377→          value={displayVal}
   378→   378→          onChange={(e) => handleNumChange(row, e.target.value)}
   379→   379→          onBlur={(e) => handleNumBlur(row, e.target.value)}
   380→   380→          onFocus={() => setFocusedNumInput(inputId)}
   381→   381→          onMouseEnter={(e) => {

   387→   387→            }
   388→   388→          }}
   389→   389→          style={focusedNumInput === inputId ? NUM_INPUT_FOCUS_STYLE : NUM_INPUT_STYLE}
   390→   390→        />
   391→   391→      </div>
   392→   392→    );
   393→   393→  };
   394→   394→

   407→   407→          style={{
   408→   408→            width: 32, height: 18, border: '1px solid var(--border)',

   410→   410→          }}
   411→   411→        />
   412→   412→        <span style={{
   413→   413→          fontSize: 9, color: 'var(--text-dim)', fontFamily: 'var(--font-mono)', flex: 1,

   415→   415→          {hexVal}

   436→   436→
   437→   437→  return (

   439→   439→      {/* ── Header with search toggle ──────────────────────────── */}

   453→   453→              letterSpacing: '0.3px',
   454→   454→            }}

   469→   469→          </svg>
   470→   470→        </button>
   471→   471→      </div>
   472→   472→
   473→   473→      {/* ── Search bar ─────────────────────────────────────────── */}

   475→   475→        <div style={{ padding: '4px 8px', borderBottom: '1px solid var(--border)' }}>
   476→   476→          <div style={{ position: 'relative' }}>

   480→   480→            >
   481→   481→              <circle cx="6.5" cy="6.5" r="5" />
   482→   482→              <path d="M10.5 10.5L15 15" />
   483→   483→            </svg>

   485→   485→              ref={searchInputRef}
   486→   486→              className="mat-search-input"
   487→   487→              type="text"
   488→   488→              placeholder="Search materials..."
   489→   489→              value={search}
   490→   490→              onChange={(e) => setSearch(e.target.value)}
   491→   491→            />
   492→   492→          </div>
   493→   493→          {search && (
   494→   494→            <div className="mat-search-results">
   495→   495→              {filteredMaterials.length === 0 && (
   496→   496→                <div style={{ padding: '8px 10px', fontSize: 10, color: 'var(--text-dim)' }}>

   500→   500→              {filteredMaterials.map((mat) => (
   501→   501→                <button
   502→   502→                  key={mat.id}

   511→   511→                    }}
   512→   512→                  />
   513→   513→                  <span style={{ flex: 1, overflow: 'hidden', textOverflow: 'ellipsis', whiteSpace: 'nowrap' }}>

   515→   515→                  </span>
   516→   516→                  <span style={{ fontSize: 9, color: 'var(--text-dim)', flexShrink: 0 }}>

   518→   518→                  </span>

   522→   522→          )}
   523→   523→        </div>
   524→   524→      )}
   525→   525→
   526→   526→      {/* ── Selected material chip ─────────────────────────────── */}

   529→   529→          <div
   530→   530→            style={{
   531→   531→              width: 12, height: 12, borderRadius: 2,

   533→   533→              border: '1px solid var(--border)',

   558→   558→          }}>

   566→   566→          </p>
   567→   567→        </div>
   568→   568→      )}
   569→   569→
   570→   570→      {/* ── Material properties ────────────────────────────────── */}
   571→   571→      {selected && (
   572→   572→        <div style={{ flex: 1, overflowY: 'auto', padding: '8px 10px' }}>
   573→   573→          {/* Material name + meshes */}
   574→   574→          <div style={{ marginBottom: 10 }}>
   575→   575→            <input
   576→   576→              type="text"
   577→   577→              value={selected.name}
   578→   578→              onChange={(e) => handlePropChange('name', e.target.value)}
   579→   579→              style={{
   580→   580→                fontSize: 11, fontWeight: 600, color: 'var(--text)',
   581→   581→                background: 'transparent', border: '1px solid transparent',
   582→   582→                borderRadius: 3, padding: '1px 4px', width: '100%',
   583→   583→                outline: 'none', fontFamily: 'inherit',
   584→   584→              }}
   585→   585→              onMouseEnter={(e) => { (e.currentTarget as HTMLInputElement).style.borderColor = 'var(--border)'; }}

   588→   588→            <div style={{ fontSize: 9, color: 'var(--text-dim)', marginTop: 1, paddingLeft: 4 }}>

   602→   602→                  {renderSectionHeader(sectionName, false)}

   605→   605→                      {TEXTURE_SLOTS.map((slotKey) => {

   609→   609→                            key={slotKey}
   610→   610→                            style={{
   611→   611→                              display: 'flex', alignItems: 'center', gap: 4,

   613→   613→                              background: slot.enabled ? 'var(--bg-card)' : 'transparent',
   614→   614→                              border: `1px solid ${slot.enabled ? 'var(--border-light)' : 'var(--border)'}`,
   615→   615→                            }}
   616→   616→                          >
   617→   617→                            <label className="mat-param-label" style={{ fontSize: 9 }}>

   621→   621→                              <>
   622→   622→                                <div
   623→   623→                                  style={{
   624→   624→                                    width: 22, height: 22, borderRadius: 3,

   629→   629→                                <span
   630→   630→                                  style={{
   631→   631→                                    flex: 1, fontSize: 8, color: 'var(--text-dim)',
   632→   632→                                    overflow: 'hidden', textOverflow: 'ellipsis', whiteSpace: 'nowrap',
   633→   633→                                  }}
   634→   634→                                  title={slot.fileName}
   635→   635→                                >
   636→   636→                                  {slot.fileName}
   637→   637→                                </span>
   638→   638→                                <button

   640→   640→                                  style={{
   641→   641→                                    fontSize: 8, color: 'var(--danger)', background: 'none',

   644→   644→                                >

   657→   657→                                    padding: '2px 6px', flexShrink: 0,
   658→   658→                                  }}
   659→   659→                                >
   660→   660→                                  + Load
   661→   661→                                </button>
   662→   662→                                <span style={{ fontSize: 8, color: 'var(--text-dim)' }}>None</span>
   663→   663→                              </>
   664→   664→                            )}
   665→   665→                            <input
   666→   666→                              ref={(el) => { textureInputRefs.current[slotKey] = el; }}
   667→   667→                              type="file" accept="image/*"
   668→   668→                              style={{ display: 'none' }}
   669→   669→                              onChange={(e) => handleTextureFileChange(slotKey, e)}
   670→   670→                            />

   672→   672→                        );
   673→   673→                      })}
   674→   674→                    </div>

   680→   680→            // ── Settings section (sliders + checkboxes) ────────
   681→   681→            if (sectionName === 'Settings') {

   694→   694→                      {/* Checkboxes */}

   700→   700→                          <input

   705→   705→                          />

   713→   713→
   714→   714→            // ── Standard property sections ────────────────────
   715→   715→            const rows = propsBySection[sectionName];

   717→   717→            const expanded = !collapsedSections.has(sectionName);
   718→   718→
   719→   719→            return (

   723→   723→                  <div style={{ display: 'flex', flexDirection: 'column', gap: 4, marginTop: 4 }}>
   724→   724→                    {rows.map((row) => {

   733→   733→        </div>
   734→   734→      )}

   736→   736→  );
