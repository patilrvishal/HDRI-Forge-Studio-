    39→function getNodeType(obj: THREE.Object3D): HierarchyNode['type'] {
    40→  if (obj.userData.isLightHelper || obj.userData.isProxy || obj.userData.isGrid) return 'helper';
    41→  if (obj instanceof THREE.Light) return 'light';
    42→  if (obj instanceof THREE.Camera) return 'camera';
    43→  if (obj instanceof THREE.Mesh) return 'mesh';
    44→  if (obj instanceof THREE.Group && obj.children.length > 0) return 'group';
    45→  return 'other';
    46→}
    47→
    48→function isCollection(obj: THREE.Object3D): boolean {
    49→  return obj instanceof THREE.Group &&
    50→    !obj.userData.isGrid &&
    51→    !obj.userData.isProxy &&
    52→    obj.userData._isCollection === true;
    53→}
    54→
    55→function buildTree(scene: THREE.Scene): HierarchyNode[] {
    56→  const nodes: HierarchyNode[] = [];
    57→  for (const child of scene.children) {
    58→    if (child.userData.isGrid || child.userData.isProxy) continue;
    59→    nodes.push(buildNode(child));
    60→  }
    61→  return nodes;
    62→}
    63→
    64→function buildNode(obj: THREE.Object3D): HierarchyNode {
    65→  const children: HierarchyNode[] = [];
    66→  for (const child of obj.children) {
    67→    if (child.userData.isGrid || child.userData.isProxy) continue;
    68→    children.push(buildNode(child));
    69→  }
    70→  return {
    71→    name: obj.name || `${obj.type}_${obj.id}`,
    72→    type: getNodeType(obj),
    73→    object: obj,
    74→    children,
    75→    isCollection: isCollection(obj),
    76→  };
    77→}
    78→
    79→function hasDescendantMatch(node: HierarchyNode, query: string): boolean {
    80→  const q = query.toLowerCase();
    81→  if (node.name.toLowerCase().includes(q)) return true;
    82→  return node.children.some((c) => hasDescendantMatch(c, q));
    83→}

    85→function findObjectByUuid(scene: THREE.Scene, uuid: string): THREE.Object3D | null {
    86→  let result: THREE.Object3D | null = null;
    87→  scene.traverse((child) => {
    88→    if (child.uuid === uuid) result = child;
    89→  });
    90→  return result;
    91→}
    92→
    93→function getAllDescendantIds(obj: THREE.Object3D): string[] {
    94→  const ids: string[] = [obj.uuid];
    95→  obj.traverse((child) => {
    96→    if (child.uuid !== obj.uuid) ids.push(child.uuid);
    97→  });
    98→  return ids;
    99→}
   100→
   101→function getAllCollections(scene: THREE.Scene): THREE.Group[] {
   102→  const groups: THREE.Group[] = [];
   103→  scene.traverse((child) => {
   104→    if (isCollection(child)) groups.push(child as THREE.Group);
   105→  });
   106→  return groups;
   107→}
   108→
   109→function countCollections(scene: THREE.Scene): number {
   110→  let count = 0;
   111→  for (const child of scene.children) {
   112→    if (isCollection(child)) count++;
   113→  }
   114→  scene.traverse((child) => {
   115→    if (child !== scene && isCollection(child)) count++;
   116→  });
   117→  return count;
   118→}
   119→
   120→/** Check if targetId is a descendant of parentId (or the same object) */
   121→function isDescendantOf(parentObj: THREE.Object3D, checkObj: THREE.Object3D): boolean {
   122→  if (parentObj.uuid === checkObj.uuid) return true;
   123→  let found = false;
   124→  parentObj.traverse((child) => {
   125→    if (child.uuid === checkObj.uuid) found = true;
   126→  });
   127→  return found;
   128→}

   651→/* ═══════════════════════════════════════════════════════════════════
   652→   Tree Node (Blender-style with drag-and-drop support)
   653→   ═══════════════════════════════════════════════════════════════════ */
   654→
   655→const TreeNode: React.FC<{
   656→  node: HierarchyNode;
   657→  depth: number;
   658→  sceneRef: React.MutableRefObject<THREE.Scene | null>;
   659→  refresh: () => void;
   660→  onContextMenu: (e: React.MouseEvent, uuid: string) => void;
   661→  onDragStart: (uuid: string, name: string) => void;
   662→  onDragEnd: () => void;
   663→  onDropOnCollection: (collectionUuid: string) => void;
   664→  dropTargetId: string | null;
   665→}> = ({ node, depth, sceneRef, refresh, onContextMenu, onDragStart, onDragEnd, onDropOnCollection, dropTargetId }) => {
   666→  const selectedId = useSceneHierarchyStore((s) => s.selectedId);
   667→  const expandedIds = useSceneHierarchyStore((s) => s.expandedIds);
   668→  const hiddenIds = useSceneHierarchyStore((s) => s.hiddenIds);
   669→  const filterType = useSceneHierarchyStore((s) => s.filterType);
   670→  const searchQuery = useSceneHierarchyStore((s) => s.searchQuery);
   671→  const select = useSceneHierarchyStore((s) => s.select);
   672→  const toggleExpanded = useSceneHierarchyStore((s) => s.toggleExpanded);
   673→  const toggleVisibility = useSceneHierarchyStore((s) => s.toggleVisibility);
   674→
   675→  const [renaming, setRenaming] = useState(false);
   676→  const [renameValue, setRenameValue] = useState(node.object.name);
   677→  const renameRef = useRef<HTMLInputElement>(null);
   678→  const rowRef = useRef<HTMLDivElement>(null);
   679→
   680→  // ── ALL hooks must be called before any conditional return (React Rules of Hooks) ──
   681→
   682→  const handleSelect = useCallback(() => {
   683→    select(node.object.uuid);
   684→    useUIStore.getState().setRightPanelTab('properties');
   685→    useUIStore.getState().showPanel('rightPanel');
   686→    if (node.type === 'mesh' && node.object instanceof THREE.Mesh) {
   687→      const matStore = useMaterialEditorStore.getState();
   688→      const mat = node.object.material;
   689→      if (mat) {
   690→        const matName = Array.isArray(mat) ? mat[0]?.name : (mat as THREE.Material).name;
   691→        if (matName) {
   692→          const match = matStore.materials.find((m) => m.name === matName);
   693→          if (match) {
   694→            matStore.selectMaterial(match.id);
   695→          }
   696→        }
   697→      }
   698→    }
   699→  }, [node, select]);
   700→
   701→  const handleToggleVisibility = useCallback(
   702→    (e: React.MouseEvent) => {
   703→      e.stopPropagation();
   704→      toggleVisibility(node.object.uuid);
   705→      const newHidden = useSceneHierarchyStore.getState().hiddenIds.includes(node.object.uuid);
   706→      node.object.visible = !newHidden;
   707→      node.object.traverse((c) => { c.visible = !newHidden; });
   708→      refresh();
   709→    },
   710→    [node, toggleVisibility, refresh],
   711→  );
   712→
   713→  const handleContextMenu = useCallback(
   714→    (e: React.MouseEvent) => {
   715→      e.preventDefault();
   716→      e.stopPropagation();
   717→      onContextMenu(e, node.object.uuid);
   718→    },
   719→    [node.object.uuid, onContextMenu],
   720→  );
   721→
   722→  const handleDoubleClickName = useCallback(() => {
   723→    setRenaming(true);
   724→    setRenameValue(node.object.name);
   725→    setTimeout(() => renameRef.current?.select(), 0);
   726→  }, [node.object.name]);
   727→
   728→  const handleRenameCommit = useCallback(() => {
   729→    setRenaming(false);
   730→    if (renameValue.trim()) {
   731→      node.object.name = renameValue.trim();
   732→      refresh();
   733→    }
   734→  }, [renameValue, node.object, refresh]);
   735→
   736→  const handleRenameKeyDown = useCallback(
   737→    (e: React.KeyboardEvent) => {
   738→      if (e.key === 'Enter') handleRenameCommit();
   739→      if (e.key === 'Escape') {
   740→        setRenaming(false);
   741→        setRenameValue(node.object.name);
   742→      }
   743→    },
   744→    [handleRenameCommit, node.object.name],
   745→  );
   746→
   747→  const handleNodeDragStart = useCallback((e: React.DragEvent) => {
   748→    e.dataTransfer.effectAllowed = 'move';
   749→    e.dataTransfer.setData('text/plain', node.object.uuid);
   750→    onDragStart(node.object.uuid, node.name);
   751→    if (rowRef.current) rowRef.current.style.opacity = '0.4';
   752→  }, [node.object.uuid, node.name, onDragStart]);
   753→
   754→  const handleNodeDragEnd = useCallback(() => {
   755→    onDragEnd();
   756→    if (rowRef.current) rowRef.current.style.opacity = '1';
   757→  }, [onDragEnd]);
   758→
   759→  const handleNodeDragOver = useCallback((e: React.DragEvent) => {
   760→    if (!node.isCollection) return;
   761→    e.preventDefault();
   762→    e.dataTransfer.dropEffect = 'move';
   763→  }, [node.isCollection]);
   764→
   765→  const handleNodeDragEnter = useCallback((e: React.DragEvent) => {
   766→    if (!node.isCollection) return;
   767→    e.preventDefault();
   768→    onDropOnCollection(node.object.uuid);
   769→  }, [node.isCollection, node.object.uuid, onDropOnCollection]);
   770→
   771→  const handleNodeDragLeave = useCallback((e: React.DragEvent) => {
   772→    if (!node.isCollection) return;
   773→    onDropOnCollection('');
   774→  }, [node.isCollection, onDropOnCollection]);
   775→
   776→  const handleNodeDrop = useCallback((e: React.DragEvent) => {
   777→    e.preventDefault();
   778→    e.stopPropagation();
   779→    if (!node.isCollection || !sceneRef.current) return;
   780→    const dragUuid = e.dataTransfer.getData('text/plain');
   781→    if (!dragUuid || dragUuid === node.object.uuid) { onDropOnCollection(''); return; }
   782→    const scene = sceneRef.current;
   783→    const dragObj = findObjectByUuid(scene, dragUuid);
   784→    if (!dragObj) { onDropOnCollection(''); return; }
   785→    if (isDescendantOf(dragObj, node.object)) { onDropOnCollection(''); return; }
   786→    dragObj.parent?.remove(dragObj);
   787→    node.object.add(dragObj);
   788→    onDropOnCollection('');
   789→    refresh();
   790→  }, [node.isCollection, node.object, sceneRef, onDropOnCollection, refresh]);
   791→
   792→  const filteredChildren = useMemo(() => {
   793→    return node.children.filter((child) => {
   794→      if (filterType !== 'all') {
   795→        const typeMap: Record<string, string[]> = {
   796→          mesh: ['mesh'],
   797→          light: ['light'],
   798→          camera: ['camera'],
   799→          group: ['group'],
   800→        };
   801→        const allowed = typeMap[filterType] || [];
   802→        if (allowed.length > 0) {
   803→          const isChildGroupMatch = child.isCollection && filterType === 'group';
   804→          if (allowed.includes(child.type) || isChildGroupMatch) return true;
   805→          let hasMatch = false;
   806→          const check = (n: HierarchyNode) => {
   807→            if (allowed.includes(n.type) || (n.isCollection && filterType === 'group')) hasMatch = true;
   808→            n.children.forEach(check);
   809→          };
   810→          check(child);
   811→          return hasMatch;
   812→        }
   813→      }
   814→      if (searchQuery) {
   815→        const q = searchQuery.toLowerCase();
   816→        return child.name.toLowerCase().includes(q) || hasDescendantMatch(child, searchQuery);
   817→      }
   818→      return true;
   819→    });
   820→  }, [node.children, filterType, searchQuery]);
   821→
   822→  // ── Computed values (after all hooks) ──
   823→
   824→  const hasChildren = node.children.length > 0;
   825→  const isSelected = selectedId === node.object.uuid;
   826→  const isHidden = hiddenIds.includes(node.object.uuid);
   827→  const isExpanded = expandedIds.includes(node.object.uuid);
   828→  const isColl = node.isCollection;
   829→  const isDropTarget = dropTargetId === node.object.uuid;
   830→
   831→  // Filter by type (early return — no hooks below this point)
   832→  if (filterType !== 'all') {
   833→    const typeMap: Record<string, string[]> = {
   834→      mesh: ['mesh'],
   835→      light: ['light'],
   836→      camera: ['camera'],
   837→      group: ['group'],
   838→    };
   839→    const allowed = typeMap[filterType] || [];
   840→    if (allowed.length > 0 && !allowed.includes(node.type)) {
   841→      const isGroupMatch = isColl && filterType === 'group';
   842→      if (!isGroupMatch) {
   843→        let hasMatch = false;
   844→        const checkDescendants = (n: HierarchyNode) => {
   845→          if (allowed.includes(n.type) || (n.isCollection && filterType === 'group')) hasMatch = true;
   846→          n.children.forEach(checkDescendants);
   847→        };
   848→        checkDescendants(node);
   849→        if (!hasMatch) return null;
   850→      }
   851→    }
   852→  }
   853→
   854→  // Filter by search (early return — no hooks below this point)
   855→  const matchesSearch = !searchQuery || node.name.toLowerCase().includes(searchQuery.toLowerCase());
   856→  const childMatchesSearch = searchQuery ? hasDescendantMatch(node, searchQuery) : false;
   857→  if (searchQuery && !matchesSearch && !childMatchesSearch) return null;
   858→
   859→  return (
   860→    <div>
   861→      <div
   862→        ref={rowRef}
   863→        draggable={!renaming}
   864→        style={{
   865→          display: 'flex',
   866→          alignItems: 'center',
   867→          gap: 1,
   868→          paddingLeft: 2 + depth * 16,
   869→          paddingRight: 2,
   870→          height: 24,
   871→          cursor: 'pointer',
   872→          background: isDropTarget
   873→            ? 'rgba(251, 191, 36, 0.15)'
   874→            : isSelected
   875→              ? 'rgba(167, 139, 250, 0.15)'
   876→              : 'transparent',
   877→          borderLeft: isSelected ? '2px solid var(--accent)' : '2px solid transparent',
   878→          borderBottom: isDropTarget ? '1px solid rgba(251, 191, 36, 0.5)' : '1px solid transparent',
   879→          borderTop: isDropTarget ? '1px solid rgba(251, 191, 36, 0.5)' : '1px solid transparent',
   880→          transition: 'background 0.1s',
   881→          userSelect: 'none',
   882→        }}
   883→        onClick={handleSelect}
   884→        onContextMenu={handleContextMenu}
   885→        onDragStart={handleNodeDragStart}
   886→        onDragEnd={handleNodeDragEnd}
   887→        onDragOver={handleNodeDragOver}
   888→        onDragEnter={handleNodeDragEnter}
   889→        onDragLeave={handleNodeDragLeave}
   890→        onDrop={handleNodeDrop}
   891→        onMouseEnter={(e) => {
   892→          if (!isSelected && !isDropTarget) {
   893→            (e.currentTarget as HTMLElement).style.background = 'rgba(255,255,255,0.03)';
   894→          }
   895→        }}
   896→        onMouseLeave={(e) => {
   897→          if (!isSelected) {
   898→            (e.currentTarget as HTMLElement).style.background = 'transparent';
   899→          } else if (!isDropTarget) {
   900→            (e.currentTarget as HTMLElement).style.background = 'rgba(167, 139, 250, 0.15)';
   901→          }
   902→        }}
   903→      >
   904→        {/* Chevron */}
   905→        <span
   906→          style={{
   907→            width: 14,
   908→            height: 14,
   909→            display: 'flex',
   910→            alignItems: 'center',
   911→            justifyContent: 'center',
   912→            fontSize: 6,
   913→            color: 'var(--text-dim)',
   914→            visibility: hasChildren ? 'visible' : 'hidden',
   915→            transform: isExpanded ? 'rotate(90deg)' : 'rotate(0deg)',
   916→            transition: 'transform 0.15s',
   917→            flexShrink: 0,
   918→          }}
   919→          onClick={(e) => {
   920→            e.stopPropagation();
   921→            toggleExpanded(node.object.uuid);
   922→          }}
   923→        >
   924→          {'\u25B6'}
   925→        </span>
   926→
   927→        {/* Type icon */}
   928→        <TypeIcon type={node.type} isCollection={isColl} />
   929→
   930→        {/* Name / Rename input */}
   931→        {renaming ? (
   932→          <input
   933→            ref={renameRef}
   934→            value={renameValue}
   935→            onChange={(e) => setRenameValue(e.target.value)}
   936→            onBlur={handleRenameCommit}
   937→            onKeyDown={handleRenameKeyDown}
   938→            onClick={(e) => e.stopPropagation()}
   939→            style={{
   940→              flex: 1,
   941→              fontSize: 10,
   942→              background: 'var(--bg-input)',
   943→              border: '1px solid var(--accent)',
   944→              borderRadius: 2,
   945→              color: 'var(--text)',
   946→              padding: '0 3px',
   947→              outline: 'none',
   948→              minWidth: 0,
   949→              fontFamily: 'var(--font-ui)',
   950→              height: 16,
   951→            }}
   952→            autoFocus
   953→          />
   954→        ) : (
   955→          <span
   956→            style={{
   957→              flex: 1,
   958→              fontSize: 10,
   959→              color: isHidden ? 'var(--text-dim)' : (isColl ? '#fbbf24' : 'var(--text-sec)'),
   960→              opacity: isHidden ? 0.5 : 1,
   961→              overflow: 'hidden',
   962→              textOverflow: 'ellipsis',
   963→              whiteSpace: 'nowrap',
   964→              fontStyle: isHidden ? 'italic' : 'normal',
   965→              minWidth: 0,
   966→              fontWeight: isColl ? 500 : 400,
   967→              lineHeight: '24px',
   968→            }}
   969→            title={node.name}
   970→            onDoubleClick={(e) => {
   971→              e.stopPropagation();
   972→              handleDoubleClickName();
   973→            }}
   974→          >
   975→            {node.name}
   976→          </span>
   977→        )}
   978→
   979→        {/* ── Blender-style row action icons ── */}
   980→
   981→        {/* Eye (viewport visibility) */}
   982→        <RowIconButton
   983→          active={!isHidden}
   984→          title={isHidden ? 'Show in Viewport' : 'Hide in Viewport'}
   985→          onClick={handleToggleVisibility}
   986→        >
   987→          {!isHidden ? (
   988→            <svg width="12" height="12" viewBox="0 0 16 16" fill="none" stroke="currentColor" strokeWidth="1.2">
   989→              <path d="M1 8s3-5 7-5 7 5 7 5-3 5-7 5-7-5-7-5z" />
   990→              <circle cx="8" cy="8" r="2" />
   991→            </svg>
   992→          ) : (
   993→            <svg width="12" height="12" viewBox="0 0 16 16" fill="none" stroke="currentColor" strokeWidth="1.2">
   994→              <path d="M1 8s3-5 7-5 7 5 7 5-3 5-7 5-7-5-7-5z" />
   995→              <line x1="2" y1="2" x2="14" y2="14" />
   996→            </svg>
   997→          )}
   998→        </RowIconButton>
   999→      </div>
  1000→
  1001→      {/* Children */}
  1002→      {hasChildren && isExpanded && (
  1003→        <div>
  1004→          {filteredChildren.map((child) => (
  1005→            <TreeNode
  1006→              key={child.object.uuid}
  1007→              node={child}
  1008→              depth={depth + 1}
  1009→              sceneRef={sceneRef}
  1010→              refresh={refresh}
  1011→              onContextMenu={onContextMenu}
  1012→              onDragStart={onDragStart}
  1013→              onDragEnd={onDragEnd}
  1014→              onDropOnCollection={onDropOnCollection}
  1015→              dropTargetId={dropTargetId}
  1016→            />
  1017→          ))}
  1018→        </div>
  1019→      )}
  1020→    </div>
  1021→  );
  1022→};

  1025→   Vec3 Input
  1026→   ═══════════════════════════════════════════════════════════════════ */
  1027→
  1028→const AXIS_COLORS = ['#f87171', '#4ade80', '#60a5fa'];
  1029→const AXIS_LABELS = ['X', 'Y', 'Z'];
  1030→
  1031→const Vec3Input: React.FC<{
  1032→  label: string;
  1033→  values: [number, number, number];
  1034→  onChange: (idx: number, value: number) => void;
  1035→  decimals?: number;
  1036→}> = ({ label, values, onChange, decimals = 3 }) => (

  1039→      {label}
  1040→    </span>
  1041→    {[0, 1, 2].map((idx) => (

  1044→          {AXIS_LABELS[idx]}
  1045→        </span>
  1046→        <input
  1047→          type="number"
  1048→          step="any"
