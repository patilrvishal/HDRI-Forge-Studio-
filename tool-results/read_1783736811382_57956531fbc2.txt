    33→    33→
    34→    34→function getNodeType(obj: THREE.Object3D): HierarchyNode['type'] {
    35→    35→  if (obj.userData.isLightHelper || obj.userData.isProxy || obj.userData.isGrid) return 'helper';
    36→    36→  if (obj instanceof THREE.Light) return 'light';
    37→    37→  if (obj instanceof THREE.Camera) return 'camera';
    38→    38→  if (obj instanceof THREE.Mesh) return 'mesh';
    39→    39→  if (obj instanceof THREE.Group && obj.children.length > 0) return 'group';
    40→    40→  return 'other';
    41→    41→}
    42→    42→
    43→    43→function isCollection(obj: THREE.Object3D): boolean {
    44→    44→  return obj instanceof THREE.Group &&
    45→    45→    !obj.userData.isGrid &&
    46→    46→    !obj.userData.isProxy &&
    47→    47→    obj.userData._isCollection === true;
    48→    48→}

    50→    50→function buildTree(scene: THREE.Scene): HierarchyNode[] {
    51→    51→  const nodes: HierarchyNode[] = [];
    52→    52→  for (const child of scene.children) {
    53→    53→    if (child.userData.isGrid || child.userData.isProxy) continue;
    54→    54→    nodes.push(buildNode(child));
    55→    55→  }
    56→    56→  return nodes;
    57→    57→}
    58→    58→
    59→    59→function buildNode(obj: THREE.Object3D): HierarchyNode {
    60→    60→  const children: HierarchyNode[] = [];
    61→    61→  for (const child of obj.children) {
    62→    62→    if (child.userData.isGrid || child.userData.isProxy) continue;
    63→    63→    children.push(buildNode(child));
    64→    64→  }
    65→    65→  return {
    66→    66→    name: obj.name || `${obj.type}_${obj.id}`,
    67→    67→    type: getNodeType(obj),
    68→    68→    object: obj,
    69→    69→    children,
    70→    70→    isCollection: isCollection(obj),
    71→    71→  };
    72→    72→}
    73→    73→
    74→    74→function hasDescendantMatch(node: HierarchyNode, query: string): boolean {
    75→    75→  const q = query.toLowerCase();
    76→    76→  if (node.name.toLowerCase().includes(q)) return true;
    77→    77→  return node.children.some((c) => hasDescendantMatch(c, q));
    78→    78→}

    80→    80→function findObjectByUuid(scene: THREE.Scene, uuid: string): THREE.Object3D | null {
    81→    81→  let result: THREE.Object3D | null = null;
    82→    82→  scene.traverse((child) => {
    83→    83→    if (child.uuid === uuid) result = child;
    84→    84→  });
    85→    85→  return result;
    86→    86→}

    88→    88→function getAllDescendantIds(obj: THREE.Object3D): string[] {
    89→    89→  const ids: string[] = [obj.uuid];
    90→    90→  obj.traverse((child) => {
    91→    91→    if (child.uuid !== obj.uuid) ids.push(child.uuid);
    92→    92→  });
    93→    93→  return ids;
    94→    94→}

    96→    96→function getAllCollections(scene: THREE.Scene): THREE.Group[] {
    97→    97→  const groups: THREE.Group[] = [];
    98→    98→  scene.traverse((child) => {
    99→    99→    if (isCollection(child)) groups.push(child as THREE.Group);
   100→   100→  });
   101→   101→  return groups;
   102→   102→}

   104→   104→function countCollections(scene: THREE.Scene): number {
   105→   105→  let count = 0;
   106→   106→  for (const child of scene.children) {
   107→   107→    if (isCollection(child)) count++;
   108→   108→  }
   109→   109→  scene.traverse((child) => {
   110→   110→    if (child !== scene && isCollection(child)) count++;
   111→   111→  });
   112→   112→  return count;
   113→   113→}
