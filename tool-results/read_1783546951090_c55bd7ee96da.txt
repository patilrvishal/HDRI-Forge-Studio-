     1→/**
     2→ * HDRIExporter — Captures the full 360° scene lighting as a standard equirectangular
     3→ * HDRI (.hdr) or EXR (.exr) file suitable for IBL in Blender, Photoshop, HDRView,
     4→ * or any HDR-capable application.
     5→ *
     6→ * The export combines:
     7→ *   - The scene's current environment (built-in presets re-created as emissive panels,
     8→ *     or custom .hdr files re-loaded from stored binary data)
     9→ *   - ALL physical lights (PointLight, SpotLight, RectAreaLight, DirectionalLight)
    10→ *     represented as bright emissive proxy meshes with HDR color values
    11→ *
    12→ * Process:
    13→ *   1. Build a dedicated capture scene with environment panels + light proxies
    14→ *   2. Render 6 cube faces via CubeCamera (HalfFloatType for Windows/Chrome compat)
    15→ *   3. Project cubemap → equirectangular via custom ShaderMaterial
    16→ *   4. Read back Uint16Array (half-float), convert to Float32Array with Y-flip
    17→ *   5. Encode as Radiance RGBE (.hdr) or OpenEXR (.exr)
    18→ *
    19→ * No external libraries — pure Three.js r165 + TypeScript.
    20→ */
    21→import * as THREE from 'three';
    22→import { getHDRIPresetById } from '../types/Environment';
    23→import type { HDRIPreset, EnvPanel } from '../types/Environment';
    24→import { getRawHDRIData } from '../store/hdriDataStore';
    25→import { RGBELoader } from 'three/addons/loaders/RGBELoader.js';

    27→// ─── IEEE 754 half-float ↔ float32 ──────────────────────────────────────────────
    28→
    29→/**
    30→ * Convert a 16-bit IEEE 754 half-float to a 32-bit JavaScript number.
    31→ * Handles signed zero, denormalized, normal, infinity, and NaN.
    32→ */
    33→export function halfToFloat(half: number): number {
    34→  const sign = (half >>> 15) & 0x1;
    35→  const exponent = (half >>> 10) & 0x1f;
    36→  const mantissa = half & 0x3ff;
    37→
    38→  if (exponent === 0) {
    39→    if (mantissa === 0) return sign === 0 ? 0 : -0;
    40→    const result = Math.pow(2, -14) * (mantissa / 1024);
    41→    return sign === 0 ? result : -result;
    42→  }
    43→  if (exponent === 31) {
    44→    if (mantissa === 0) return sign === 0 ? Infinity : -Infinity;
    45→    return NaN;
    46→  }
    47→  const result = Math.pow(2, exponent - 15) * (1 + mantissa / 1024);
    48→  return sign === 0 ? result : -result;
    49→}
    50→
    51→/** Convert a 32-bit float to 16-bit half-float (for EXR HALF encoding). */
    52→function floatToHalf(value: number): number {
    53→  const buf = new ArrayBuffer(4);
    54→  const f32 = new Float32Array(buf);
    55→  const u32 = new Uint32Array(buf);
    56→  f32[0] = value;
    57→  const x = u32[0];
    58→  const sign = (x >>> 16) & 0x8000;
    59→  const absX = x & 0x7fffffff;
    60→  if (absX > 0x477fe000) return sign | 0x7c00;
    61→  if (absX < 0x33000000) return sign;
    62→  if (absX < 0x38800000) {
    63→    const shift = 125 - ((absX >>> 23) & 0xff);
    64→    return sign | (((absX & 0x7fffff) | 0x800000) >>> shift);
    65→  }
    66→  const exponent = ((absX >>> 23) & 0xff) - 127 + 15;
    67→  const mantissa = (absX >>> 13) & 0x3ff;
    68→  return sign | (exponent << 10) | mantissa;
    69→}

    71→// ─── Scene capture ─────────────────────────────────────────────────────────────
    72→
    73→/**
    74→ * Captures the full 360° scene lighting (environment + all physical lights)
    75→ * into an equirectangular Float32Array (RGB, linear, top-to-bottom).
    76→ *
    77→ * Environment capture strategy:
    78→ *   - Built-in presets: Re-creates the original emissive panels from preset data.
    79→ *     These ARE the light sources that produce the environment lighting, so
    80→ *     capturing them directly gives the most accurate HDRI representation.
    81→ *   - Custom .hdr files: Re-loads the raw HDR binary data (stored in hdriDataStore),
    82→ *     creates an equirect DataTexture, and sets it as scene.background so the
    83→ *     CubeCamera renders it as the visible sky dome.
    84→ *
    85→ * Light proxy strategy:
    86→ *   Each THREE.Light (Point, Spot, Area, Directional) gets an emissive proxy mesh
    87→ *   with HDR-bright color values. The proxy brightness is scaled so that even
    88→ *   default-intensity lights (intensity≈1) produce values well above 1.0 in the
    89→ *   HDRI, ensuring the exposure slider in Photoshop/Blender reveals the light sources.
    90→ */
    91→export async function captureSceneToHDRI(
    92→  renderer: THREE.WebGLRenderer,
    93→  scene: THREE.Scene,
    94→  resolution: number,
    95→): Promise<Float32Array> {
    96→  // ── Save renderer state ──────────────────────────────────────────────────
    97→  const savedToneMapping = renderer.toneMapping;
    98→  const savedToneMappingExposure = renderer.toneMappingExposure;
    99→  const savedOutputColorSpace = renderer.outputColorSpace;
   100→
   101→  renderer.toneMapping = THREE.NoToneMapping;
   102→  renderer.toneMappingExposure = 1.0;
   103→  renderer.outputColorSpace = THREE.LinearSRGBColorSpace;
   104→
   105→  // ── Build a dedicated capture scene ──────────────────────────────────────
   106→  const captureScene = new THREE.Scene();
   107→  const disposables: { geometry?: THREE.BufferGeometry; material?: THREE.Material; texture?: THREE.Texture }[] = [];
   108→
   109→  // Track any Object URL created for custom HDRI (for cleanup)
   110→  let customHDRIObjURL: string | null = null;
   111→
   112→  try {
   113→    // ── 1) Environment: re-create the light sources that produce it ────────
   114→    await addEnvironmentToCaptureScene(captureScene, renderer, disposables);
   115→
   116→    // ── 2) Emissive proxies for every physical light in the scene ─────────
   117→    addLightProxies(scene, captureScene, disposables);
   118→
   119→    // ── 3) Capture cubemap at scene center (0, 0, 0) ────────────────────
   120→    const cubeSize = Math.max(512, resolution);
   121→    const cubeRT = new THREE.WebGLCubeRenderTarget(cubeSize, {
   122→      format: THREE.RGBAFormat,
   123→      type: THREE.HalfFloatType,
   124→      generateMipmaps: false,
   125→      minFilter: THREE.LinearFilter,
   126→      magFilter: THREE.LinearFilter,
   127→    });
   128→    const cubeCamera = new THREE.CubeCamera(0.01, 200, cubeRT);
   129→    cubeCamera.position.set(0, 0, 0);
   130→    cubeCamera.update(renderer, captureScene);
   131→
   132→    // ── 4) Cubemap → Equirectangular projection ────────────────────────────
   133→    const eqWidth = resolution;
   134→    const eqHeight = Math.floor(resolution / 2);
   135→
   136→    const eqRT = new THREE.WebGLRenderTarget(eqWidth, eqHeight, {
   137→      format: THREE.RGBAFormat,
   138→      type: THREE.HalfFloatType,
   139→      minFilter: THREE.LinearFilter,
   140→      magFilter: THREE.LinearFilter,
   141→      generateMipmaps: false,
   142→    });
   143→
   144→    const equirectMaterial = new THREE.ShaderMaterial({
   145→      uniforms: {
   146→        envMap: { value: cubeRT.texture },
   147→      },
   148→      vertexShader: `
   149→        varying vec2 vUv;
   150→        void main() {
   151→          vUv = uv;
   152→          gl_Position = vec4(position.xy, 0.0, 1.0);
   153→        }
   154→      `,
   155→      fragmentShader: `
   156→        uniform samplerCube envMap;
   157→        varying vec2 vUv;
   158→        #define PI 3.14159265359
   159→        #define TWO_PI 6.28318530718
   160→        void main() {
   161→          float phi = (1.0 - vUv.y) * PI;
   162→          float theta = vUv.x * TWO_PI;
   163→          vec3 dir = vec3(
   164→            sin(phi) * cos(theta),
   165→            cos(phi),
   166→            sin(phi) * sin(theta)
   167→          );
   168→          gl_FragColor = textureCube(envMap, dir);
   169→        }
   170→      `,
   171→      depthWrite: false,
   172→      depthTest: false,
   173→    });
   174→
   175→    const quad = new THREE.Mesh(new THREE.PlaneGeometry(2, 2), equirectMaterial);
   176→    const orthoCamera = new THREE.OrthographicCamera(-1, 1, 1, -1, 0, 1);
   177→    const renderScene = new THREE.Scene();
   178→    renderScene.add(quad);
   179→
   180→    const prevRT = renderer.getRenderTarget();
   181→    renderer.setRenderTarget(eqRT);
   182→    renderer.render(renderScene, orthoCamera);
   183→    renderer.setRenderTarget(prevRT);
   184→
   185→    // ── 5) Read pixels as Uint16Array (half-float) ────────────────────────
   186→    const halfPixels = new Uint16Array(eqWidth * eqHeight * 4);
   187→    renderer.readRenderTargetPixels(eqRT, 0, 0, eqWidth, eqHeight, halfPixels);
   188→
   189→    // ── 6) Convert half → float32 with Y-axis flip ────────────────────────
   190→    const floatPixels = new Float32Array(eqWidth * eqHeight * 3);
   191→    for (let row = 0; row < eqHeight; row++) {
   192→      const srcRow = eqHeight - 1 - row;
   193→      for (let col = 0; col < eqWidth; col++) {
   194→        const srcIdx = (srcRow * eqWidth + col) * 4;
   195→        const dstIdx = (row * eqWidth + col) * 3;
   196→        floatPixels[dstIdx]     = halfToFloat(halfPixels[srcIdx]);
   197→        floatPixels[dstIdx + 1] = halfToFloat(halfPixels[srcIdx + 1]);
   198→        floatPixels[dstIdx + 2] = halfToFloat(halfPixels[srcIdx + 2]);
   199→      }
   200→    }
   201→
   202→    // ── Cleanup GPU resources ──────────────────────────────────────────────
   203→    quad.geometry.dispose();
   204→    equirectMaterial.dispose();
   205→    eqRT.dispose();
   206→    cubeRT.dispose();
   207→
   208→    return floatPixels;
   209→
   210→  } finally {
   211→    // ── Restore renderer state ─────────────────────────────────────────────
   212→    renderer.toneMapping = savedToneMapping;
   213→    renderer.toneMappingExposure = savedToneMappingExposure;
   214→    renderer.outputColorSpace = savedOutputColorSpace;
   215→
   216→    // ── Dispose all temporary resources ────────────────────────────────────
   217→    for (const item of disposables) {
   218→      item.geometry?.dispose();
   219→      item.material?.dispose();
   220→      item.texture?.dispose();
   221→    }
   222→    if (customHDRIObjURL) {
   223→      URL.revokeObjectURL(customHDRIObjURL);
   224→    }
   225→  }
   226→}

   228→// ─── Environment capture helpers ────────────────────────────────────────────────
   229→
   230→/**
   231→ * Adds the environment light sources to the capture scene.
   232→ *
   233→ * For built-in presets: Re-creates the emissive panels that define the environment.
   234→ * For custom .hdr: Re-loads the raw file and sets it as scene.background.
   235→ * For 'none': No environment added.
   236→ */
   237→async function addEnvironmentToCaptureScene(
   238→  captureScene: THREE.Scene,
   239→  renderer: THREE.WebGLRenderer,
   240→  disposables: { geometry?: THREE.BufferGeometry; material?: THREE.Material; texture?: THREE.Texture }[],
   241→): Promise<void> {
   242→  // Lazy imports to avoid circular dependencies at module level
   243→  const { useSceneStore } = await import('../store/sceneStore');
   244→  const envState = useSceneStore.getState().environment;
   245→
   246→  // Handle environment rotation
   247→  const rotationDeg = envState.rotation ?? 0;
   248→
   249→  if (envState.presetId === 'none') {
   250→    // No environment — dark background only
   251→    captureScene.background = new THREE.Color(0x000000);
   252→    return;
   253→  }
   254→
   255→  if (envState.presetId === '__custom__') {
   256→    // Custom .hdr file: re-load from stored binary data
   257→    const rawBuffer = getRawHDRIData();
   258→    if (rawBuffer) {
   259→      try {
   260→        const texture = await loadHDRITexture(rawBuffer);
   261→        if (texture) {
   262→          // Use as scene.background — the CubeCamera renders this as the sky dome
   263→          texture.mapping = THREE.EquirectangularReflectionMapping;
   264→          captureScene.background = texture;
   265→          disposables.push({ texture });
   266→        }
   267→      } catch {
   268→        // Fallback: dark background if custom HDR can't be re-loaded
   269→        captureScene.background = new THREE.Color(0x000000);
   270→      }
   271→    } else {
   272→      captureScene.background = new THREE.Color(0x000000);
   273→    }
   274→    return;
   275→  }
   276→
   277→  // Built-in preset: re-create the emissive panels
   278→  const preset = getHDRIPresetById(envState.presetId);
   279→  if (!preset) {
   280→    captureScene.background = new THREE.Color(0x000000);
   281→    return;
   282→  }
   283→
   284→  // Add each emissive panel from the preset
   285→  // HDR_BOOST: environment panel intensities (0.3–1.2) are designed for PMREM PBR rendering,
   286→  // not for direct HDRI export. Multiply by 30 so panels produce values of 9–36 in linear
   287→  // HDR space — bright enough that exposure adjustment in Photoshop/Blender reveals them
   288→  // as distinct light sources while the black (0.0) background stays black.
   289→  const HDR_BOOST = 30;
   290→  for (const panel of preset.panels) {
   291→    const color = new THREE.Color(panel.color);
   292→    color.multiplyScalar(panel.intensity * HDR_BOOST);
   293→    const geo = new THREE.PlaneGeometry(panel.size[0], panel.size[1]);
   294→    const mat = new THREE.MeshBasicMaterial({
   295→      color,
   296→      side: THREE.DoubleSide,
   297→    });
   298→    const mesh = new THREE.Mesh(geo, mat);
   299→    mesh.position.set(panel.position[0], panel.position[1], panel.position[2]);
   300→    if (panel.rotationX !== undefined) mesh.rotation.x = panel.rotationX;
   301→    if (panel.rotationY !== undefined) mesh.rotation.y = panel.rotationY;
   302→    captureScene.add(mesh);
   303→    disposables.push({ geometry: geo, material: mat });
   304→  }
   305→
   306→  // Apply rotation to the entire environment (same as EnvironmentLoader does)
   307→  if (rotationDeg !== 0) {
   308→    captureScene.rotation.y = (rotationDeg * Math.PI) / 180;
   309→  }
   310→
   311→  // Pure black background — critical for proper HDRI dynamic range.
   312→  // backgroundHint is only for the viewport UI, NOT for HDRI data.
   313→  // Any non-zero floor (e.g. '#2a2a3e' ≈ 0.16) would cause the entire image
   314→  // to shift when adjusting exposure in Photoshop/Blender, instead of only
   315→  // the light source pixels responding. With 0.0 background, 0×exposure = 0
   316→  // (dark stays dark) while light panels (values 9–36) respond to exposure.
   317→  captureScene.background = new THREE.Color(0x000000);
   318→}

   320→/**
   321→ * Load a raw .hdr ArrayBuffer into a THREE.DataTexture via RGBELoader.
   322→ */
   323→function loadHDRITexture(buffer: ArrayBuffer): Promise<THREE.DataTexture | null> {
   324→  return new Promise<THREE.DataTexture | null>((resolve) => {
   325→    const blob = new Blob([buffer], { type: 'application/octet-stream' });
   326→    const url = URL.createObjectURL(blob);
   327→    const loader = new RGBELoader();
   328→    loader.load(
   329→      url,
   330→      (texture) => {
   331→        URL.revokeObjectURL(url);
   332→        resolve(texture);
   333→      },
   334→      undefined,
   335→      () => {
   336→        URL.revokeObjectURL(url);
   337→        resolve(null);
   338→      },
   339→    );
   340→  });
   341→}

   343→// ─── Light proxy creation ───────────────────────────────────────────────────────
   344→
   345→/**
   346→ * Traverses the main scene, finds all physical lights, and creates bright
   347→ * emissive proxy meshes in the capture scene for each one.
   348→ *
   349→ * Proxy brightness strategy:
   350→ *   Light intensity in Three.js r165 (physically correct) is in real-world units
   351→ *   (candela, lux, nits). For HDRI purposes, we need the proxy to produce values
   352→ *   well above 1.0 so the light source is clearly visible as an HDR hotspot.
   353→ *
   354→ *   Mapping: proxyColor = baseColor * max(20, intensity * 50)
   355→ *   - intensity 0.1  → color value ~20.0  (minimum HDR)
   356→ *   - intensity 1.0  → color value ~50.0  (clearly HDR, responds to exposure)
   357→ *   - intensity 5.0  → color value ~250.0 (very bright)
   358→ *   - intensity 50.0 → color value ~1000.0 (capped)
   359→ *
   360→ *   Cap at 1000 to avoid numerical issues in the HalfFloat render target.
   361→ *   With a pure black (0.0) background, these values create proper dynamic range
   362→ *   so exposure adjustment in Photoshop/Blender only affects light source pixels.
   363→ */
   364→function addLightProxies(
   365→  mainScene: THREE.Scene,
   366→  captureScene: THREE.Scene,
   367→  disposables: { geometry?: THREE.BufferGeometry; material?: THREE.Material; texture?: THREE.Texture }[],
   368→): void {
   369→  mainScene.traverse((child) => {
   370→    if (!(child instanceof THREE.Light)) return;
   371→
   372→    // Skip ambient/hemisphere — they are omnidirectional with no clear position
   373→    if (child instanceof THREE.AmbientLight || child instanceof THREE.HemisphereLight) {
   374→      return;
   375→    }
   376→
   377→    const light = child;
   378→    const intensity = light.intensity ?? 1;
   379→    if (intensity <= 0) return;
   380→
   381→    const baseColor = light.color ? light.color.clone() : new THREE.Color(1, 1, 1);
   382→
   383→    // Scale to HDR range: ensure even dim lights produce values well above 1.0
   384→    const hdrBrightness = Math.min(1000, Math.max(20, intensity * 50));
   385→    const emissiveColor = baseColor.clone().multiplyScalar(hdrBrightness);
   386→
   387→    let proxy: THREE.Mesh | null = null;
   388→
   389→    if (light instanceof THREE.DirectionalLight) {
   390→      // DirectionalLight → large emissive plane far away in the light's direction
   391→      const dir = new THREE.Vector3();
   392→      light.getWorldDirection(dir);
   393→      const planeSize = 40;
   394→      const geo = new THREE.PlaneGeometry(planeSize, planeSize);
   395→      const mat = new THREE.MeshBasicMaterial({
   396→        color: emissiveColor,
   397→        side: THREE.DoubleSide,
   398→      });
   399→      proxy = new THREE.Mesh(geo, mat);
   400→      const pos = dir.clone().multiplyScalar(48);
   401→      proxy.position.copy(pos);
   402→      proxy.lookAt(0, 0, 0);
   403→
   404→    } else if (light instanceof THREE.SpotLight) {
   405→      // SpotLight → emissive disc at the light's position, facing its direction
   406→      const pos = new THREE.Vector3();
   407→      light.getWorldPosition(pos);
   408→      const dir = new THREE.Vector3();
   409→      light.getWorldDirection(dir);
   410→      const angle = light.angle ?? Math.PI / 6;
   411→      const discRadius = Math.max(0.5, Math.tan(angle) * 4);
   412→      const geo = new THREE.CircleGeometry(discRadius, 32);
   413→      const mat = new THREE.MeshBasicMaterial({
   414→        color: emissiveColor,
   415→        side: THREE.DoubleSide,
   416→      });
   417→      proxy = new THREE.Mesh(geo, mat);
   418→      proxy.position.copy(pos);
   419→      proxy.lookAt(pos.clone().add(dir));
   420→
   421→    } else if (light instanceof THREE.RectAreaLight) {
   422→      // RectAreaLight → emissive rectangle matching the light's dimensions
   423→      const pos = new THREE.Vector3();
   424→      light.getWorldPosition(pos);
   425→      const w = light.width ?? 2;
   426→      const h = light.height ?? 2;
   427→      const geo = new THREE.PlaneGeometry(w, h);
   428→      const mat = new THREE.MeshBasicMaterial({
   429→        color: emissiveColor,
   430→        side: THREE.DoubleSide,
   431→      });
   432→      proxy = new THREE.Mesh(geo, mat);
   433→      proxy.position.copy(pos);
   434→      proxy.quaternion.copy(light.quaternion);
   435→
   436→    } else {
   437→      // PointLight (and IES, which is mapped to PointLight)
   438→      // → bright emissive sphere at the light's world position
   439→      const pos = new THREE.Vector3();
   440→      light.getWorldPosition(pos);
   441→      const radius = Math.max(0.3, Math.min(2.0, hdrBrightness * 0.003));
   442→      const geo = new THREE.SphereGeometry(radius, 16, 12);
   443→      const mat = new THREE.MeshBasicMaterial({ color: emissiveColor });
   444→      proxy = new THREE.Mesh(geo, mat);
   445→      proxy.position.copy(pos);
   446→    }
   447→
   448→    if (proxy) {
   449→      captureScene.add(proxy);
   450→      disposables.push({ geometry: proxy.geometry, material: proxy.material as THREE.Material });
   451→    }
   452→  });
   453→}

   455→// ─── Radiance HDR (RGBE) Encoding ──────────────────────────────────────────────
   456→
   457→/**
   458→ * Encode HDR pixel data into a Radiance RGBE .hdr file (ArrayBuffer).
   459→ *
   460→ * Uses uncompressed (flat) RGBE encoding for maximum compatibility.
   461→ *
   462→ * @param pixels - Float32Array of RGB (3 floats/pixel, top-to-bottom, linear).
   463→ * @param width  - Image width in pixels.
   464→ * @param height - Image height in pixels.
   465→ * @returns Complete .hdr file as ArrayBuffer.
   466→ */
   467→export function encodeHDR(
   468→  pixels: Float32Array,
   469→  width: number,
   470→  height: number,
   471→): ArrayBuffer {
   472→  const headerText =
   473→    '#?RADIANCE\n' +
   474→    'FORMAT=32-bit_rle_rgbe\n' +
   475→    'EXPOSURE=1.0\n' +
   476→    `\n-Y ${height} +X ${width}\n`;
   477→  const headerBytes = new TextEncoder().encode(headerText);
   478→
   479→  const pixelDataSize = width * height * 4;
   480→  const pixelData = new Uint8Array(pixelDataSize);
   481→
   482→  for (let i = 0; i < width * height; i++) {
   483→    const r = pixels[i * 3];
   484→    const g = pixels[i * 3 + 1];
   485→    const b = pixels[i * 3 + 2];
   486→    rgbFloatToRGBE(r, g, b, pixelData, i * 4);
   487→  }
   488→
   489→  const buffer = new ArrayBuffer(headerBytes.length + pixelDataSize);
   490→  const view = new Uint8Array(buffer);
   491→  view.set(headerBytes, 0);
   492→  view.set(pixelData, headerBytes.length);
   493→  return buffer;
   494→}
   495→
   496→/**
   497→ * Convert a single RGB float triplet to Radiance RGBE (4 bytes).
   498→ *
   499→ * RGBE encoding with shared exponent:
   500→ *   maxVal = max(r, g, b)
   501→ *   if maxVal < 1e-32: output [0,0,0,0]
   502→ *   else:
   503→ *     exp = ceil(log2(maxVal))
   504→ *     if 2^exp <= maxVal: exp++
   505→ *     scale = 2^(-exp) * 256
   506→ *     R = clamp(floor(r * scale), 0, 255)
   507→ *     G = clamp(floor(g * scale), 0, 255)
   508→ *     B = clamp(floor(b * scale), 0, 255)
   509→ *     E = exp + 128
   510→ */
   511→function rgbFloatToRGBE(
   512→  r: number,
   513→  g: number,
   514→  b: number,
   515→  out: Uint8Array,
   516→  off: number,
   517→): void {
   518→  const maxVal = Math.max(r, g, b);
   519→
   520→  if (maxVal < 1e-32) {
   521→    out[off] = 0;
   522→    out[off + 1] = 0;
   523→    out[off + 2] = 0;
   524→    out[off + 3] = 0;
   525→    return;
   526→  }
   527→
   528→  let exp = Math.ceil(Math.log2(maxVal));
   529→  if (Math.pow(2, exp) < maxVal) exp++;
   530→  if (Math.pow(2, exp) < maxVal) exp++;
   531→
   532→  const scale = Math.pow(2, -exp) * 256.0;
   533→  out[off]     = Math.max(0, Math.min(255, Math.floor(r * scale)));
   534→  out[off + 1] = Math.max(0, Math.min(255, Math.floor(g * scale)));
   535→  out[off + 2] = Math.max(0, Math.min(255, Math.floor(b * scale)));
   536→  out[off + 3] = Math.max(0, Math.min(255, exp + 128));
   537→}

   539→// ─── OpenEXR Encoding ──────────────────────────────────────────────────────────
   540→
   541→/**
   542→ * Encode HDR pixel data into an OpenEXR .exr file (ArrayBuffer).
   543→ *
   544→ * Channels: A, B, G, R as FLOAT (32-bit) — matches standard HDRI EXR convention
   545→ * (alphabetical channel order with alpha). Compression: NONE.
   546→ * Alpha is set to 1.0 (fully opaque) for all pixels.
   547→ *
   548→ * OpenEXR header format per attribute:
   549→ *   name\0  type\0  size(u32LE)  value(size bytes, padded to 4-byte boundary)
   550→ *
   551→ * @param pixels - Float32Array of RGB (3 floats/pixel, top-to-bottom, linear).
   552→ * @param width  - Image width in pixels.
   553→ * @param height - Image height in pixels.
   554→ * @returns Complete .exr file as ArrayBuffer.
   555→ */
   556→export function encodeEXR(
   557→  pixels: Float32Array,
   558→  width: number,
   559→  height: number,
   560→): ArrayBuffer {
   561→  const CHANNEL_NAMES = ['A', 'B', 'G', 'R'] as const;
   562→  const PIXEL_TYPE_FLOAT = 2;
   563→  const CHANNEL_BYTES = 4;
   564→  const NUM_CHANNELS = 4; // A, B, G, R
   565→  const BYTES_PER_PIXEL = NUM_CHANNELS * CHANNEL_BYTES;
   566→  const SCANLINE_DATA_SIZE = width * BYTES_PER_PIXEL;
   567→
   568→  // ── Low-level write helpers ──────────────────────────────────────────────
   569→  const writeU32LE = (arr: number[], val: number): void => {
   570→    arr.push(val & 0xff, (val >>> 8) & 0xff, (val >>> 16) & 0xff, (val >>> 24) & 0xff);
   571→  };
   572→  const writeI32LE = (arr: number[], val: number): void => writeU32LE(arr, val >>> 0);
   573→  const writeF32LE = (arr: number[], val: number): void => {
   574→    const buf = new ArrayBuffer(4);
   575→    new DataView(buf).setFloat32(0, val, true);
   576→    const bytes = new Uint8Array(buf);
   577→    arr.push(bytes[0], bytes[1], bytes[2], bytes[3]);
   578→  };
   579→
   580→  /** Convert a 32-bit integer to 4 bytes (little-endian). */
   581→  const intToBytes = (val: number): number[] => [
   582→    val & 0xff, (val >>> 8) & 0xff, (val >>> 16) & 0xff, (val >>> 24) & 0xff,
   583→  ];
   584→
   585→  /** Convert a 32-bit float to 4 bytes (little-endian). */
   586→  const floatToBytes = (val: number): number[] => {
   587→    const buf = new ArrayBuffer(4);
   588→    new DataView(buf).setFloat32(0, val, true);
   589→    return [...new Uint8Array(buf)];
   590→  };
   591→
   592→  /** Write a null-terminated string (for attribute names and type names). */
   593→  const writeName = (arr: number[], str: string): void => {
   594→    const bytes = new TextEncoder().encode(str);
   595→    for (const b of bytes) arr.push(b);
   596→    arr.push(0);
   597→  };
   598→
   599→  /**
   600→   * Write a channel-list entry: name\0 + padding to 4-byte boundary,
   601→   * then pixel_type(i32) + pLinear(u32) + x_sampling(u32) + y_sampling(u32).
   602→   * Each entry is exactly 4 + 16 = 20 bytes.
   603→   */
   604→  const writeChannelEntry = (arr: number[], name: string): void => {
   605→    const bytes = new TextEncoder().encode(name);
   606→    for (const b of bytes) arr.push(b);
   607→    arr.push(0); // null terminator
   608→    // Pad name to 4-byte boundary (including null)
   609→    const nameLen = bytes.length + 1;
   610→    const pad = (4 - (nameLen % 4)) % 4;
   611→    for (let i = 0; i < pad; i++) arr.push(0);
   612→    // Channel properties (16 bytes)
   613→    writeI32LE(arr, PIXEL_TYPE_FLOAT); // pixel type
   614→    writeU32LE(arr, 0);                // pLinear = false
   615→    writeU32LE(arr, 1);                // x sampling
   616→    writeU32LE(arr, 1);                // y sampling
   617→  };
   618→
   619→  /**
   620→   * Write an attribute value with proper size field and 4-byte padding.
   621→   * Format: size(u32LE) + value_bytes + padding_to_4_byte_boundary
   622→   */
   623→  const writeAttrValue = (arr: number[], valueBytes: number[]): void => {
   624→    writeU32LE(arr, valueBytes.length); // size field
   625→    for (const b of valueBytes) arr.push(b);
   626→    // Pad value to 4-byte boundary
   627→    const pad = (4 - (valueBytes.length % 4)) % 4;
   628→    for (let i = 0; i < pad; i++) arr.push(0);
   629→  };
   630→
   631→  // ── Build header attributes ──────────────────────────────────────────────
   632→  const hdr: number[] = [];
   633→
   634→  // 1) channels (chlist): 4 entries × 20 bytes + 1 terminator byte = 81 bytes
   635→  writeName(hdr, 'channels');
   636→  writeName(hdr, 'chlist');
   637→  const channelData: number[] = [];
   638→  for (const chName of CHANNEL_NAMES) {
   639→    writeChannelEntry(channelData, chName);
   640→  }
   641→  channelData.push(0); // channel list terminator
   642→  writeAttrValue(hdr, channelData);
   643→
   644→  // 2) compression: 1 byte (0 = NO_COMPRESSION)
   645→  writeName(hdr, 'compression');
   646→  writeName(hdr, 'compression');
   647→  writeAttrValue(hdr, [0]);
   648→
   649→  // 3) dataWindow (box2i): 4 × int32 = 16 bytes
   650→  writeName(hdr, 'dataWindow');
   651→  writeName(hdr, 'box2i');
   652→  writeAttrValue(hdr, [
   653→    ...intToBytes(0), ...intToBytes(0),
   654→    ...intToBytes(width - 1), ...intToBytes(height - 1),
   655→  ]);
   656→
   657→  // 4) displayWindow (box2i): 4 × int32 = 16 bytes
   658→  writeName(hdr, 'displayWindow');
   659→  writeName(hdr, 'box2i');
   660→  writeAttrValue(hdr, [
   661→    ...intToBytes(0), ...intToBytes(0),
   662→    ...intToBytes(width - 1), ...intToBytes(height - 1),
   663→  ]);
   664→
   665→  // 5) lineOrder: 1 byte (0 = INCREASING_Y)
   666→  writeName(hdr, 'lineOrder');
   667→  writeName(hdr, 'lineOrder');
   668→  writeAttrValue(hdr, [0]);
   669→
   670→  // 6) pixelAspectRatio (float): 4 bytes
   671→  writeName(hdr, 'pixelAspectRatio');
   672→  writeName(hdr, 'float');
   673→  writeAttrValue(hdr, floatToBytes(1.0));
   674→
   675→  // 7) screenWindowCenter (v2f): 8 bytes
   676→  writeName(hdr, 'screenWindowCenter');
   677→  writeName(hdr, 'v2f');
   678→  writeAttrValue(hdr, [...floatToBytes(0.0), ...floatToBytes(0.0)]);
   679→
   680→  // 8) screenWindowWidth (float): 4 bytes
   681→  writeName(hdr, 'screenWindowWidth');
   682→  writeName(hdr, 'float');
   683→  writeAttrValue(hdr, floatToBytes(1.0));
   684→
   685→  // End of header: empty name (single null byte), then pad to 8-byte boundary
   686→  hdr.push(0);
   687→  while (hdr.length % 8 !== 0) hdr.push(0);
   688→
   689→  const headerSize = hdr.length;
   690→  const fileHeaderSize = 8;
   691→  const offsetTableSize = height * 8;
   692→  const scanlineDataStart = fileHeaderSize + headerSize + offsetTableSize;
   693→  const scanlineBlockSize = 4 + 4 + SCANLINE_DATA_SIZE;
   694→
   695→  // ── Offset table ────────────────────────────────────────────────────────
   696→  const offsets: number[] = [];
   697→  for (let y = 0; y < height; y++) {
   698→    offsets.push(scanlineDataStart + y * scanlineBlockSize);
   699→  }
   700→
   701→  // ── Scanline pixel data ─────────────────────────────────────────────────
   702→  // Each scanline: [y_coord (u32)] [data_size (u32)] [A(u32) B(u32) G(u32) R(u32) per pixel]
   703→  const scanlines: number[] = [];
   704→  for (let y = 0; y < height; y++) {
   705→    writeU32LE(scanlines, y);
   706→    writeU32LE(scanlines, SCANLINE_DATA_SIZE);
   707→    for (let x = 0; x < width; x++) {
   708→      const srcIdx = (y * width + x) * 3;
   709→      writeF32LE(scanlines, 1.0);              // A = 1.0 (fully opaque)
   710→      writeF32LE(scanlines, pixels[srcIdx + 2]); // B
   711→      writeF32LE(scanlines, pixels[srcIdx + 1]); // G
   712→      writeF32LE(scanlines, pixels[srcIdx]);     // R
   713→    }
   714→  }
   715→
   716→  // ── Assemble file ───────────────────────────────────────────────────────
   717→  const totalSize = fileHeaderSize + headerSize + offsetTableSize + scanlines.length;
   718→  const file = new Uint8Array(totalSize);
   719→  const dv = new DataView(file.buffer);
   720→
   721→  dv.setUint32(0, 20000630, true);
   722→  dv.setUint32(4, 2, true);
   723→
   724→  file.set(new Uint8Array(hdr), 8);
   725→
   726→  const offsetBase = 8 + headerSize;
   727→  for (let y = 0; y < height; y++) {
   728→    const off = offsets[y];
   729→    dv.setUint32(offsetBase + y * 8, off & 0xffffffff, true);
   730→    dv.setUint32(offsetBase + y * 8 + 4, Math.floor(off / 0x100000000) & 0xffffffff, true);
   731→  }
   732→
   733→  file.set(new Uint8Array(scanlines), scanlineDataStart);
   734→
   735→  return file.buffer;
   736→}

   738→// ─── Download helper ───────────────────────────────────────────────────────────
   739→
   740→/**
   741→ * Download HDR pixel data as a .hdr or .exr file.
   742→ */
   743→export function downloadHDRI(
   744→  pixels: Float32Array,
   745→  width: number,
   746→  height: number,
   747→  format: 'hdr' | 'exr',
   748→  filename?: string,
   749→): void {
   750→  const baseName = filename ?? `hdri_export_${Date.now()}`;
   751→  const ext = format === 'hdr' ? '.hdr' : '.exr';
   752→  const mimeType = format === 'hdr' ? 'image/vnd.radiance' : 'image/x-exr';
   753→
   754→  const buffer: ArrayBuffer =
   755→    format === 'hdr'
   756→      ? encodeHDR(pixels, width, height)
   757→      : encodeEXR(pixels, width, height);
   758→
   759→  const blob = new Blob([buffer], { type: mimeType });
   760→  const url = URL.createObjectURL(blob);
   761→  const link = document.createElement('a');
   762→  link.href = url;
   763→  link.download = `${baseName}${ext}`;
   764→  link.style.display = 'none';
   765→  document.body.appendChild(link);
   766→  link.click();
   767→
   768→  setTimeout(() => {
   769→    URL.revokeObjectURL(url);
   770→    if (link.parentNode) link.parentNode.removeChild(link);
   771→  }, 250);
   772→}
